gef_fild10,242,202,0	script	次元峽谷的碎片	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"獸人的記憶";
	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "只有等級在 ^ff000030 ~ 80^000000 之間的玩家才能進入這個副本。";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "隊伍狀態已確認。你想預約進入「"+.@md_name$+"」嗎？";
			next;
			switch(select("預約「"+.@md_name$+"」:進入副本:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "隊伍名稱: "+ getpartyname(.@party_id);
					mes "隊長: "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000 - 預約失敗。";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000- 正在嘗試預約進入";
				mes "預約成功後，若要進入「"+.@md_name$+"」，請從選單中選擇「進入副本」。";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + " 進入紀念副本:取消") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "如果副本已經建立，你就可以進入了。";
		next;
		if(select("進入副本「"+.@md_name$+"」:取消") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ff所有與獸人的記憶副本相關的紀錄和影響已刪除。你現在可以重新建立或進入副本了。^000000";
		erasequest 12059;
		close;
	}
	mes "為了建立副本，你必須是隊長，並且隊伍中至少有2名成員。";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "發生了未知的錯誤。";
		close;
	case IE_NOINSTANCE:
		mes "紀念副本「獸人的記憶」不存在。";
		mes "紀念副本已被隊長銷毀，或是時間已到。請在2小時後再試一次。";
		close;
	case IE_NOMEMBER:
		mes "只有隊伍成員才能進入紀念副本。";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +" 隊伍的成員 "+strcharinfo(0)+" 已經進入了獸人的記憶。",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

1@orcs,180,30,0	script	#Resurrect Monsters1	844,{
	end;

OnInstanceInit:
	disablenpc instance_npcname("#Resurrect Monsters1");
	end;

OnDisable:
	stopnpctimer;
	killmonster instance_mapname("1@orcs"),instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnEnable:
	enablenpc instance_npcname("#Resurrect Monsters1");
	monster instance_mapname("1@orcs"),0,0,"獸人戰士",1023,30,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnMyMobDead:
	set .@map$, instance_mapname("1@orcs");
	set .@mob_dead_num, 30 - mobcount(.@map$,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead");
	if (.@mob_dead_num > 0) {
		set .@mob_ran,rand(1,30);
		if (.@mob_ran > 29) {
			monster .@map$,0,0,"獸人戰士",1023,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
		}
		else if ((.@mob_ran > 28) && (.@mob_ran < 30)) {
			monster .@map$,0,0,"獸人猛將",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "獸人猛將: 我們需要更多防禦！叫更多人過來！",bc_map,"0xff4444";
			}
		}
		else if ((.@mob_ran > 26) && (.@mob_ran < 29)) {
			areamonster .@map$,41,91,51,81,"獸人猛將",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "獸人猛將在哪裡！？讓他們去阻止敵人！",bc_map,"0xff4444";
			}
		}
		else {
			areamonster .@map$,17,187,27,177,"獸人猛將",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,5) == 3) {
				mapannounce .@map$, "警告: 軍隊開始在 4 號區域集結了。",bc_map,"0x77ff77";
			}
			if (rand(1,100) == 50) {
				initnpctimer;
			}
		}
	}
	end;

OnTimer10:
	mapannounce instance_mapname("1@orcs"), "守衛隊長大喊: 看來這會比預期的要花更長的時間。召喚鐘乳巨石怪！",bc_map,"0xff4444";
	end;

OnTimer4010:
	set .@map$, instance_mapname("1@orcs");
	mapannounce .@map$, "鐘乳巨石怪正從深層地底鑽出來。",bc_map,"0x77ff77";
	areamonster .@map$,17,187,27,177,"鐘乳巨石怪",1278,20,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	stopnpctimer;
	end;
}
