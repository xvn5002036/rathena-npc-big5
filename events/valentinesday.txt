// 史蒂芬 - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	史蒂芬	58,{
	mes "[史蒂芬]";
	mes "你猜我拿到了什麼？";
	mes "一種在米德加爾特很難找到的美味點心……";
	next;
	mes "[史蒂芬]";
	mes "巧克力！";
	mes "沒錯，你難道不愛巧克力嗎……我很愛。";
	mes "而且你很幸運，因為我一個只賣 5,000 Zeny！";
	next;
	if(select("我想要一些巧克力！:不用了，謝謝。")==2) {
		mes "[史蒂芬]";
		mes "你不想要巧克力嗎？";
		mes "我跟你說！你會後悔的！";
		mes "你最好現在就買一些……你再也碰不到這樣的巧克力了！";
		mes "好好考慮一下，有空再來找我。";
		close; }
	mes "[史蒂芬]";
	mes "哈！";
	mes "我就知道！";
	mes "但我一次最多只能賣你 5 個……不過，如果你真的還需要……";
	mes "你可以再回來找我。";
	mes "那麼你想要多少個？";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[史蒂芬]";
	if (.@i > 5)  {
		mes "抱歉，我不能給你那麼多。";
		close; }
	if (Zeny < .@i*5000) {
		mes "抱歉，看來你身上的錢不夠買這些。";
		close;
	}
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "拿去吧！";
	mes "你可以把它當作禮物送人，或者自己享用！";
	mes "嗯……香甜的巧克力……";
	mes "隨時歡迎再來找我……！";
	close;
}

prt_castle,54,34,4	script	卡爾 奧爾良	47,{
	mes "[卡爾 奧爾良]";
	mes "有什麼事嗎？";
	next;
	if(select("我想要一些手工巧克力……:我迷路了，抱歉打擾你。")==2) {
		mes "[卡爾 奧爾良]";
		mes "噢……好吧，如果你想要我特製的手工巧克力……";
		mes "你至少得給我 ^0000FF 3 個巧克力^000000。";
		next;
		mes "[卡爾 奧爾良]";
		mes "沒錯，只要 ^0000FF 3 個巧克力^000000";
		mes "把它們帶來給我，你就能得償所願。";
		next;
		mes "[卡爾 奧爾良]";
		mes "再見。";
		close; }
	mes "[卡爾 奧爾良]";
	mes "嗯，我或許能滿足你的需求……";
	next;
	mes "[卡爾 奧爾良]";
	if (countitem(558) < 3) {
		mes "抱歉，你沒有足夠的巧克力塊來做這個。";
		close; }
	delitem 558,3;
	mes "我看看，你有 3 個純巧克力。";
	mes "把它們交給我……";
	next;
	mes "[卡爾 奧爾良]";
	mes "好，如果你保證會好好使用它，我才會為你製作特製的手工巧克力。";
	next;
	mes "[卡爾 奧爾良]";
	mes "……嗯…………";
	mes "好了……";
	next;
	mes "[卡爾 奧爾良]";
	getitem 559,1;
	mes "拿去。";
	mes "希望你能把它送給特別的人，因為這是一份特別的巧克力。";
	mes "如你所知……只有我能做出這個。";
	next;
	mes "[卡爾 奧爾良]";
	mes "好好享用吧。";
	close;
}
