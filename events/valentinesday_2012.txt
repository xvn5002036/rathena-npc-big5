mosk_in,21,246,5	script	粉紅狂熱	71,{
	set .@n$,"^0000FF[粉紅狂熱]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "你好 "+strcharinfo(0)+"，你";
		mes "想要買些道具嗎？";
		mes "你必須給我愛之花";
		mes "來交換，這是一定的！";
		next;
		if(select("好的，沒問題！","不了，當我沒說。") == 2) close;
		next;
		mes .@n$;
		mes "請選擇一個道具。";
		mes "你需要的愛之花數量標示在括號內。";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000 "+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "如果你想要那個，你需要更多的愛之花！";
			close;
		}
		mes "你確定要用 ^00aa00"+.@SweetsCost[.@choice]+" 個愛之花 ^000000 交換 ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000 嗎？";
		if(select("不了，我改變主意了。:是的，我要交換！") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "祝你使用愉快！";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "你必須帶 15 個愛之花給我！";
			close;
		}
		mes "非常感謝你";
		mes "幫我們把花找回來！";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "收下這個當作一點";
		mes "「謝禮」吧。";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "如果你獲得更多的愛之花，";
		mes "你可以在這裡用它們";
		mes "交換一些甜點。";
		mes "期待再相見！";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "請帶一頂 +8 蛋糕帽";
		mes "給 烘焙大師，";
		mes "他就在我旁邊！";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "你好 "+strcharinfo(0)+"，";
		mes "情人節到了，我們";
		mes "很樂意為大家送上";
		mes "甜美的巧克力。";
		next;
		mes .@n$;
		mes "但我們的問題是，我們";
		mes "需要一位像你一樣";
		mes "強大的人來幫忙，";
		mes "不過首先你必須";
		mes "帶一頂 +8 蛋糕帽";
		mes "給 烘焙大師，";
		mes "他就在我旁邊！";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "你好 "+strcharinfo(0)+"！";
	close;
OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

mosk_in,21,244,3	script	烘焙大師	410,{
	set .@n$,"^0000FF[烘焙大師]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "去跟 粉紅狂熱 交換你的愛之花吧！";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "帶 15 個愛之花給 粉紅狂熱！";
		close;
	} else if(#V_QUE12==1) { 
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "啊，所以是 粉紅狂熱 叫你";
			mes "帶 +8 蛋糕帽來";
			mes "找我的。現在我要";
			mes "把你的 +8 蛋糕帽";
			mes "換成另一頂蛋糕帽。";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "現在，如果你戴著";
			mes "這頂新的蛋糕帽，就有";
			mes "機率從";
			mes "人形系魔物身上";
			mes "打到愛之花！";
			next;
			mes .@n$;
			mes "魔物們偷走了我們的花，";
			mes "我們需要用它來製作";
			mes "巧克力。現在去收集";
			mes "15 個愛之花，並把它們";
			mes "交給 粉紅狂熱 吧！";
			set #V_QUE12,2;
			close;
		} else {
			mes "抱歉，你的";
			mes "+8 蛋糕帽呢？把它";
			mes "帶來給我，並且記得";
			mes "要裝備在身上！";
			close;
		}
	} else {
		mes .@n$;
		mes "你好，"+strcharinfo(0)+"！";
		close;
	}
}
