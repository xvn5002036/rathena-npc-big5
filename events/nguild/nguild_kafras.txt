function	script	F_GKafra	{
	cutin "kafra_01",2;
	set @GID, GetCastleData(getarg(0),1);
	if (getcharid(2) == @GID && getgdskilllv(@GID,10001)) goto L_StartG;
	mes "[卡普拉服務人員]";
	mes "我們有簽訂合約，我只負責為 ^ff0000" + GetGuildName(@GID) + "^000000 公會提供專屬服務。請您尋求附近其他的卡普拉服務人員服務。造成您的不便敬請見諒。";
	cutin "",255;
	close;
L_StartG:
	set @wrpP[0], 200;
	set @wrpD$[0], getarg(1);
	setarray @wrpC$[0], @wrpD$[0]+" ^880000"+@wrpP[0]+"^000000 z", "取消", "", "", "","";
	callfunc "F_Kafra",2,0,0,0,800;
	end;
}

// Castle 1 ===============================================
nguild_alde,218,170,0	script	卡普拉服務人員#N01	117,{
	callfunc "F_GKafra", "nguild_alde", "普隆德拉";
	end;
OnRecvCastleN01:
	if (GetCastleData("nguild_alde",9) < 1) disablenpc "卡普拉服務人員#N01";
	end;
}
