function	script	F_GldDunSw	{
	set .@GID, GetCastleData(getarg(0),1);
	if (.@GID == 0) {
		mes "[迴盪的聲音]";
		mes " ' 克服試煉並展現真正勇氣之人……將會找到出路…… ' ";
		close;
	}
	else {
		mes "[迴盪的聲音]";
		mes " ' 只有展現真正勇氣之人，才能接受這個考驗。 '";
		next;
		mes " ";
		mes "這裡有一個小拉桿。你要拉下它嗎？";
		next;
		if (select("拉下。:不要。") == 1) {
			if (getcharid(2) == .@GID) {
				warp "gld_dun"+getarg(1),getarg(2),getarg(3);
				end;
			}
			mes " ";
			mes " 什麼事也沒發生。";
		}
		return;
	}
}

	callfunc "F_GldDunSw","nguild_alde","02",32,122;
	close;
}
