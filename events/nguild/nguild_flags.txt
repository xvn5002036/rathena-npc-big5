/// Flag Function
//============================================================
function	script	F_Flags	{
	set .@GID, GetCastleData(getarg(1),1);
	if (getarg(5) == 0) return;
	if (.@GID == 0) {
		mes "[ 盧恩米德加茲王國 神聖法令 ]";
		mes " ";
		mes "1. 遵循盧恩米德加茲王國的神聖法令，";
		mes "我們在此宣布";
		mes "這座城堡目前沒有正式的主人。";
		mes " ";
		mes "2. 能夠";
		mes "克服所有考驗";
		mes "並摧毀華麗金屬的人，";
		mes "國王將會賦予他";
		mes "這座城堡的所有權。";
		return;
	}
	else {
		if (getcharid(2) == .@GID && getarg(4) == 1) {
			mes "勇敢的戰士啊...";
			mes "你想要返回你的榮耀之地嗎？";
			next;
			if (select("返回公會城堡。:取消。") == 1) {
				if (getcharid(2) == GetCastleData(getarg(1),1))
					warp getarg(1),getarg(2),getarg(3);
			}
			return;
		}
		mes "[ 盧恩米德加茲王國 神聖法令 ]";
		mes " ";
		mes "1. 根據盧恩米德加茲王國的";
		mes "神聖法令，";
		mes "我們正式核准這個地方";
		mes "為 ^ff0000" + GetGuildName(.@GID) + "^000000 公會的私人領地。";
		mes " ";
		mes "2. ^ff0000"+ GetGuildName(.@GID) + "^000000 公會的會長是";
		mes "^FF0000" + GetGuildMaster(.@GID) + "^000000";
		mes "若有任何人對此提出異議，";
		mes "請拿起手中的鋼刃，證明你的實力與榮耀。";
		return;
	}
}
