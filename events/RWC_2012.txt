prontera,147,61,3	script	打洞專家#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你的物品欄裡有太多東西了。請去卡普拉服務人員那裡存放一些物品後再來。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你因為超重而無法繼續。";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[打洞專家]";
		mes "我的工作是為 RWC紀念飾品打洞。";
		next;
		mes "[打洞專家]";
		mes "很抱歉，你右邊的飾品欄位沒有裝備任何物品。";
		close;
	}
	mes "[打洞專家]";
	mes "我的工作是為 RWC紀念飾品打洞。而且，我只處理^ff0000純淨的物品^000000，也就是那些還沒有被附魔過的物品。";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "[打洞專家]";
		mes "然而，我看到你右邊裝備的飾品無法處理。請裝備RWC紀念飾品。";
		close;
	}
	mes "[打洞專家]";
	mes "你也應該知道，打洞是非常危險的。^ff0000成功的機率大約是 50%。^000000 你希望繼續嗎？";
	next;
	if(select("取消:來吧！") == 1) {
		mes "[打洞專家]";
		mes "那麼再見了。";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967;
//RWC_2012_Ring_
		set .@name$,"RWC 2012 Memorial Ring";
		set .@str$,"ring";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969;
//RWC_2012_Pendant_
		set .@name$,"RWC 2012 Memorial Pendant";
		set .@str$,"pendant";
	} else {
		mes "[打洞專家]";
		mes "我無法辨識你右手裝備的飾品，我沒辦法處理它。";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "[打洞專家]";
		mes "這個物品已經被附魔過了。我不能處理這個，因為這違反了規定。";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "[打洞專家]";
		mes "耶！成功了！你的 "+.@name$+" 現在有卡片插槽了。快看看吧！";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "[打洞專家]";
		mes "啊... 這脆弱的 "+.@str$+"... 它在過程中碎裂了。我很抱歉。";
		close;
	}
}

prontera,147,59,3	script	戈德堡#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你攜帶了太多物品。請整理好你的物品欄後再來。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你因為超重而無法繼續。";
		close;
	}
	mes "[戈德堡]";
	mes "你好！我負責為 RWC紀念飾品附加上神秘力量。";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[戈德堡]";
		mes "很抱歉，你右邊的飾品欄位沒有裝備任何物品。";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[戈德堡]";
		mes "然而，我看到你裝備的飾品不是我能處理的東西。請裝備 RWC紀念飾品。";
		close;
	}
	set .@select, select("抱歉，我沒興趣。:請強化我的飾品。:移除附魔。")-1;
	if (.@select == 0) {
		mes "[戈德堡]";
		mes "好吧，那麼下次見...";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[戈德堡]";
			mes "很抱歉，但我無法處理你目前佩戴的飾品。";
			mes "如果你把 RWC紀念飾品裝備在左邊，請試著換到右邊。";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[戈德堡]";
			mes "你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("取消:鬥志:ATK (%):最大 HP:HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("取消:魔力:MATK (%):SP")-2;
			break;
		case 2:
			mes "[戈德堡]";
			mes "^ff0000請小心！附魔大約有 25% 的機率會失敗。如果失敗，該物品將會被摧毀。^000000 你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[戈德堡]";
			mes "^ff0000大約有 25% 的機率附魔會失敗。如果失敗，該物品將會被摧毀。^000000 你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:MHP:HP")-2;
			break;
		case 0:
			mes "[戈德堡]";
			mes "你的飾品已經有太多附魔了，我幾乎無法再對它進行任何處理。";
			close;
		}
		if (.@i == -1) {
			mes "[戈德堡]";
			mes "好吧，那麼下次見。";
			close;
		}
		mes "[戈德堡]";
		mes "附魔的力量將會是隨機選擇的。^ff0000一旦注入，附魔將無法被移除。^000000 我們要繼續嗎？";
		next;
		if(select("不，請停止。:是的，請繼續。") == 1) {
			mes "[戈德堡]";
			mes "好吧，那麼下次見...";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[戈德堡]";
			mes "在我工作的時候不要脫下你的裝備，好嗎？";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806;
//Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742;
//Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867;
//MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[戈德堡]";
			mes "我們遇到問題了，讓我檢查一下。";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.
		if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;
// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[戈德堡]";
			mes "天啊！";
			mes "這個物品不夠堅固，無法承受附魔的力量，因此被摧毀了。我很抱歉。";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[戈德堡]";
			mes "喔... 看來注入的所有力量之間產生了某種不穩定。這導致所有的附魔都消失了。真可惜，請再試一次！";
		} else {
			mes "[戈德堡]";
			mes "太棒了！";
			mes "附魔成功！它將被應用在第 ^990000"+(.@slot+1)+"^000000 個插槽中。";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[戈德堡]";
		mes "我只會初始化附魔選項，不會對插槽中的卡片做任何變動。你想要繼續嗎？";
		next;
		if(select("我要停止。:是的，請繼續。") == 1) {
			mes "[戈德堡]";
			mes "如果你改變主意再來找我。";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[戈德堡]";
			mes "很抱歉。你沒有 RWC附魔重置券。你能檢查一下你的物品欄嗎？";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[戈德堡]";
			mes "嗯...這個裝備是乾淨的。如果上面什麼都沒有，我就無法初始化它！再檢查一下。";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[戈德堡]";
		mes "你的物品上的附魔選項將被初始化。";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		close;
	}
}
