schg_dun01,1,1,1	script	怪物控制器#sch_gd	81,{
OnInit:
	donpcevent "怪物控制器1#sch::OnKill";

	end;

schg_dun01,1,2,1	script	怪物控制器1#sch	81,{
OnEnable:
	set .@callwhere,rand(1,4);
	if (.@callwhere == 1) {
		monster "schg_dun01",150,340,"庫柏林",1980,1,"怪物控制器1#sch::OnMyMobDead";
	}
	else if (.@callwhere == 2) {
		monster "schg_dun01",232,293,"庫柏林",1980,1,"怪物控制器1#sch::OnMyMobDead";
	}
	else if (.@callwhere == 3) {
		monster "schg_dun01",156,167,"庫柏林",1980,1,"怪物控制器1#sch::OnMyMobDead";
	}
	else {
		monster "schg_dun01",243,56,"庫柏林",1980,1,"怪物控制器1#sch::OnMyMobDead";
	}
	end;

OnKill:
	killmonster "schg_dun01","怪物控制器1#sch::OnMyMObDead";
	end;

OnMyMobDead:
	if (mobcount("schg_dun01","怪物控制器1#sch::OnMyMObDead") == 0) {
		mapannounce "schg_dun01", "庫柏林：啊啊！",bc_map,"0x99CC00";
		mapannounce "schg_dun01", "莫爾斯通：我好像聽到庫柏林的慘叫聲！！是誰在那裡？庫柏林怎麼了？嘿，你！",bc_map,"0x99CC00";
		donpcevent "矮人#sch_gd::OnEnable";
	}
	end;
}
