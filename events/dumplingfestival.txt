payon,93,81,4	script	驅魔大師 法海	834,{

	mes "[驅魔大師 法海]";
	mes "你好，年輕的戰士，我是驅魔大師 法海。";
	mes "我追蹤那神出鬼沒的白素貞已經有一段時間了。";
	next;
	mes "[驅魔大師 法海]";
	mes "我覺得是時候讓另一位戰士來嘗試完成";
	mes "我的任務了。我有一些物品可以協助你進行任務。";
	next;
	menu "我想購買",-, "不用了，謝謝",M_FIN;
  
	mes "[驅魔大師 法海]";
	mes "我有 2 件對你可能會很有用的物品。";
	next;
	menu "雄黃酒",-, "驅魔草",M_HERB;
  
	mes "[驅魔大師 法海]";
	mes "你需要多少個？";
	mes "你一次最多只能購買 5 個。";
	mes "每個售價 20000 Zeny。";
	mes "(輸入 0 取消)";
	next;
	input @input;
	if(@input==0) close;
	if(@input>5) goto L_SORRY;
	if(Zeny< @input*20000) goto L_NOZENY;
	set Zeny,Zeny-20000*@input;
	getitem 682,@input;
	mes "[驅魔大師 法海]";
	mes "給你，希望你能成功完成我的任務。";
	close;
  
L_SORRY:
	mes "[驅魔大師 法海]";
	mes "你不該貪心，誠實才是神之道。";
	close;
  
M_FIN:
	mes "[驅魔大師 法海]";
	mes "徹底消滅邪惡的道路還很漫長，在神之道上協助我吧。";
	close;
M_HERB:
	mes "[驅魔大師 法海]";
	mes "你需要多少個？";
	mes "你一次最多只能購買 5 個。";
	mes "每個售價 10000 Zeny。";
	mes "(輸入 0 取消)";
	next;
	input @input;
	if(@input==0) close;
	if(@input>5) goto L_SORRY;
	if(Zeny< @input*10000) goto L_NOZENY;
	set Zeny,Zeny-10000*@input;
	getitem 683,@input;
	mes "[驅魔大師 法海]";
	mes "給你，希望你能成功完成我的任務。";
	close;

L_NOZENY:
	mes "[驅魔大師 法海]";
	mes "金錢不能為每個人帶來快樂，但我們需要它來維持寺廟和我的生計。請嘗試去擊敗一些魔物並拿走牠們的掉落物吧。";
	close;
}
