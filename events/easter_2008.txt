							mes "下次再見！";

	switch(select("製作聖蛋需要什麼？:我想製作聖蛋。:也許下次吧。")) {
	case 1:
		mes "[奈倫]";

		mes "製作的每一個聖蛋，";
		mes "你需要";
		mes "帶來...";
		next;
		mes "[奈倫]";

		mes "^4D4DFF1 黃色藥草^000000，";
		mes "^4D4DFF1 聖水^000000，以及";
		mes "^4D4DFF1 雞蛋^000000。";
		close;
	case 2:
		mes "[奈倫]";
		mes "好的，你希望我";
		mes "為你製作多少個";
		mes "聖蛋呢？";
		mes "我一次最多只能製作";
		mes "10 個聖蛋。";
		mes "輸入 ''0'' 取消。";
		next;
		while(1) {
			input .@input;
			if (.@input == 0) {
				mes "[奈倫]";
				mes "喔，你改變主意了？";
				mes "隨時歡迎你來找我";
				mes "為你製作一些聖蛋~";
				close;

			else if (.@input < 0 || .@input > 10) {
				mes "[奈倫]";
				mes "很抱歉，我一次最多";
				mes "只能製作 10 個聖蛋。";
				mes "如果你想";
				mes "取消，只需輸入 ''0''。";
				next;
			}
			else {
				if ((countitem(509) < .@input) || (countitem(511) < .@input) || (countitem(508) < .@input) || (countitem(523) < .@input) || (countitem(7605) < .@input)) {
					mes "[奈倫]";
					mes "很抱歉，但你沒有";
					mes "足夠的物品來製作";
					mes "聖蛋。你需要";
					mes "以下物品，我才能為你";
					mes "製作每個聖蛋。";
					next;
					mes "[奈倫]";
					mes "^4D4DFF1 白色藥草^000000，";
					mes "^4D4DFF1 綠色藥草^000000，";
					mes "^4D4DFF1 黃色藥草^000000，";
					mes "^4D4DFF1 聖水^000000，以及";
					mes "^4D4DFF1 雞蛋^000000。";
					close;
				}
				else {
					break;
				}
			}
		}
		if (checkweight(12019,.@input) == 0) {
			mes "[奈倫]";
			mes "很抱歉，但你現在";
			mes "拿不了那麼多";
			mes "聖蛋...";
			mes "你最好先騰出";
			mes "一些物品欄空間。";
			close;
		}
		else {
			delitem 509,.@input; //White_Herb
			delitem 511,.@input; //Green_Herb
			delitem 508,.@input; //Yellow_Herb
			delitem 523,.@input; //Holy_Water
			delitem 7605,.@input; //Egg
			getitem 12019,.@input; //Holy_Egg
			mes "[奈倫]";
			mes "謝謝你！";
			mes "希望你喜歡";
			mes "這些聖蛋~";
			mes "願上帝的愛與你同在。";
			close;
		}
	case 3:
		mes "[奈倫]";
		mes "好的，我明白了。";
		mes "願上帝保佑你。";
		close;
	}
}
