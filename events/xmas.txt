xmas_in,100,96,4	script	聖誕老人::Santa2	718,{
	mes "[聖誕老人]";
	if(xmas_npc==0) set xmas_npc, 1; // For Lutie & Bard quest
	if(#event_xmas > 0 && #event_xmas < 30 ) goto L_Start;
	mes "聖誕快樂！";
	if(Class==0 || #event_xmas>=30 ) close; //anti exploit protection
	mes "我有一份禮物要給你！呵呵呵！";
	getitem rand(664,667),1; //gives one of 4 gift boxes
	set #event_xmas,#event_xmas+1;
	close;

L_Start:
	mes "我遇到了一點問題……";
	mes "你願意聽聽看嗎？";
	next;
	menu "聽聖誕老人說。",M_0, "給聖誕老人證據。",M_1, "取消。",M_End;

	M_0:
		mes "[聖誕老人]";
		mes "我的問題是這樣的。";
		mes "外面似乎有一個人";
		mes "在冒充我，並在各地散播";
		mes "恐懼。";
		next;
		mes "[聖誕老人]";
		mes "就像傳說中的鬼靈精一樣，他拿走了";
		mes "所有孩子們的玩具，並把它們據為己有";
		mes "。";
		next;
		mes "[聖誕老人]";
		mes "我在這裡太忙了，要製作明年";
		mes "的玩具，所以我無法";
		mes "親自出去找他。";
		mes "所以我希望你能出去";
		mes "替我消滅這個人。";
		next;
		mes "[聖誕老人]";
		mes "不過，他手裡有我的一個";
		mes "魔法袋，所以每次你攻擊他時，";
		mes "他都會逃進袋子裡，轉移到";
		mes "另一個地方。";
		next;
		mes "[聖誕老人]";
		mes "然而，";
		mes "他在慌亂中經常會掉東西。";
		mes "如果他碰巧掉了一隻他用來偷";
		mes "可憐孩子們玩具的破洞襪子，";
		mes "請把它撿起來。";
		next;
		mes "[聖誕老人]";
		mes "如果你收集了 3 個這個，我會給";
		mes "你一個我一直放在實驗室的";
		mes "神秘原型箱。";
		mes "它會隨機吐出禮物，";
		mes "能省下我很多工作。";
		close;
	M_1:
		mes "[聖誕老人]";
		if(countitem(7034) < 3) goto L_NotEnuf;
		delitem 7034,3;
		mes "看來你在消滅那些";
		mes "假聖誕老人方面做得很好。";
		mes "繼續保持！";
		next;
		getitem 644,1; //Gift Box Prototype
		set #event_xmas,#event_xmas+1;
		mes "[聖誕老人]";
		mes "這是你的獎勵。";
		mes "如果你再拿到 3 個，我會再給你一個。";
		mes "希望你能拿到好東西。";
		close;
		L_NotEnuf:
			mes "你沒有足夠的襪子作為證據。";
			mes "去打倒那些邪惡的聖誕老人";
			mes "並幫我拿更多回來，我會獎勵你的。";
			close;
	M_End:
		mes "[聖誕老人]";
		mes "我明白了。好吧，至少";
		mes "我們會在聖誕節早晨再次相見。";
		close;

OnInit:
	disablenpc "聖誕老人";
	end;
}
