mosk_dun02,53,220,4	script	#babayagain	WARPNPC,1,1,{
	end;
OnTouch:
	if (mos_nowinter == 11) {
		donpcevent "士兵1#mos::OnEnable";
		donpcevent "士兵2#mos::OnEnable";
		donpcevent "士兵3#mos::OnEnable";
		mes "[士兵]";
		mes "在那裡停下來！";
		next;
		mes "[士兵]";
		mes "我們收到情報，";
		mes "指出你一直暗中";
		mes "與可怕的";
		mes "芭芭亞加聯繫。";
		next;
		mes "[士兵]";
		mes "如果你有什麼話想說，";
		mes "去跟沙皇說吧。";
		mes "跟我們走。";
		close2;
		donpcevent "士兵1#mos::OnInit";
		donpcevent "士兵2#mos::OnInit";
		donpcevent "士兵3#mos::OnInit";
		set mos_nowinter,12;
		warp "mosk_in",125,92;
		end;
	}
	else if (mos_nowinter == 12 || mos_nowinter == 13) {
		donpcevent "士兵1#mos::OnEnable";
		donpcevent "士兵2#mos::OnEnable";
		donpcevent "士兵3#mos::OnEnable";
		mes "[士兵]";
		mes "我們早就料到";
		mes "你會再次出現。";
		mes "你找沙皇有什麼事？";
		next;
		if(select("逃跑。:解釋情況。") == 1) {
			mes "["+strcharinfo(0)+"]";
			mes "我應該離開這裡...";
			close2;
			warp "mosk_dun02",100,100;
		}
		else {
			mes "-我解釋了到目前為止所發生的事情。-";
			next;
			mes "[士兵]";
			mes "如果你有什麼話想說，去跟";
			mes "沙皇說吧。";
			mes "如果你離開這裡，";
			mes "將會被視為我們的敵人。";
			close2;
		}
		donpcevent "士兵1#mos::OnInit";
		donpcevent "士兵2#mos::OnInit";
		donpcevent "士兵3#mos::OnInit";
		end;
	}
	else {
		warp "mosk_in",215,41;
		end;
	}
}
