	if (strnpcinfo(2) == "template") end;

		maprespawnguildid strnpcinfo(2),getcastledata(strnpcinfo(2),CD_GUILD_ID),2;

	end;

	if (strnpcinfo(2) == "template") end;

		set .@str$,substr(strnpcinfo(2),0,1)+substr(strnpcinfo(2),8,9);
		killmonster strnpcinfo(2),"管家#"+.@str$+"::OnStartArena";
		donpcevent strnpcinfo(0)+"::OnReset";
		donpcevent "管家#"+.@str$+"::OnStop";
	}
	end;

	if (strnpcinfo(2) == "template") end;
	killmonster strnpcinfo(2),"gard1#"+strnpcinfo(2)+"::OnGuardianDied";
	killmonster strnpcinfo(2),"gard2#"+strnpcinfo(2)+"::OnGuardianDied";

		else if (strnpcinfo(2) == "schg_cas03") setarray .@i[0],338,202;
		else setarray .@i[0],120,272;	// Castles 1,4,5 are identical.

	end;

	donpcevent "df2#"+strnpcinfo(2)+"::OnDisable";

		donpcevent "RL"+.@i+"#"+strnpcinfo(2)+"::OnDisable";

	donpcevent "第一守護石#"+.@str$+"::OnEnable";
	donpcevent "第二守護石#"+.@str$+"::OnEnable";
	end;

-	script	管家#template	-1,{
	set .@GID, getcastledata(strnpcinfo(4),CD_GUILD_ID);
	if (!.@GID) {
		mes "[ 管家 ]";
		mes "我在等待著命運為我挑選的主人。";
		mes "你認為你有足夠的勇氣與實力";
		mes "來征服這座城池嗎？";
		close;
	}
	if (is_guild_leader(.@GID) == false) {
		mes "[ 管家 ]";
		mes "哼。你的威脅嚇不倒我！";

	mes "想與我討論？";

	switch(select("城池狀態簡報:投資商業發展:投資防禦發展:雇用/解雇儲藏室員工:前往會長房間")) {
	case 1:
		mes "[ 管家 ]";
		mes "城池的商業發展度是 ^0000ff"+getcastledata(strnpcinfo(4),CD_CURRENT_ECONOMY)+".";
		if (getcastledata(strnpcinfo(4),CD_INVESTED_ECONOMY) > 0) {
			mes "上次，您投資了";
			mes "商業發展 "+getcastledata(strnpcinfo(4),CD_INVESTED_ECONOMY)+".";
		}
		next;
		mes "[ 管家 ]";
		mes "我們城池的";
		mes "防禦度是 "+getcastledata(strnpcinfo(4),CD_CURRENT_DEFENSE)+".";
		if (getcastledata(strnpcinfo(4),CD_INVESTED_DEFENSE) > 0) {
			mes "上次，您投資了";
			mes "防禦發展 "+getcastledata(strnpcinfo(4),CD_INVESTED_DEFENSE)+" 次。";
		}
		mes " ";
		mes "報告完畢，會長。";
		close;
	case 2:
		set .@Economy,getcastledata(strnpcinfo(4),CD_CURRENT_ECONOMY);
		setarray .@cost[0],5000,10000,20000,35000,55000,80000,110000,145000,185000,230000,280000,335000,395000,460000,530000,605000,685000,770000,860000,955000;
		set .@j,0;
		for(set .@i,6; .@i<101; set .@i,.@i+5) {
			if (.@Economy < .@i) {
				set .@eco_invest,.@cost[.@j];
				break;
			}
			set .@j, .@j+1;
		}
		// Quadruple the cost of investing if you've already invested once.
		if (getcastledata(strnpcinfo(4),CD_INVESTED_ECONOMY)) set .@eco_invest,.@eco_invest*4;
		mes "[ 管家 ]";
		mes "提升城池的商業發展";
		mes "可以增加公會生產的物資數量。";
		mes "投資商業發展";
		mes "將有助於公會的未來。";
		next;
		mes "[ 管家 ]";
		mes "您每天可以進行一次投資，";
		mes "但如果您支付更多 Zeny，";
		mes "可以進行兩次投資：";
		mes "這將加速商業發展，";
		mes "不過花費會相當昂貴。";
		next;
		if (.@Economy == 100) {
			mes "[ 管家 ]";
			mes "然而，我們城池的";
			mes "商業發展度已經達到 100%。";
			mes "目前無法再進一步";
			mes "發展商業了。";
			close;
		}
		if (getcastledata(strnpcinfo(4),CD_INVESTED_ECONOMY) >= 2) {
			mes "[ 管家 ]";
			mes "您今天已經進行了兩次投資，";
			mes "所以您必須等到明天";
			mes "才能再次進行投資。";
			close;
		}
		if (getcastledata(strnpcinfo(4),CD_INVESTED_ECONOMY) == 0) {
			mes "[ 管家 ]";
			mes "您必須支付 ^FF0000"+.@eco_invest+"^000000 Zeny";

			mes "明智地運用了公會的資金，會長。";

		for(set .@i,6; .@i<101; set .@i,.@i+5) {
			if (.@Defence < .@i) {
				set .@def_invest,.@cost[.@j];
				break;
			}
			set .@j, .@j+1;
		}
		// Quadruple the cost of investing if you've already invested once.
		if (getcastledata(strnpcinfo(4),CD_INVESTED_DEFENSE)) set .@def_invest,.@def_invest*4;
		mes "[ 管家 ]";
		mes "投資城池的防禦發展";
		mes "將提升我們的守護者";
		mes "與華麗金屬的耐久度。";
		mes "我們需要利用一切優勢";
		mes "來保護自己免受敵人的攻擊。";
		next;
		mes "[ 管家 ]";
		mes "您每天可以投資一次防禦發展，";
		mes "但如果您支付更多 Zeny，";
		mes "您每天最多可以投資兩次。";
		next;
		mes "[ 管家 ]";

			mes "這座城池的防禦度已經";
			mes "達到 100%，";
			mes "無法再進一步提升了。";
			close;
		}
		if (getcastledata(strnpcinfo(4),CD_INVESTED_DEFENSE) >= 2) {
			mes "會長，您今天已經";
			mes "投資了兩次防禦發展。";
			mes "如果您真的想增強我們的防禦，";
			mes "您需要等到明天。";

			mes "您現在要投資嗎？";
		}

			mes "防禦投資。";
			mes "您現在要投資嗎？";
		}
		next;
		switch(select("投資防禦發展:取消")) {
		case 1:
			if (getcastledata(strnpcinfo(4),CD_INVESTED_DEFENSE) >= 2) {
				mes "[ 管家 ]";
				mes "會長，您今天已經";
				mes "投資了兩次防禦發展。";

				close;
			}
			if (Zeny < .@def_invest) {
				mes "[ 管家 ]";
				mes "很抱歉，會長，但您沒有";
				mes "足夠的 Zeny 來為公會";

				close;

		mes "華麗金屬中心嗎？^000000";

			warp strnpcinfo(4),getarg(1),getarg(2);
		close;
	}
	mes "^3355FF這裡是城池傳送服務。";
	mes "請選擇";
	mes "城池內的目的地。^000000";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+3)
		set .@menu$, .@menu$+getarg(.@i)+":";
	set .@menu$, .@menu$+"取消";

	close;
}
