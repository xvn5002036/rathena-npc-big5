				close2;
				warp "einbroch",174,204;
				end;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "很抱歉，您身上攜帶了";
				mes "太多的物品。在購買";
				mes "蘋果套餐之前，請先將一些物品";
