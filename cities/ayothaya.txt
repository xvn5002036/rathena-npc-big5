-	script	::Aibakthing_ayothaya	-1,{
	mes "[艾巴克辛]";
	mes "呼！哈！呼！哈！";
	mes "讓我帶你前往";
	mes "遠方的精神聖地，";
	mes "一個充滿異國神秘色彩的國度，";
	mes "我的故鄉阿育陀耶！";
	next;
	switch(select("關於阿育陀耶。:前往阿育陀耶。:取消。")) {
	case 1:
		mes "[艾巴克辛]";
		mes "我們阿育陀耶人是虔誠的民族，重視和平與誠意。我們努力在與自然的和諧中，過著簡單而高尚的生活。";
		next;
		mes "[艾巴克辛]";
		mes "我們的傳統料理聞名世界。當我們端出像冬蔭功（Tom Yum Goong）這樣由蝦、檸檬和辣椒製成的料理時，那誘人的香氣和超乎想像的美味，絕對不會讓你感到驚訝。";
		next;
		mes "[艾巴克辛]";
		mes "對於那些渴望冒險的人來說，有個古老的故事，關於一隻潛伏在我們寺廟遺跡中的邪惡生物——薩明虎（Sa-mhing Tiger）。";
		next;
		mes "[艾巴克辛]";
		mes "呼！哈！呼！哈！";
		mes "沒有人能抗拒這片";
		mes "美麗土地的召喚。";
		mes "米德加爾特人";
		mes "永遠受到歡迎！";
		close;
	case 2:
		mes "[艾巴克辛]";
		mes "啊，如果你想訪問阿育陀耶，";
		mes "必須支付 10,000 Zeny。";
		mes "如果你準備好了，我們";
		mes "現在就可以出發?";
		next;
		if (select("我準備好了，走吧！:不。") == 1) {
			if (Zeny > 9999) {
				mes "[艾巴克辛]";
				mes "呼！哈！呼！哈！";
				mes "讓我們出發吧！回到我";
				mes "美麗的阿育陀耶！";
				close2;
				set Zeny, Zeny-10000;
				warp "ayothaya",149,71;
				end;
			}
			mes "[艾巴克辛]";
			mes "我很抱歉，但你沒有 10,000 Zeny 可以前往阿育陀耶。與等待著你的體驗相比，這樣的價格根本不算什麼！";
			close;
		}
	case 3:
		mes "[艾巴克辛]";
		mes "你是否曾夢想過";
		mes "一個充滿精神寧靜的";
		mes "美麗地方？朋友，你";
		mes "一定是在夢見阿育陀耶。";
		close;
	}
}

	mes "慢慢來吧。";
	close;
}
