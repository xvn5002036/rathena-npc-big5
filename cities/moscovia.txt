	mes "[莫斯科比亞宣傳人員]";
	mes "莫斯科比亞！你知道莫斯科比亞嗎？";
	mes "那片在無盡海洋上展開的樂園...";
	mes "歡迎來到莫斯科比亞。";
	mes "這是一個充滿冒險與神秘的地方。";
	next;
	switch(select("關於莫斯科比亞...:前往莫斯科比亞。:取消")) {
		case 1:
			mes "[莫斯科比亞宣傳人員]";
			mes "莫斯科比亞是一個美麗的王國，";
			mes "位於盧恩";
			mes "米德加爾特北方的一座島嶼上。";
			next;
			mes "[莫斯科比亞宣傳人員]";
			mes "我確信你一定會";
			mes "深深著迷於";
			mes "莫斯科比亞的美麗風景";
			mes "與華麗的宮殿。";
			next;
			mes "[莫斯科比亞宣傳人員]";
			mes "現在我們漫長的冬天已經";
			mes "過去了，";
			mes "我很高興現在能向你展示";
			mes "我們華麗的故鄉。";
			next;
			mes "[莫斯科比亞宣傳人員]";
			mes "如果你想拜訪莫斯科比亞，";
			mes "現在就把握機會吧！";
			mes "我會幫助你有一趟";
			mes "美好的莫斯科比亞之旅！";
			close;

		case 2:
			mes "[莫斯科比亞宣傳人員]";
			mes "好，那我們現在就出發吧。";
			mes "你必須付給我 10,000 Zeny";
			mes "才能前往莫斯科比亞。";
			mes "但是當你回來時，";
			mes "你就不需要付錢了。";
			next;
			mes "[莫斯科比亞宣傳人員]";
			mes "我們現在可以出發了嗎？";
			next;
			if (select("出發吧！:取消") == 2) {
				mes "[莫斯科比亞宣傳人員]";
				mes "如果你現在太忙，";
				mes "請在你想去的時候隨時告訴我。";
				mes "我隨時準備好引導任何人前往莫斯科比亞。";
				close;
			}
			if (Zeny < 10000) {
				mes "[莫斯科比亞宣傳人員]";
				mes "很抱歉，但你現在沒有";
				mes "足夠的 Zeny。";
				mes "你需要 10,000 Zeny";
				mes "才能前往莫斯科比亞。";
				mes "謝謝你。";
				close;
			} else {
				mes "[莫斯科比亞宣傳人員]";
				mes "好，我們現在出發。";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[莫斯科比亞宣傳人員]";
			mes "如果你現在太忙，";
			mes "請在你想去的時候隨時告訴我。";
			mes "我隨時準備好引導任何人前往莫斯科比亞。";
			close;
	}
}
