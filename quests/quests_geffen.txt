// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	妮亞#yagu	700,{
	mes "[妮亞]";
	mes "你好，我是妮亞，遊戲大師";
	mes "馬翁的第五代弟子。現在，";
	mes "你想和我玩一局";
	mes "猜數字遊戲嗎？";
	next;
	switch(select("猜數字遊戲？:練習猜數字:開始猜數字:紀錄保持者:取消")) {
	case 1:
		mes "[妮亞]";
		mes "啊，這個嘛，猜數字遊戲的";
		mes "目的就是猜中我心裡想的";
		mes "數字。當然，有一些規則";
		mes "限制了我可以選擇的數字。";
		next;
		mes "[妮亞]";
		mes "我設計的所有數字";
		mes "都必須是3位數，任何一位數";
		mes "都不能使用0，";
		mes "而且我不能重複使用";
		mes "同一個數字。";
		next;
		mes "[妮亞]";
		mes "所以，舉例來說，我不能";
		mes "使用 103、112、";
		mes "252 或 701 這些數字。這些數字";
		mes "要麼含有 0，要麼就是";
		mes "重複使用了同一個數字。";
		mes "現在懂了嗎？";
		next;
		mes "[妮亞]";
		mes "現在，你只有 5 次猜測的機會，";
		mes "每次猜測後，我會告訴你";
		mes "你猜對了幾個數字，";
		mes "接著是猜對的數字中，";
		mes "位置也正確的數量。";
		next;
		mes "[妮亞]";
		mes "舉例來說，假設我";
		mes "想的數字是 168。";
		mes "如果你的第一次猜測是 678，那麼你猜對了兩個數字，";
		mes "6 和 8。然而，";
		mes "只有 8 的位置是正確的。";
		next;
		mes "[妮亞]";
		mes "你知道，先嘗試練習模式";
		mes "是個好主意。";
		mes "然後，等你比較";
		mes "熟悉了，再玩真正的";
		mes "猜數字遊戲。別忘了速度也會影響你的成績！";
		next;
		mes "[妮亞]";
		mes "啊，如果你碰巧在";
		mes "猜數字遊戲中創下了新紀錄，";
		mes "你的名字會被記錄下來，";
		mes "所以別錯過這個";
		mes "名利雙收的機會哦?";
		close;
	case 2:
		mes "[妮亞]";
		mes "太好了，我們來練習一局";
		mes "猜數字遊戲。請";
		mes "試著猜我的數字，好嗎？";
		mes "如果你想取消，請輸入大於 1,000 的數字。";
		set .@practice,1;
		break;
	case 3:
		mes "[妮亞]";
		mes "太好了，我們來玩猜數字吧！";
		mes "你有 5 次機會來猜我心裡想的數字。";
		mes "如果你想取消，請輸入大於 1,000 的數字。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[妮亞]";
		mes "^ff0000" + .@name_record$ + "^000000 是";
		mes "猜數字遊戲的紀錄保持者，";
		mes "完成時間為";
		mes "^ff0000" + .@score_min + " 分鐘 " + .@score_sec + " 秒^000000。";
		close;
	case 5:
		mes "[妮亞]";
		mes "保重自己，";
		mes "冒險者。如果你隨時";
		mes "想玩一局猜數字";
		mes "遊戲，就來找我說話，";
		mes "好嗎？再見?";
		close;
	}
	next;
	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);
	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[妮亞]";
				mes "你取消了";
				mes "猜數字遊戲。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[妮亞]";
			mes "你輸入的數字中";
			mes "包含了 0，或者";
			mes "重複輸入了";
			mes "同一個數字。";
			mes "請輸入其他猜測。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[妮亞]";
		mes "^ff0000"+strcharinfo(0)+"^000000，";
		mes "你的猜測是 ^0000ff"+.@input+"^000000。";
		mes "給我一點時間";
		mes "來計算你的結果。";
		next;
		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[妮亞]";
				mes "沒錯，我心裡想的數字";
				mes "就是 ^ff0000"+.@yagu+"^000000！";
				mes "恭喜你，你";
				mes "剛剛贏了這場練習";
				mes "遊戲！現在你應該";
				mes "準備好迎接真正的挑戰了?";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;
				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[妮亞]";
					mes "太棒了！你猜中了";
					mes "我的數字，也就是";
					mes "^ff0000" + .@yagu + "^000000。你完成了，耗時";
					mes "^ff0000" + .@total_min + " 分鐘 " + .@total_sec + " 秒^000000，";
					mes "打破了之前的紀錄";
					mes " ^ff0000" + .@score_min + " 分鐘 " + .@score_sec + " 秒^000000。";
					next;
					mes "[妮亞]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "創下了那個舊紀錄，";
					mes "但現在你是猜數字遊戲中";
					mes "新的挑戰目標。恭喜";
					mes "創下新紀錄！";
					close;
				}
				mes "[妮亞]";
				mes "做得好，你猜中了";
				mes "我的數字！答案";
				mes "當然是 ^ff0000" + .@yagu + "^000000。你成功";
				mes "猜對了這個數字，耗時";
				mes "^ff0000" + .@total_min + " 分鐘 " + .@total_sec + " 秒^000000。";
				next;
				mes "[妮亞]";
				mes "^ff0000" + .@name_record$ + "^000000 是";
				mes "目前猜數字的";
				mes "紀錄保持者，成績為";
				mes "^ff0000" + .@score_min + " 分鐘 " + .@score_sec + " 秒^000000。";
				mes "祝你好運，希望你";
				mes "總有一天能打破這個紀錄?";
				close;
			}
		} else {
			if (.@practice) {
				mes "[妮亞]";
				mes "你的猜測有 ^ff0000" + .@ball + "^000000 個";
				mes "正確的數字。你猜測的數字中";
				mes "有 ^ff0000" + .@strike + "^000000 個";
				mes "不僅數字正確，";
				mes "位置也正確。";
				mes "好了，再試一次吧?";
				next;
			} else {
				mes "[妮亞]";
				mes "你上一次的猜測有 ^ff0000" + .@ball + "^000000 個";
				mes "正確的數字，";
				mes "並且有 ^ff0000" + .@strike + "^000000 個";
				mes "數字不僅正確";
				mes "而且位置也正確。";
				switch(.@retry) {
					case 1: mes "你還剩下四次猜測機會。"; break;
					case 2: mes "你還剩下三次猜測機會。"; break;
					case 3: mes "你還剩下兩次猜測機會。"; break;
					case 4: mes "你只剩下最後一次猜測機會了..."; break;
					case 5: mes "答案是 ^ff0000" + .@yagu + "^000000。"; close;
				}
				next;
			}
		}
	}
}
