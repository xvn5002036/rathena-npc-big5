	next;

	emotion ET_THINK;
	next;
	switch(select("喔是嗎？聽起來不錯。:不了，謝謝妳，夫人。")) {
	case 1:
		if ((countitem(724) > 0) && (countitem(949) > 199) && (countitem(5001) > 0) && (Zeny > 4999)) {
			mes "[神秘女子]";

			delitem 5001,1; //Headset
			set Zeny,Zeny-5000;

			next;
			mes "[神秘女子]";

			emotion ET_THINK;
			close;
		}
	case 2:
		mes "[神秘女子]";
		mes "沒關係。";
		mes "我相信我們";
		mes "會再見面的。";
		mes "...總有一天。";

		close;
	}
}
