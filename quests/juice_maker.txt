payon_in03,188,146,5	script	馬克斯·漢森#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！ -";
		mes "- 你現在攜帶了 -";
		mes "- 太多物品了。 -";
		mes "- 請稍後再回來 -";
		mes "- 把一些物品存入卡普拉倉庫後再來。 -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人 馬克斯·漢森]";
		mes "歡迎。";
		mes "你也是來";
		mes "加工水果的嗎？";
		next;
		switch(select("製作果汁。:說話並獲取水果加工的資訊。:取消。")) {
		case 1:
			mes "[商人 馬克斯·漢森]";
			mes "你想製作哪一種果汁？";
			next;
			switch(select("蘋果汁:香蕉汁:胡蘿蔔汁:葡萄汁:取消")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人 馬克斯·漢森]";
				mes "那麼...";
				mes "下次見。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人 馬克斯·漢森]";
				mes "喔不...";
				mes "你沒有所有必要的材料。要製作 "+getitemname(.@juice)+"，我需要1個"+getitemname(.@fruit)+"和1個空瓶。我還需要 3 Zeny 的手續費。";
				next;
				mes "[商人 馬克斯·漢森]";
				mes "當你準備好所有東西後，我會把水果打成美味的果汁給你。";
				close;
			}

			mes "[商人 馬克斯·漢森]";
			mes "你想要多少瓶？";
			next;
			switch(select("盡可能多。:我想要指定的數量。:取消。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人 馬克斯·漢森]";
				mes "請選擇一個小於100的數字。如果你不想製作，請輸入 '0'。你最多可以製作 " + countitem(.@fruit) + " 瓶果汁。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人 馬克斯·漢森]";
						mes "那麼...";
						mes "歡迎下次光臨。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人 馬克斯·漢森]";
						mes "超過100瓶是不可能的。請選擇不同的數量。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人 馬克斯·漢森]";
				mes "那麼...";
				mes "歡迎下次光臨。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人 馬克斯·漢森]";
				mes "喔不...";
				mes "你沒有所有必要的材料。在這種情況下我無法幫你。我想你需要收集足夠的材料。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人 馬克斯·漢森]";
			mes "給！如約為你製作的新鮮美味果汁。它應該非常清爽可口。";
			next;
			mes "[商人 馬克斯·漢森]";
			mes "那麼...";
			mes "歡迎下次光臨。";
			close;
		case 2:
			mes "[商人 馬克斯·漢森]";
			mes "在人類能夠發展出如此龐大的社會之前，他們從樹上採集水果來生存。水果可能是大自然的恩賜，讓我們能夠存在於這個世界上。";
			next;
			mes "[商人 馬克斯·漢森]";
			mes "自從生活變得如此繁榮後，年輕一代似乎不再吃水果了。所以，我開始思考讓水果更容易食用的方法。";
			next;
			mes "[商人 馬克斯·漢森]";
			mes "我意識到如果把水果做成果汁，不僅更方便食用，而且味道也更好。";
			next;
			mes "# 果汁資訊 #";
			mes "^CC4E5C- 蘋果汁 -^000000";
			mes "蘋果 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^E3CF57- 香蕉汁 -^000000";
			mes "香蕉 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^ED9121- 胡蘿蔔汁 -^000000";
			mes "胡蘿蔔 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^CC00FF- 葡萄汁 -^000000";
			mes "葡萄 x 1 個，空瓶 x 1 個，3 Zeny。";
			close;
		case 3:
			mes "[商人 馬克斯·漢森]";
			mes "嘿！";
			mes "如果你拜訪";
			mes "某人，跟他們說話！";
			close;
		}
	}
	else {
		mes "[商人 馬克斯·漢森]";
		mes "歡迎。";
		mes "有什麼我可以幫忙的嗎？";
		next;
		if (select("說話:取消") == 1) {
			mes "[商人 馬克斯·漢森]";
			mes "在人類能夠發展出如此龐大的社會之前，他們從樹上採集水果來生存。水果可能是大自然的恩賜，讓我們能夠存在於這個世界上。";
			next;
			mes "[商人 馬克斯·漢森]";
			mes "自從生活變得如此繁榮後，年輕一代似乎不再吃水果了。所以，我開始思考讓水果更容易食用的方法。";
			next;
			mes "[商人 馬克斯·漢森]";
			mes "我意識到如果把水果做成果汁，不僅更方便食用，而且味道也更好。";
			close;
		}
		mes "[商人 馬克斯·漢森]";
		mes "嘿！";
		mes "如果你拜訪";
		mes "某人，跟他們說話！";
		close;
	}
}
