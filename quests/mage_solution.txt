	next;
	if (select("好的，成交。:不了，算了。") == 1) {
		mes "[魔法師公會人員]";
		if (Zeny < 50) {
			mes "嘿！你身上的錢不夠支付 50 Zeny 的費用。";
			close;
		}
		if (countitem(1092) == 0) {
			mes "沒有瓶子你怎麼裝水溶液！帶個空試管來給我吧。";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
