			mes "[古德曼]";
			mes "第一件事。我需要";
			mes "考驗你，看看你是否有資格";

			mes "[古德曼]";
			mes "嗯。你回來得";
			mes "太早了。而且你沒有";

			mes "[古德曼]";
			mes "去用任何物品";

			mes "重量 500 的物品都放不下為止。";
			mes "當你完成這個任務時，";
			mes "回來向我報告。";
			close;
		}
		mes "[古德曼]";
		mes "這麼快就回來啦？";
		mes "好吧，讓我檢查一下";
		mes "你帶著的東西。";
		mes "讓我看看，物品欄，物品欄...";
		next;
		mes "[古德曼]";
		mes "做得好：你及格了。";
		mes "你現在意識到，";
		mes "收集這些物品所花費的";
		mes "時間和經驗，比它們在";
		mes "Zeny 上的價格更有價值了嗎？";
		next;
		mes "[古德曼]";
		if (Upper == 1)
			mes "我相信神工匠";
		else
			mes "我相信鐵匠";
		mes "首先也必須是";
		mes "工匠。變得富有和出名";
		mes "只不過是附帶的效果。";

			mes "值得尊敬的神工匠知道價值在於手藝...";

			mes "值得尊敬的鐵匠知道價值在於手藝...";

	mes "[古德曼]";

	mes "為所有夥伴鍛造";
	mes "更高品質物品的過程中，";
	mes "能為你帶來便利。";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
