S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[瑪德蓮 丘]";
	mes "你要買幾個 "+ (.@item_id == 12125 ? "攜帶用":"家庭用");
	mes "料理器具？如果你";
	mes "想要取消，請";
	mes "輸入數字0。";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[瑪德蓮 丘]";
			mes "你改變主意了？";
			mes "好吧，如果之後需要";
			mes "買 "+ getitemname(.@item_id);
			mes "的話，隨時都可以回來";
			mes "找我，好嗎？";
			close;
		}
		else if (.@sell > 100) {
			mes "[瑪德蓮 丘]";
			mes "喔，很抱歉，但為了";
			mes "安全起見，我一次";
			mes "不賣超過100個 "+ getitemname(.@item_id) + "。";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[瑪德蓮 丘]";
		mes "喔，很抱歉，但你";
		mes "身上的錢不夠買這麼多";
		mes getitemname(.@item_id);
		mes "請在購買我的器具前";
		mes "確認一下你的Zeny~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[瑪德蓮 丘]";
		mes "很抱歉，但你的物品欄";
		mes "沒有足夠的空間";
		mes "容納這麼多";
		mes getitemname(.@item_id) +"......";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[瑪德蓮 丘]";
	mes "給你~";
	mes "祝你的料理訓練";
	mes "一切順利！";
	return;
}
