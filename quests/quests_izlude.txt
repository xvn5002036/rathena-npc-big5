// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[艾德加]";
		mes "所以你又要去艾爾貝塔了嗎？讓我給你跟上次一樣的折扣，只收你 250 Zeny。";
		mes "你覺得如何呢？";
		next;
		if (select("好哇?！:你為什麼對我這麼好？！") == 1) {
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯...";
				mes "這筆錢";
				mes "不夠喔。";
				mes "你要不要去";
				mes "多拿點錢來？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "我這人就是這樣。而且，你那惡魔般的微笑讓我想起了我心愛的金髮兒子，他多年前離家，到自己的船上當水手去了。願神保佑他，無論他在哪裡。";
		close;
	}
	mes "[艾德加]";
	mes "我的城鎮，伊斯魯得島，西邊有著通往艾爾貝塔的港口。我們兩地之間的交通非常頻繁，我幾乎都要變成艾爾貝塔人了。呵呵呵?";
	next;
	mes "[艾德加]";
	mes "我認識一個住在艾爾貝塔叫菲利克斯的傢伙，跟他還挺熟的。那傢伙真的很小氣...他什麼都要收錢！";
	next;
	mes "[艾德加]";
	mes "但他其實是個好人，喜歡幫助別人。他有一副好心腸，只要你付得起他的價錢，他就會協助你。";
	next;
	mes "[艾德加]";
	mes "最近，艾爾貝塔的人說他真的在努力幫助大家，而他要求傑勒比結晶只是個幌子。嗯，你應該去看看他能提供什麼。";
	next;
	if (select("你能告訴我怎麼去艾爾貝塔嗎？:結束對話") == 1) {
		mes "[艾德加]";
		mes "咦？這個嘛，你可以靠你的雙腳走過去。但如果你有錢，我建議你搭船。";
		next;
		if (select("好，我知道了。:但我不想走路，而且我身無分文！") == 1) {
			mes "[艾德加]";
			mes "好的，保重囉?";
			close;
		}
		mes "[艾德加]";
		mes "好吧...";
		mes "你不想走路『而且』你沒有錢，但你還是想去那裡？喔，天啊...";
		next;
		mes "[艾德加]";
		mes "好吧好吧。身為船長的我，可以負擔得起用較低的價格載你過去。250 Zeny 你覺得如何？";
		next;
		if (select("好哇?！:呸，真是坑人！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯...";
				mes "這筆錢";
				mes "不夠喔。";
				mes "去多準備";
				mes "一些再來吧。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "天啊，";
		mes "如果你覺得";
		mes "這是在坑人...";
		close;
	}
	mes "[艾德加]";
	mes "嗯，好吧。";
	mes "回頭見?";
	close;
}
