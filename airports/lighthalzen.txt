lhz_airport,143,43,5	script	機場服務員#lhz_air1a::lhz_airport1	90,{
	mes "[機場服務員]";
	mes "歡迎來到";
	mes "里希塔樂鎮機場，";
	mes "我們提供直達";
	mes "艾因布羅克、尤諾和毀葛的航班。";
	next;
	if (select("搭乘飛空艇。:取消。") == 1) {
		mes "[機場服務員]";
		mes "搭乘費用是";
		mes "1,200 zeny，但如果您";
		mes "使用免費飛空艇搭乘券，";
		mes "就可以免費搭乘。";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[飛空艇服務員]";
			mes "很抱歉，但您身上";
			mes "沒有足夠的 1,200 zeny";
			mes "來支付搭乘費用。";
			close;
		}
	}
	mes "[機場服務員]";
	mes "謝謝您，";
	mes "歡迎下次光臨。";
	mes "祝您有美好的一天?";
	close;
}
