y_airport,143,43,5	script	機場服務員#y_air1a::y_airport1	90,{
	mes "[機場服務員]";
	mes "歡迎來到尤諾機場，我們提供前往艾因布羅克、里希塔樂鎮和毀葛的國內航班，";
	mes "以及前往伊斯魯得島和拉赫的國際航班。";
	mes "請問有什麼我可以為您服務的嗎？";
	next;
	if (select("搭乘飛空艇。:取消。") == 1) {
		mes "[機場服務員]";
		mes "所有航班的";
		mes "搭乘費用皆為 1,200 zeny。如果您";
		mes "使用免費飛空艇搭乘券，";
		mes "就可以免費搭乘。那麼，您想要搭乘嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[機場服務員]";
			mes "很抱歉，但您身上";
			mes "沒有足夠的 1,200 zeny";
			mes "來支付搭乘費用。";
			close;
		}
	}
	mes "[機場服務員]";
	mes "謝謝您，";
	mes "祝您有美好的一天。";
	close;
}
