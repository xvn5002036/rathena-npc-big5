airport,143,43,5	script	機場服務員#airport1a::airport1	90,{
	mes "[機場服務員]";
	mes "歡迎來到";
	mes "艾因布羅克機場，";
	mes "我們提供直達";
	mes "尤諾、里希塔樂鎮以及毀葛";
	mes "的航班。";
	next;
	if (select("搭乘飛空艇:取消") == 1) {
		mes "[機場服務員]";
		mes "飛空艇的搭乘費用";
		mes "是 1,200 zeny，但如果您";
		mes "有免費飛空艇搭乘券，";
		mes "就可以免費搭乘。您";
		mes "要搭乘飛空艇嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[機場服務員]";
			mes "很抱歉，您沒有";
			mes "免費飛空艇搭乘券，";
			mes "且身上沒有足夠的";
			mes "zeny 來搭乘飛空艇。";
			mes "請記住，搭乘費用是 1,200 zeny。";
			close;
		}
	}
	mes "[機場服務員]";
	mes "謝謝您，";
	mes "祝您有美好的一天。";
	close;
}
