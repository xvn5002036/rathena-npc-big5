hugel,178,142,0	script	往飛空艇#hugel	45,1,1,{
	end;
OnTouch:
	mes "搭乘飛空艇需要支付 1,200 zeny 或是使用免費飛空艇搭乘券。";
	mes "您想要搭乘嗎？";
	next;
	if (select("是:否") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1;
//Free_Flying_Ship_Ticket
			warp "airplane",244,58;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane",244,58;
			end;
		}
		mes "很抱歉，但您身上的錢不夠。";
		mes "請記住，搭乘飛空艇需要支付 1,200 zeny。";
		close;
	}
	mes "謝謝您，歡迎下次光臨。";
	close;
}
