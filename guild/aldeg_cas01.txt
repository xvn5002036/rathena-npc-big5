// Guild Flags (Outside Castle)
//============================================================
-	script	諾伊施萬石#1::OutsideFlagsA1	722,{

	set .@GID, GetCastleData("aldeg_cas01",1);

	if (.@GID == 0) {
		mes "[ 神聖盧恩米德加茲王國詔書 ]";
		mes " ";
		mes "1. 遵循神聖盧恩米德加茲王國的法令，";
		mes "我們宣布";
		mes "這座城池目前沒有正式的主人。";
		mes " ";
		mes "2. 凡是能夠";
		mes "克服所有考驗";
		mes "並擊碎華麗金屬的人，";
		mes "國王將賦予他";
		mes "這座城池的所有權。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 迴盪的聲音 ]";
		mes "勇敢的人們...";
		mes "你們希望回到屬於你們的榮耀之地嗎？";
		next;
		if(select("回到公會城池。:離開。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("aldeg_cas01",1)) {
				warp "aldeg_cas01",218,170;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神聖盧恩米德加茲王國詔書 ]";
	mes " ";
	mes "1. 遵循神聖盧恩米德加茲王國的法令，";
	mes "我們承認這個地方";
	mes "歸 ^ff0000"+GetGuildName(.@GID)+"^000000 公會私有。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公會的會長是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如果有人對此有異議，";
	mes "請用你手中的鋼刃來證明你的實力與榮耀。";
	close;

OnRecvCastleA01:
	FlagEmblem GetCastleData("aldeg_cas01",1);
	end;
}
