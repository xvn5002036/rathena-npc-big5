// Guild Flag (Outside Castle)
//============================================================
-	script	艾歐布里加#1::OutsideFlagsG2	722,{

	set .@GID, GetCastleData("gefg_cas02",1);

	if (.@GID == 0) {
		mes "[ 盧恩米德加茲王國 神聖的公告 ]";
		mes " ";
		mes "1. 遵從神聖的盧恩米德加茲王國的法令，";
		mes "我們在此宣布";
		mes "這座城池目前沒有正式的主人。";
		mes " ";
		mes "2. 對於那些能夠";
		mes "克服所有試煉";
		mes "並摧毀華麗金屬的人，";
		mes "國王將賦予其";
		mes "這座城池的所有權。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 迴盪的聲音 ]";
		mes "勇敢的人們...";
		mes "你們希望回到那充滿榮耀的地方嗎？";
		next;
		if(select("回到公會城池。:離開。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("gefg_cas02",1)) {
				warp "gefg_cas02",178,43;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 盧恩米德加茲王國 神聖的公告 ]";
	mes " ";
	mes "1. 遵從神聖的盧恩米德加茲王國的法令，";
	mes "我們承認這個地方是";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公會的私人領地。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公會的會長是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如果有人對此有異議，";
	mes "請用你手中的鋼刃來證明你的力量與榮耀。";
	close;

OnRecvCastleG02:
	FlagEmblem GetCastleData("gefg_cas02",1);
	end;
}
