morocc,208,90,6	script	包頭巾的盜賊	58,{
	mes "[包頭巾的盜賊]";
	if(event_umbala < 2) {
		mes "你想找我做什麼？快滾！";
		close;
	}
	if(sphmask_q) {
		mes "你跟我已經沒瓜葛了，走開！";
		close;
	}
	mes "哈囉我的朋友，你對這面稀有的面具感興趣嗎？我向你保證它的價值絕對是真的，我的朋友。這可是極品啊！";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "[包頭巾的盜賊]";
	mes "呃！算了！我去別的地方能賣得更好！";
	set sphmask_q,1;
	close;

L_Menu:
	mes "[包頭巾的盜賊]";
	switch(getarg(0)) {
		case 1: mes "哇？喔呵呵呵，所以你知道這是我從那些野蠻人那裡偷來的，對吧？我可不會還回去。不過如果你肯出個好價錢... 你覺得如何？";
		break;
		case 2: mes "啊，你還真是個精明的生意人，對吧？ 好吧，那這樣如何..."; break;
		case 3: mes "嗯... 你真會殺價，好吧... 我最後的報價..."; break; }
	next;
	if(select("支付 "+getarg(1)+"z:不成交")==2) return;
	mes "[包頭巾的盜賊]";
	if (Zeny < getarg(2)) {
		mes "你在耍我嗎？你根本沒那麼多錢！";
		close;
	}
	set Zeny, Zeny-getarg(2);
	mes "喔呵呵，那就成交啦！";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
