prontera,158,356,4	script	皇家信使	105,{
	
	setarray .@Req1[0],1161,999,120,984,10,996,10,994,10,995,10,997,10;
	setarray .@Req2[0],5013,2298,1,1001,10,996,10,994,10,995,10,997,10;
	
	mes "[皇家信使]";
	mes "歡迎來到普隆德拉。";
	mes "我是負責發布皇家任務的信使。";
	next;
	if(select("聽聽看...:走開")==2) close;
	mes "[皇家信使]";
	mes "這裡有兩個任務。";
	mes "選擇一個你喜歡的吧。";
	next;
	set .@i, select("巴爾蒙克:太陽神頭盔:^777777算了。^000000");
	mes "[皇家信使]";
	if (.@i==3) {
		mes "祝你有個美好的一天。";
		close; }
	mes "你需要以下的物品：";
	for(set .@j,1; .@j<13; set .@j,.@j+2) {
		mes " ~ "+getd(".@Req"+.@i+"["+(.@j+1)+"]")+"個 ^0055FF"+getitemname(getd(".@Req"+.@i+"["+.@j+"]"))+"^000000";
		if (!.@nr && countitem(getd(".@Req"+.@i+"["+.@j+"]")) < getd(".@Req"+.@i+"["+(.@j+1)+"]")) set .@nr,1;
	}
	mes " ~ 1,000,000 Zeny";
	next;
	mes "[皇家信使]";
	if (.@nr || Zeny < 1000000) {
		mes "當你滿足所有條件後再來找我吧。";
		close; }
	for(set .@j,1; .@j<13; set .@j,.@j+2)
		delitem getd(".@Req"+.@i+"["+.@j+"]"), getd(".@Req"+.@i+"["+(.@j+1)+"]");
	set Zeny, Zeny-1000000;
	mes "我看到你已經準備好所有需要的物品了。這是答應給你的 "+getitemname(getd(".@Req"+.@i+"[0]"))+"。";
	getitem getd(".@Req"+.@i+"[0]"),1;
	close;
}
