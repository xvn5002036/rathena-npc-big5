lighthalzen,108,229,2	script	拉芬	766,{
	mes "[拉芬]";
	if(BaseLevel<99) {
		if(Q_99LVL) {
			mes "你... 你變年輕了！！！但你絕對就是你！";
			mes "我沒想到你真的能達成我的條件...";
			emotion ET_SCRATCH;
			next;
			mes "[拉芬]";
			mes "這是給你的獎勵。它永遠屬於你了。";
			set Q_99LVL,0;
			getitem 12106,1; //Jewel Case
			emotion ET_CONGRATULATION;
		} else {
			mes callfunc("F_Hi");
			mes "...";
			emotion ET_SLEEPY;
			next;
			mes "[拉芬]";
			mes "我有一個謎題，專門出給達到2轉且 Base Level 99 的強者。";
			mes "*打哈欠* 如果你遇到這樣的人，請告訴他們我的謎題。";
		}
		close;
	}
	if(BaseLevel==99 && Upper==0) {
		if(Q_99LVL) {
			mes "現在去吧，讓自己變年輕！";
			mes "如果你解開了我的謎題，我就會給你一個裝著稀有裝飾品的神奇珠寶盒。";
		} else {
			mes callfunc("F_Hi")+" 英雄！";
			if(BaseExp<1000000) {
				mes "現在你很強了。";
				mes "但是... 要解開我的謎題還不夠強。";
				if(rand(2)) {
					mes "再去擊殺 "+((1000000-BaseExp)/3765+1)+" 隻海豹寶寶，然後再回來吧。";
				} else {
					mes "再去擊殺 "+((1000000-BaseExp)/81+1)+" 隻波波利，然後再回來吧。";
				}
			} else {
				mes "現在你真的很強。";
				mes "如果你變得比現在更年輕，我就會把我的寶物給你。";
				set Q_99LVL,1;
				emotion ET_BEST;
			}
		}
		close;
	}
	emotion ET_SCRATCH;
	mes "...";
	close;
}
