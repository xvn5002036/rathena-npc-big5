prt_castle,80,170,4	script	普隆德拉國王	108,{
	mes "[普隆德拉國王]";
	mes "你好，你能幫我個忙嗎？";
	next;
	if(select("當然，我能做什麼？:抱歉，我很忙。")==2) {
		mes "[普隆德拉國王]";
		mes "好吧，沒關係，我相信會有其他人願意幫忙的。";
		close; }
	mes "[普隆德拉國王]";
	mes "謝謝你。是這樣的，我旁邊的女兒有點吵鬧...";
	mes "如果你能幫她弄到這些物品，我相信她會冷靜下來的。";
	mes "我會把我的王冠或是劍作為獎勵送給你。";
	next;
	setarray .@Items1[0],5007,742,30,741,50,753,20,732,1,710,20,629,20;
	setarray .@Items2[0],1137,2254,1,2282,1,714,1;
	switch(select("你需要什麼？:我已經準備好物品了！:算了吧。")) {
		case 1:
			for(set .@i,1; .@i<3; set .@i,.@i+1) {
				mes "[普隆德拉國王]";
				mes "如果你能幫我弄到：";
				for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
					mes " ~ ^0055FF"+getd(".@Items"+.@i+"["+(.@j+1)+"]")+"^000000個 "+getitemname(getd(".@Items"+.@i+"["+.@j+"]"));
				mes "我就會把我的"+((.@i == 2 )?"劍。":"王冠。");
				if (.@i==2) close;
				next; }
		case 2:
			mes "[普隆德拉國王]";
			mes "你想要哪個物品？";
			next;
			set .@i, select("王冠:劍");
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				if (countitem(getd(".@Items"+.@i+"["+.@j+"]")) < getd(".@Items"+.@i+"["+(.@j+1)+"]")) set .@nr,1;
			mes "[普隆德拉國王]";
			if (.@nr) { 
				mes "你少了一些東西。";
				mes "等你收集齊所有的物品後再來吧。";
				close; }
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				delitem getd(".@Items"+.@i+"["+.@j+"]"), getd(".@Items"+.@i+"["+(.@j+1)+"]");
			getitem getd(".@Items"+.@i+"[0]"),1;
			mes "非常感謝你！！";
			mes "這一定能讓她冷靜下來。";
			mes "照約定，這是我的"+((.@i==1)?"王冠。":"劍。");
			close;
		case 3:
			mes "[普隆德拉國王]";
			mes "好吧，沒關係，我相信會有其他人願意幫忙的。";
			close;
	}
}
