alberta,88,193,5	script	太陽眼鏡商人	73,{
	mes "[太陽眼鏡商人]";
	if(SG_QUEST1 == 1) {
		mes "我沒有什麼能再告訴你的了。";
		next;
		mes "[太陽眼鏡商人]";
		mes "去找瑪瑟夫（Maseph）。她就在夢羅克北邊的某處。";
		close;
	}
	mes "你好。有什麼我可以幫你的嗎？";
	next;
	if(select("我聽說你可以製作 ^0000FF太陽眼鏡[1]^000000。:沒事，抱歉打擾了。")==2) {
		mes "[太陽眼鏡商人]";
		mes "等你準備好再來找我吧。";
		close; }
	mes "[太陽眼鏡商人]";
	mes "我不會做，但我可以告訴你去哪裡找會做的人。只要一點點小費...";
	next;
	if(select("多少錢？:想得美，我自己會去找她！")==2) {
		mes "[太陽眼鏡商人]";
		mes "隨你便，除非製作人知道你要去，否則她是不會幫你做 ^0000FF太陽眼鏡[1]^000000 的。";
		mes "只有我能事先通知她你要過去。";
		next;
		mes "[太陽眼鏡商人]";
		mes "等你放棄了再來找我吧。哈哈哈...";
		close; }
	mes "[太陽眼鏡商人]";
	mes "為了讓我告訴你關於 ^0000FF太陽眼鏡[1]^000000 的情報，你需要給我：";
	mes " - ^0000881個 1克拉鑽石^000000";
	mes " - ^00008850個 羽毛^000000";
	mes " - ^000088100000z^000000";
	next;
	if(select("好吧，給你。:這太貴了吧！")==2) {
		mes "[太陽眼鏡商人]";
		mes "隨你便。";
		close; }
	mes "[太陽眼鏡商人]";
	if (countitem(730) < 1 || countitem(949) < 50 || Zeny < 100000) {
		mes "你少了一些東西。等你收集齊所有的材料後再來吧。";
		close; }
	delitem 730,1;
	delitem 949,50;
	set Zeny, Zeny-100000;
	set SG_QUEST1,1;
	mes "[太陽眼鏡商人]";
	mes "太好了。現在，仔細聽好。";
	next;
	mes "[太陽眼鏡商人]";
	mes "在夢羅克北邊的某處，尋找一位名叫瑪瑟夫（Maseph）的人。";
	mes "我會傳個訊息給她，讓她知道你要過去。";
	close;
}
