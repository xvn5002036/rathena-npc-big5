prontera,160,155,4	script	變裝活動	795,{

	// Currently set to run every two hours.
	// To change times, edit the OnClock labels below.

	set .@GMLevel,60;	// GM level required to access NPC.
	set .@n$,"[^0000FF變身猜謎 NPC^000000]";

	if (getgmlevel()>=.@GMLevel) {
		mes .@n$;
		mes "請選擇一個選項。";
		next;
		switch(select("開啟/關閉 活動:活動設定")) {
		case 1:
			mes .@n$;
			if (.EventON) {
				mes "活動目前狀態：[^0000FF開啟^000000]";
				mes "請問您想要關閉它嗎？";
			} else {
				mes "活動目前狀態：[^FF0000關閉^000000]";
				mes "請問您想要開啟它嗎？";
			}
			if(select("是:否")==2) close;
			if (.EventON) {
				set .EventON,0; set .Timer,0;
				setnpctimer 0; stopnpctimer;
				announce "GM 已關閉變身猜謎活動。因此將不再發放獎品。",bc_map |
				bc_blue;
				deletepset 1;
				setnpcdisplay "變裝活動",795;
				close;
			}
			set .EventON,1; set .Timer,1; setnpctimer 0; initnpctimer;
			set .ResetCounter,.ResetCounter+1;
			announce "變身猜謎活動將於 3 分鐘後開始。",bc_all | bc_blue;
			announce "活動將在普隆德拉舉行。",bc_all |
			bc_blue;
			close;
		case 2:
			mes .@n$;
			mes "請選擇要修改的設定。";
			next;
			switch(select("怪物顯示設定:回合數設定:獎品設定")) {
			case 1:
				setarray .@r$[0],"變身為所有怪物。","僅變身為 MVP。";
				mes .@n$;
				mes "請選擇變身規則。";
				next;
				set .Rule, select(implode(.@r$,":"));
				mes .@n$;
				mes "變身規則已設定為：";
				mes "  > ^0055FF"+.@r$[.Rule-1]+"^000000";
				close;
			case 2:
				mes .@n$;
				mes "請輸入您希望活動持續的回合數。";
				mes "目前回合數：[^0000FF"+.Rounds+"^000000]";
				next;
				input .@Rounds;
				set .Rounds,.@Rounds;
				mes .@n$;
				mes "回合數已更改為 "+.Rounds+"。";
				close;
			case 3:
				mes .@n$;
				mes "請輸入每回合發放的獎品物品 ID。";
				mes "目前物品：[^0000FF"+getitemname(.Prize)+"^000000] (ID #"+.Prize+")";
				next;
				input .@Prize;
				mes .@n$;
				if (getitemname(.@Prize)=="" || getitemname(.@Prize)=="null") {
					mes "該物品不存在。請再試一次。";
					close;
				}
				set .Prize,.@Prize;
				mes "請輸入要發放的數量。";
				next;
				input .@amount;
				mes .@n$;
				if (.@amount<=0 || .@amount>=10000) {
					mes "數量無效。將使用預設數量 1。";
					set .@amount,1;
					next;
					mes .@n$;
				}
				set .PrizeAmt,.@amount;
				mes "獎品已成功更改。";
				mes "獎品："+.PrizeAmt+" 個 [^0000FF"+getitemname(.Prize)+"^000000]";
				close;
			}
		}
	}
	if (.EventON) end;
	mes .@n$;
	mes "歡迎。";
	mes "有什麼我可以幫忙的嗎？";
	if(select("活動資訊:沒事，只是路過。")==2) close;
	next;
	mes .@n$;
	mes "這個活動非常簡單。";
	mes "活動開始時，我會";
	mes "隨機變身成一種怪物。";
	mes "你必須大聲喊出";
	mes "那隻怪物的名字。";
	next;
	mes "如果猜對了，你就會獲得";
	mes "一份獎品。如果猜錯了，請繼續嘗試！";
	mes "這就是活動的全部內容。";
	close;

OnInit:
	set .EventON,0;
	set .Wait,0;
	set .Winner,0;
	set .ResetCounter,0;
	set .Rounds,10;
	set .Prize,512;
	set .PrizeAmt,1;
	set .Rule,1;
	setarray .MVP[0],1038,1039,1046,1059,1086,1087,1112,1115,1147,1150,1157,1159,1190,1251,1252,1272,1312,1373,
	                  1389,1399,1418,1492,1502,1511,1583,1623,1630,1646,1647,1648,1649,1650,1651,1658,1685,1688,
				   1708,1719,1734,1751,1768,1779,1785,1802,1832,1871,1874,1885,1917,1980,2022,2068,2087,2131,
				   2156,2165;
	set .BlackList$, "1003,1006,1017,1021,1022,1027,1043,1075,1136,1137,1168," + 
	    "1171,1172,1173,1181,1187,1210,1217,1218,1222,1223,1224,1225,1226,1227,1228," + 
	    "1233,1284,1407,1411,1414,1495,1501,1900,1996,2000,2001,2002,2003,2004," + 
	    "2005,2006,2007,2011,2012,2025,2028,2029,2030,2031,2032,2033,2034,2035," + 
	    "2036,2037,2038,2039,2040,2041,2042,2043,2044,2045,2046,2047,2048,2049," + 
	    "2050,2051,2052,2053,2054,2055,2056,2057,2058,2059,2060,2061,2062,2063," + 
	    "2064,2065,2066,2067,2075,2076,2077,2078,2079,2080,2081,2083,2084,2085," + 
	    "2086,2087,2088,2089,2090,2091,2092,2093,2094,2095,2096,2097,2098,2099," + 
	    "2100,2101,2012,2103,2104,2105,2106,2107,2108,2109,2110,2111,2112,2113," + 
	    "2114,2115,2116,2117,2118,2119,2120,2121,2123,2124,2125,1496,";
	end;
OnClock0000:
OnClock0200:
OnClock0400:
OnClock0600:
OnClock0800:
OnClock1000:
OnClock1200:
OnClock1400:
OnClock1600:
OnClock1800:
OnClock2000:
OnClock2200:
	set .ResetCounter,.ResetCounter+1;
	set .EventON,1;
	set .Timer,1;
	set .Wait,1;
	announce "變身猜謎活動將於 3 分鐘後開始。",bc_all | bc_blue;
	announce "活動將在普隆德拉舉行。",bc_all | bc_blue;
	setnpctimer 0;
	initnpctimer;
	end;
OnTimer10000:
	if (.Timer || .Change) end;
	set .Wait,0;
	goto iDisguise;
	end;
OnTimer30000:
	if (.Timer) end;
	set .Change,0;
	setnpcdisplay "變裝活動",795;
	npctalk "變身猜謎 NPC : 大家猜太久囉。請稍等 10 秒，我將再次變身。";
	specialeffect EF_DETECT2;
	set $MonsterName$,"";
	deletepset 1;
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;
OnTimer60000:
	if (.Timer!=1) end;
	announce "變身猜謎活動將於 2 分鐘後開始。",bc_all | bc_blue;
	announce "活動將在普隆德拉舉行。",bc_all |
	bc_blue;
	end;
OnTimer120000:
	if (.Timer!=1) end;
	announce "變身猜謎活動將於 1 分鐘後開始。",bc_all | bc_blue;
	announce "活動將在普隆德拉舉行。",bc_all | bc_blue;
	end;
OnTimer180000:
	if (.Timer!=1) end;
	announce "變身猜謎活動正式開始！",bc_all | bc_blue;
	announce "活動將在普隆德拉舉行。",bc_all | bc_blue;
	set .Timer,0;
	stopnpctimer;
	setnpctimer 0; initnpctimer;
iDisguise:
	if (.Rule==1) {
		set .Winner,0;
		set .Monster,1000+rand(1,995);
		if (compare(","+.BlackList$+"," , ","+.Monster+",")) goto iDisguise;
		if (.Monster==.LastMonster) goto iDisguise;
		set .LastMonster,.Monster;
		set $MonsterName$,getmonsterinfo(.Monster, MOB_NAME);
	}
	if (.Rule==2) {
		set .Winner,0;
		set .Monster,rand(49);
		set $MonsterName$,getmonsterinfo(.MVP[.Monster], MOB_NAME);
	}
	deletepset 1;
	defpattern 1,"([^:]+):.\\s*"+$MonsterName$+".*", "iCorrect";
	activatepset 1;
	if (.Rule==1) setnpcdisplay "變裝活動",.Monster;
	if (.Rule==2) setnpcdisplay "變裝活動",.MVP[.Monster];
	set .Change,1;
	setnpctimer 0;
	end;
iCorrect:
	if (.Winner) {
		dispbottom "已經有人贏得這回合了。";
		end;
	}
	set .Winner,1;
	set .RoundCount,.RoundCount+1;
	deletepset 1;
	activatepset 1;
	getitem .Prize,.PrizeAmt;
	announce strcharinfo(0)+" 猜對了！我剛才變身的是："+$MonsterName$+"",bc_map | bc_blue;
	if (.RoundCount>=.Rounds) {
		setnpcdisplay "變裝活動",795;
		set .RoundCount,0; set .Change,0;
		set .EventON,0;
		setnpctimer 0; stopnpctimer;
		npctalk "變身猜謎 NPC : 感謝大家的參與。這已經是變身猜謎活動的最後一回合了。下次再來玩吧。";
		end;
	}
	setnpcdisplay "變裝活動",795;
	set .Change,0;
	setnpctimer 0;
	end;
}
