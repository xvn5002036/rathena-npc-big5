prontera,3,3,3	script	不安的檢查	-1,{
	end;

OnHour00:
	set $UNEASY_DL,$UNEASY_DL-1;
	set $UNEASY_BL,$UNEASY_BL+30; //add need of HW for 30 bottles per day
	if ($UNEASY_BL>666) set $UNEASY_BL,666;
	//keep needed bottles not <=666
	if ($UNEASY_DL < 0) goto L_Start_Undead;
//The Cemetery is OK yet.
	disablenpc "瑪莎娜修女";
	end;
OnInit:
	if ($UNEASY_DL >= 0) disablenpc "瑪莎娜修女";
	end;

OnHour06:
	killmonsterall "prontera";	//The Sun kills undead in the morning
	end;
OnHour01:
	if ($@UNEASY_MOB > 0) mapannounce "prontera","[瑪莎娜修女]: 以奧丁之名，請消滅這些四處遊蕩的殘餘不死生物！",0;
	end;
OnZombieDead:
	set $@UNEASY_MOB,$@UNEASY_MOB-1;
	if ($@UNEASY_MOB>0) end;
	set $UNEASY_DL,0;
	set $UNEASY_H$,strcharinfo(0);
	if (Sex==SEX_MALE) mapannounce "prontera","[瑪莎娜修女]: 勇敢的 "+$UNEASY_H$+" 剛剛消滅了普隆德拉的最後一隻不死生物！",0;
	else mapannounce "prontera","[瑪莎娜修女]: 淑女 "+$UNEASY_H$+" 剛剛消滅了普隆德拉的最後一隻不死生物！",0;
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	end;
L_Start_Undead:
	killmonsterall "prontera";	//kills any left monsters
	enablenpc "瑪莎娜修女";
//call some monsters in the city
	set $@UNEASY_MOB, 65;
	areamonster "prontera",0,0,0,0,"Zombie",1015,30,"不安的檢查::OnZombieDead";
//in the Cemetery
	monster "prontera",268,349,"殭屍",1015,30,"不安的檢查::OnZombieDead";
	monster "prontera",269,350,"腐屍",1036,5,"不安的檢查::OnZombieDead";
//announce
	mapannounce "prontera","[瑪莎娜修女]: 墓地變得焦躁不安了！以奧丁之名，快點前往聖域！拯救普隆德拉城！",0;
}

prontera,257,313,5	script	瑪莎娜修女	79,{
	mes "[瑪莎娜修女]";
	if ($UNEASY_DL <= 0) goto L_Undead_Walk;
	mes "恐怕我們古老的墓地出了點問題...";
	if ($UNEASY_H$==strcharinfo(0)) mes "但多虧了你，"+$UNEASY_H$+"，我們接下來的 " + $UNEASY_DL + " 個夜晚都能安穩入睡了！";
	if ($UNEASY_H$!=strcharinfo(0)) mes "但多虧了 "+$UNEASY_H$+" 的幫忙，我們換來了 " + $UNEASY_DL + " 個安穩的夜晚！";
	emotion ET_SURPRISE;
	close;
L_Undead_Walk:
	if ($UNEASY_DL == 0) mes "他們明晚可能還會再回來！";
	if ($UNEASY_DL == 0 && $UNEASY_H$==strcharinfo(0)) mes "謝謝你，"+$UNEASY_H$+"！現在我們可以安息到下一個午夜了！";
	if ($UNEASY_DL == 0 && $UNEASY_H$!=strcharinfo(0)) mes "因為有 "+$UNEASY_H$+" 的幫忙，我們可以安穩休息到下個午夜了！";
	mes "為了平息焦躁不安的墓地，我們必須在這些墳墓上灑滿聖水。但是我們的弟兄姐妹們已經把聖水用光了。";
	mes "你能提供我們一些聖水嗎？";
	next;
	menu "好，我所有的聖水都給妳！",-, "不行，我自己也需要。",M_NO, "我身上沒有聖水。",M_DONT_HAVE;
	if ( countitem(523)<1 ) goto M_DONT_HAVE;
	set $UNEASY_BL,$UNEASY_BL-countitem(523);
	delitem 523,countitem(523);

	if ( $UNEASY_BL > 0 ) goto L_NEED_MORE;
//set quiet days!!! no more undead for this period!
	set $UNEASY_DL,5+((0-$UNEASY_BL)/30);
	set $UNEASY_H$,strcharinfo(0);
	mes "[瑪莎娜修女]";
	mes "謝謝你，"+$UNEASY_H$+"！現在我們有足夠的聖水了！";
	next;
	mes "[瑪莎娜修女]";
	mes "在墓地灑上這些聖水後，我們將獲得 " + $UNEASY_DL + " 個安全的夜晚！";
	next;
	killmonsterall "prontera";
//kills any left monsters
	mes "[瑪莎娜修女]";
	mes "看到了嗎，"+ $UNEASY_H$ +"？他們現在都消失了！";
	next;
	mes "[瑪莎娜修女]";
	mes "我們的教會將會親自向你致謝...";
	next;
	if (Sex==SEX_MALE) mapannounce "prontera","[瑪莎娜修女]: 以奧丁之名，我們宣布英俊的 "+$UNEASY_H$+" 成為普隆德拉的救世主！",0;
	else mapannounce "prontera","[瑪莎娜修女]: 以奧丁之名，我們宣布美麗的 "+$UNEASY_H$+" 成為普隆德拉的救世主！",0;
	mes "[瑪莎娜修女]";
	mes "以奧丁之名，我們為你祈福，並鄭重獻上這份來自馬雷烏西斯酒窖的微薄禮物。";
	getitem 505,1; //Blue_Potion
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	close;

L_NEED_MORE:
	mes "[瑪莎娜修女]";
	mes "謝謝你，好心的 "+strcharinfo(0)+"，但我們還需要 " + $UNEASY_BL + " 瓶聖水。";
	close;
M_NO:
	mes "[瑪莎娜修女]";
	mes "恐怕古老的墓地很快就會失控... 請盡量為我們帶回所有的聖水。";
	close;

M_DONT_HAVE:
	mes "[瑪莎娜修女]";
	mes "唉！我們還需要 " + $UNEASY_BL + " 瓶聖水... 你何不去向其他人要點多餘的聖水呢？";
	mes "古老的墓地很快就會失控了...";
	mes "請以奧丁之名，幫助普隆德拉城吧。";
	close;
}
