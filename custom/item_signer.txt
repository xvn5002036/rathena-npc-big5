prt_in,24,61,7	script	佩爾契克	47,{
	
	setarray .@Item[0],644,3;	// Item requirements: <ID>,<Count>{,...} (0 to disable)
	setarray .@Cost[0],0,5000;
	// Zeny requirements: <base price>,<price per refine>

	disable_items;
	mes "[佩爾契克]";
	mes "我可以在你持有的幾乎任何稀有物品上 ^0055FF簽上你的名字^000000。";
	next;
	if(select("Tell me more...:Sign my items, please!") == 1) {
		mes "[佩爾契克]";
		mes "我可以在任何沒有插槽的裝備或武器上簽上你的名字。";
		emotion ET_AHA;
		next;
		mes "[佩爾契克]";
		if (getarraysize(.@Item) || getarraysize(.@Cost)) {
			mes "作為報酬我需要：";
			if (getarraysize(.@Item)) for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2)
				mes " ~ "+.@Item[.@i+1]+" 個 "+getitemname(.@Item[.@i]);
			if (.@Cost[0]) mes " ~ "+callfunc("F_InsertComma",.@Cost[0])+" Zeny";
			if (.@Cost[1]) mes " ~ 每精煉一次 "+callfunc("F_InsertComma",.@Cost[1])+" Zeny";
		}
		else mes "我免費為你服務，但是...";
		next;
		emotion ET_CRY;
		mes "[佩爾契克]";
		mes "唉，我有 12 個嗷嗷待哺的孩子";
		mes "和一個非常生氣的老婆。";
		next;
		mes "[佩爾契克]";
		mes "或者應該說是 12 個生氣的孩子";
		mes "和一個非常飢餓的老婆...";
		emotion ET_SCRATCH;
		close;
	}
	mes "[佩爾契克]";
	mes "讓我看看你要簽名的物品...";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i]))
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) +" [^0055FF"+getequipname(.@indices[.@i])+"^000000]";
		set .@menu$, .@menu$+":";
	}
	set .@part, .@indices[ select(.@menu$) ];
	set .@id, getequipid(.@part);
	set .@ref, getequiprefinerycnt(.@part);
	mes "[佩爾契克]";
	if (!getequipisequiped(.@part)) {
		mes "那個部位沒有裝備任何東西！";
		emotion ET_KEK;
		close;
	}
	for(set .@i,0; .@i<4; set .@i,.@i+1)
		set .@slot[.@i], getequipcardid(.@part,.@i);
	if (.@slot[0]==255 || .@slot[0]==254 || .@slot[0]<0) {
		mes "唉，這件物品已經簽過名了。";
		mes "我絕對不會去更動大師的作品。";
		emotion ET_SCRATCH;
		close;
	}
	if (.@slot[0]>4000 && .@slot[0]<5000) {
		mes "卡片？這裡？！";
		mes "我之前說過，我不會在插了卡片的物品上簽名。";
		emotion ET_SCRATCH;
		close;
	}
	if (getiteminfo(.@id, ITEMINFO_SLOT)) {
		mes "抱歉，我不會在有插槽的物品上簽名。";
		emotion ET_SORRY;
		close;
	}
	getinventorylist;
	for(set .@i,0; .@i<@inventorylist_count; set .@i,.@i+1)
		if (@inventorylist_expire[.@i] != 0) {
			mes "抱歉，我不會在租賃物品上簽名！";
			emotion ET_SCRATCH;
			close;
		}
	set .@price, .@Cost[0]+(.@Cost[1]*.@ref);
	if (getarraysize(.@Item) || .@price) {
		mes "我會需要：";
		if (getarraysize(.@Item)) for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2) {
			mes " ~ "+.@Item[.@i+1]+" 個 "+getitemname(.@Item[.@i]);
			if (countitem(.@Item[.@i]) < .@Item[.@i+1]) set .@nr,1;
		}
		if (.@price) mes " ~ "+.@price+" Zeny";
	}
	mes "我要在你的 ^0055FF"+getitemname(.@id)+"^000000 上簽名嗎？";
	next;
	if(select("Ok!:Leave")==2) {
		mes "[佩爾契克]";
		mes "再會...";
		emotion ET_SLEEPY;
		close;
	}
	mes "[佩爾契克]";
	if (Zeny < .@price || .@nr) {
		mes "我可不是為了『謝謝』才工作的。";
		emotion ET_SORRY;
		close;
	}
	if (!countitem2(.@id,1,.@ref,0,.@slot[0],.@slot[1],.@slot[2],.@slot[3])) {
		mes "你的 "+getitemname(@id)+" 在哪裡...？";
		npctalk "Perchik : You're a snoozy cheater!";
		logmes "Hack: Tried to sign an item not having it: "+getitemname(@id);
		emotion ET_KEK;
		close;
	}
	if (.@price) set Zeny, Zeny-.@price;
	if (getarraysize(.@Item))  for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2)
		delitem .@Item[.@i], .@Item[.@i+1];
	delitem2 .@id,1,1,.@ref,0,.@slot[0],.@slot[1],.@slot[2],.@slot[3];
	mes "完成了！";
	emotion ET_OK;
	getitem2 .@id,1,1,.@ref,0,254,0,getcharid(0)&0xffff,(getcharid(0)>>16)&0xffff;
	equip .@id;
	close;
}
