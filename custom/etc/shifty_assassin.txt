morocc,148,86,5	script	可疑的刺客	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "[可疑的刺客]";
	mes "你想要什麼？";
	next;
	menu "雇用忍者",M_Buy,"暗殺某人",M_Kill,"查看你的忍者",M_Check, ( getgmlevel() > 90 ? "增加忍者數量" : "" ),-,"取消",M_Exit;
// GM MENU TO ADD NINJAS //
	mes "[可疑的刺客]";
	mes "你想增加多少名可雇用的忍者？";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes "已增加 " + .@add +" 名忍者。";
	close;

	// BUY NINJAS //
M_Buy:
	mes "[可疑的刺客]";
	mes "你想雇用多少名忍者？";
	mes "目前有 ^0000FF" + $ninja_avail + "^000000 名忍者可供雇用。";
	mes "每名忍者的費用是 ^0000FF" + .@ninja_price + " Zeny^000000。";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "[可疑的刺客]";
	mes "總共需要 ^0000FF"+ .@price +" Zeny^000000。";
	next;
	menu "繼續",-,"取消",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "[可疑的刺客]";
	mes "謝謝惠顧。";
	close;
// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "[可疑的刺客]";
	mes "請輸入目標的名字。";
	mes "^FF0000請輸入完全正確的名字，否則我會找不到目標。^000000";
	next;
	menu "繼續",-,"取消",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "[可疑的刺客]";
		mes .@name$ +" 目前不在線上。";
		close;
	}
	next;
	mes "[可疑的刺客]";
	mes "活動中的忍者：" + #ninjas;
	mes "休息中的忍者：" + #ninjasr;
	mes "你想派出多少名忍者？";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;
// SUCCESSFUL ATTACK //
	mes "現在派出忍者。";
	next;
	mes "[可疑的刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "你的暗殺行動成功了，但只有 ^FF0000" + #ninjasr + "^000000 名忍者活了下來。";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +" 已經被 " + strcharinfo(0) +" 的忍者暗殺了。",bc_npc;
	close;
// FAILED ATTACK //
M_Failure:
	mes "現在派出忍者。";
	next;
	mes "[可疑的刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "你的暗殺行動失敗了，只有 ^FF0000" + #ninjasr + "^000000 名忍者活了下來。";
	announce .@name$+" 在 " + strcharinfo(0) +" 的忍者暗殺中存活了下來。",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "[可疑的刺客]";
	mes "抱歉，我所有的忍者都忙著參加城戰。";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "[可疑的刺客]";
	mes "你目前擁有：";
	mes "^FF0000" + #ninjas + "^000000 名活動中的忍者。";
	mes "^0000FF" + #ninjasr + "^000000 名休息中的忍者。";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "[可疑的刺客]";
	mes "沒有忍者，你誰也殺不了。";
	next;
	goto M_Start;

TooManyNinjas:
	mes "[可疑的刺客]";
	mes "你最多只能派出 10 名忍者。";
	next;
	goto M_Start;

NoZeny:
	mes "[可疑的刺客]";
	mes "你沒有足夠的 Zeny。";
	close;

NotEnoughNinjas:
	mes "[可疑的刺客]";
	mes "沒有那麼多忍者可以雇用。";
	next;
	goto M_Start;

NoNinjas:
	mes "[可疑的刺客]";
	mes "目前沒有忍者可以雇用了。";
	close;

NotEnoughNinjas1:
	mes "[可疑的刺客]";
	mes "你沒有那麼多名忍者。";
	next;
	goto M_Start;

M_Exit:
	mes "[可疑的刺客]";
	mes "再見。";
	close;
// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
