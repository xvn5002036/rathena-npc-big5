prontera,148,174,5	script	私人 MVP 房間	100,{
	mes "[私人 MVP 房間]";
	mes "請選擇一個私人 MVP 房間。";
	if ( getvariableofnpc( .rentcost, "MVP 召喚員" ) )
		mes "租用房間 "+ getvariableofnpc( .timeout, "MVP 召喚員" ) +" 分鐘的費用是 "+ callfunc("F_InsertComma", getvariableofnpc( .rentcost, "MVP 召喚員" ) ) +" Zeny。";
	else
		mes "你只能使用這個房間 "+ getvariableofnpc( .timeout, "MVP 召喚員" ) +" 分鐘。";
	mes " ";
	for ( .@i = 1; .@i <= 8; .@i++ )
		if ( getvariableofnpc( .renttime[.@i], "MVP 召喚員" ) )
			mes "房間 #"+ .@i +" = "+ .color$[ .type[.@i] ] + .whoinuse$[.@i] +"^000000";
	next;
	.@room = select(
		"MVP 房間 1 ["+ getmapusers("06guild_01") +"]",
		"MVP 房間 2 ["+ getmapusers("06guild_02") +"]",
		"MVP 房間 3 ["+ getmapusers("06guild_03") +"]",
		"MVP 房間 4 ["+ getmapusers("06guild_04") +"]",
		"MVP 房間 5 ["+ getmapusers("06guild_05") +"]",
		"MVP 房間 6 ["+ getmapusers("06guild_06") +"]",
		"MVP 房間 7 ["+ getmapusers("06guild_07") +"]",
		"MVP 房間 8 ["+ getmapusers("06guild_08") +"]");
	if ( getvariableofnpc( .renttime[.@room], "MVP 召喚員" ) ) {
		if ( .inuseid[.@room] == getcharid( .type[.@room] ) ) {
			warp "06guild_0"+ .@room, 0,0;
			close;
		} else {
			mes "[私人 MVP 房間]";
			mes "這個房間已經被預訂給 ";
			mes .color$[ .type[.@room] ] + .whoinuse$[.@room] +"^000000。";
			mes "請選擇其他房間。";
			close;
		}
	}
	mes "[私人 MVP 房間]";
	mes "將此房間預訂給...";
	next;
	set .@type, select( "我的隊伍成員", "我的公會成員", "個人帳號使用" );
	if ( !getcharid(.@type) ) {
		mes "[私人 MVP 房間]";
		mes "你沒有所屬的"+( ( .@type == 1 )? "隊伍" : "公會" )+"。";
		close;
	}
	else if ( Zeny < getvariableofnpc( .rentcost, "MVP 召喚員" ) ) {
		mes "你沒有足夠的 Zeny 來租用房間。";
		close;
	}
	else if ( getvariableofnpc( .renttime[.@room], "MVP 召喚員" ) ) {
		mes "[私人 MVP 房間]";
		mes "抱歉，已經有其他人搶先登記了這個房間。";
		close;
	}
	for ( .@i = 1; .@i <= 8; .@i++ ) {
		if ( ( getvariableofnpc( .renttime[.@i], "MVP 召喚員" ) ) && .@type == .type[.@i] && getcharid(.@type) == .inuseid[.@i] ) {
			mes "[私人 MVP 房間]";
			mes "你已經租用了房間 #"+ .@i +"。請直接使用那個房間。";
			close;
		}
	}
	set .type[.@room], .@type;
	set .inuseid[.@room], getcharid(.@type);
	set .whoinuse$[.@room], strcharinfo( ( .@type == 3 )? 0 : .@type );
	Zeny -= getvariableofnpc( .rentcost, "MVP 召喚員" );
	warp "06guild_0"+ .@room, 0,0;
	killmonsterall "06guild_0"+ .@room;
	donpcevent "MVP 召喚員#"+ .@room +"::OnEnterMap";
	close;
OnInit:
	.color$[1] =  "^EE8800"; // party color
	.color$[2] =  "^70CC11"; // guild color
	.color$[3] =  "^0000FF"; // account color
	end;
}

		if ( mobcount( "this", strnpcinfo(3)+"::OnMobDead" ) ) {

			mes "當附近有魔物時，我無法召喚下一隻 MVP。";

			mes "[MVP 召喚員]";

		close2;

		Zeny -= .mvpcost;
		monster "this", 0, 0, "--ja--", .mvpid[.@menu], 1, strnpcinfo(3)+"::OnMobDead";
		end;
	case 3:
		if ( mobcount( "this", strnpcinfo(3)+"::OnMobDead" ) ) {
			mes "[MVP 召喚員]";
			mes "當附近有魔物時，我無法召喚下一隻小王。";
			close;
		} else if ( .bosscost ) {
			mes "[MVP 召喚員]";
			mes "召喚一隻小王的費用是 "+ callfunc( "F_InsertComma", .bosscost ) +" Zeny。";
			next;
			set .@menu, select(.bossid_menu$) -1;
			mes "[MVP 召喚員]";
			if ( Zeny < .bosscost ) {
				mes "你沒有足夠的 Zeny 來召喚小王。";
				close;
			}
		} else {
			set .@menu, select(.bossid_menu$) -1;
			mes "[MVP 召喚員]";
		}
		mes "請做好準備。";
		close2;
		if ( Zeny < .bosscost ) end;
		Zeny -= .bosscost;
		monster "this", 0, 0, "--ja--", .bossid[.@menu], 1, strnpcinfo(3)+"::OnMobDead";
		end;
	case 4:
		if ( mobcount( "this", strnpcinfo(3)+"::OnMobDead" ) > 0 ) {
			mes "[MVP 召喚員]";
			mes "當附近有魔物時，我無法提供這項服務。";
			close;
		}
		close2;
		callshop "MVP room#branch", 1;
		end;
	case 5:
		mes "[MVP 召喚員]";
		mes "你確定要離開這個房間嗎？";
		next;
		if ( select( "是:否") == 1 )
			warp .respawnmap$, .respawnx, .respawny;
		close;
	case 6:
		mes "[MVP 召喚員]";
		if ( getmapusers( strcharinfo(3) ) > 1 ) {
			mes "房間內還有其他玩家。請確認你是房間裡的最後一人，再來使用這個選項。";
			close;
		}
		mes "你確定要放棄這個房間嗎？";
		if ( .rentcost )
			mes "如果還要進入，你需要重新支付費用。";
		next;
		if ( select( "是", "否" ) == 2 ) close;
		awake strnpcinfo(0);
		end;
	}
	close;

OnMobDead:
	end;

	.renttime[.@id] = gettimetick(2);
	sleep .timeout * 60000;
	mapwarp strnpcinfo(4), .respawnmap$, .respawnx, .respawny;
