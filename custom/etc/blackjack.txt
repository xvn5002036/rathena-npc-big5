//===== Description: =========================================
//= Black Jack card game.
//= Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.
//= Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	21點 發牌員	57,{
	mes "[發牌員]";
	mes "你好啊！想來玩一局 21 點（Black Jack）嗎？";
M_Menu:
	next;
	menu "遊戲規則是什麼？",M_0, "我想玩。",L_Play, "不了，謝謝。",M_End;

M_0:
	mes "[發牌員]";
	mes "你想了解什麼？";
sM_Menu0:
	next;
	menu "基本規則。",sM_0a, "輸贏判定。",sM_0b, "「雙倍下注（Double Down）」。",sM_0c,
	"Ace 的點數。",sM_0d, "沒事。",M_Menu;

sM_0a:
	mes "[發牌員]";
	mes "21 點是一款紙牌遊戲，目標是讓點數達到 21 點，或是";
	mes "盡可能接近 21 點，但不能超過 21 點。";
	next;
	mes "[發牌員]";
	mes "遊戲開始前，玩家必須先下注。一旦下注完成，";
	mes "發牌員和玩家都會各自發到 2 張牌。";
	mes "根據你手上的牌，你可以選擇 ^5533FF「停牌（stay）」^000000，";
	mes "或者你也可以選擇 ^5533FF「要牌（pull）」^000000。";
	next;
	mes "[發牌員]";
	mes "當你選擇「停牌」時，代表你告訴發牌員你已經";
	mes "不需要更多的牌了。這時就輪到發牌員決定是否要牌。";
	mes "若要停牌，你必須 ^FF3355至少有 13 點^000000。而發牌員";
	mes "則必須在 ^FF3355達到 17 點或以上^000000 時才能強制停牌。";
	next;
	mes "[發牌員]";
	mes "當你選擇「要牌」時，代表你告訴發牌員你想要";
	mes "再拿一張牌。藉由要牌，你可以增加你的";
	mes "總點數。只要你的點數 ^FF3355低於 21 點^000000，你就可以繼續";
	mes "從牌堆中要牌。";
	goto sM_Menu0;
sM_0b:
	mes "[發牌員]";
	mes "在 21 點中，有 3 種判定輸贏的方式。";
	next;
	mes "[發牌員]";
	mes "1.) 在一局 21 點結束時，如果你的點數比";
	mes "發牌員多，你就會贏得這局。反之，如果你";
	mes "的點數比發牌員少，你就會輸掉。";
	next;
	mes "[發牌員]";
	mes "2.) 如果你要了一張牌，導致你的總點數超過 21 點，你";
	mes "將自動輸掉這局。這稱為 ^5533FF「爆牌（bust）」^000000。如果";
	mes "發牌員爆牌了，你就會贏得這局。";
	next;
	mes "[發牌員]";
	mes "3.) 如果你最初的 2 張牌總點數剛好是 21 點，你";
	mes "將自動贏得這局。這被稱為 ^5533FF「Black Jack」^000000，";
	mes "當你拿到一張「Ace」和一張「10 點數值」的牌時就會發生。如果";
	mes "發牌員拿到了 Black Jack，他將自動贏得這局。";
	next;
	mes "[發牌員]";
	mes "4.) 除了輸和贏之外，你也有可能和發牌員平手。如果";
	mes "在一局結束時，你和發牌員的總點數相同，";
	mes "這局就會以平局收場，沒有人獲勝，也不會損失或贏得";
	mes "任何金錢。";
	next;
	mes "[發牌員]";
	mes "這在遊戲中被稱為與發牌員 ^5533FF「平手（push）」^000000。這也";
	mes "適用於你和發牌員同時拿到 Black Jack 的";
	mes "情況。";
	goto sM_Menu0;
sM_0c:
	mes "[發牌員]";
	mes "「雙倍下注」選項允許你將目前的賭注加倍，";
	mes "但缺點是你將只能夠再要一張";
	mes "額外的牌。這個選項只有在每一局的";
	mes "最開始時才能選擇。";
	next;
	mes "[發牌員]";
	mes "一個使用雙倍下注的好時機範例，是當";
	mes "你最初的 2 張牌總點數為 11 點時。你有很大的";
	mes "機率在下一張抽到的牌中湊成 21 或 20 點。這";
	mes "就會是一個適合使用雙倍下注的好牌型。";
	goto sM_Menu0;
sM_0d:
	mes "[發牌員]";
	mes "「Ace」在 21 點遊戲中是一張非常獨特的牌，因為";
	mes "它可以有 2 種點數。一張 Ace 可以被算作 11 點，";
	mes "或者只有 1 點。舉例來說，如果你有一張 Ace 和一張 4，這將會";
	mes "給你 15 點或 5 點的總和。";
	next;
	mes "[發牌員]";
	mes "如果你決定停牌，這張 Ace 將會自動被算作";
	mes "11 點，讓你的總點數為 15 點。";
	next;
	mes "[發牌員]";
	mes "如果你當時決定要牌，並且拿到了一張 9，這張 Ace 將會";
	mes "自動被算作 1 點，讓你的總點數變成 14";
	mes "點。如果這張 Ace 被算作 11 點的話，你的總點數";
	mes "就會超過 21 點並且輸掉這局。";
	next;
	mes "[發牌員]";
	mes "正是因為「Ace」擁有的這種靈活性，讓它成為";
	mes "這個遊戲中最強大的牌。";
	goto sM_Menu0;
M_End:
	mes "[發牌員]";
	mes "隨時歡迎你再來玩。";
	close;

//================
L_Play:
	mes "[發牌員]";
	mes "請下注...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "這賭注對我來說太大了....",M_End;
M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;
sL_NotEnuf:
	mes "[發牌員]";
	mes "很抱歉，你沒有足夠的 Zeny 來下這個注。";
	close;
L_Cont0:
	mes "(發牌中)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;
	mes "- 這是 ^FF5533發牌員^000000 的牌：";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    發牌員 點數為： ^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- 這是 ^5533FF你^000000 的牌：";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    你 點數為： ^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    你 點數為： ^5533FF" +@playTot+ "^000000，或 ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "要牌(pull)",M_Hit, "停牌(Stay)",M_Stay, "雙倍下注(Double Down)",M_Double;
	if(@dealerTurn == 0) menu "要牌(pull)",M_Hit, "停牌(Stay)",M_Stay;

M_Stay:
	mes "[發牌員]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "發牌員選擇停牌。";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "發牌員選擇要牌。";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;
sL_PlayToLow:
	mes "很抱歉，你的總點數不夠高，無法停牌。你必須要牌。";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[發牌員]";
	mes "玩家選擇了雙倍下注。你目前的賭注將會";
	mes "加倍，而且你只能夠再多要 1 張牌。";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;
sL_Win:
	mes "[發牌員]";
	mes "恭喜你，你贏了！";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[發牌員]";
	mes "很遺憾，你輸了。";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[發牌員]";
	mes "平局（Push）。你和發牌員平手了。";
	next;
	goto L_Play;
//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;
//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;
GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;
//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
