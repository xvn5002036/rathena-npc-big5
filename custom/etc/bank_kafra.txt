//===== Current Version: =====================================
//= 1.2a
//===== Compatible With: =====================================
//= rAthena Project
//===== Description: =========================================
//= A bank which has an interest %
//===== Additional Comments: =================================
// Look for this formula and setup your Bank daily % interest
// #kafrabank/1000 = 0.1% of interest per day
// #kafrabank/100  =  1%  of interest per day
// #kafrabank/10   =  10% of interest per day
//
// 1.1 Added log of bank operation -> logmes "some info";
// 1.2 Set max income to 100,000z. It would help to avoid
//     zeny exploits when you change DATE at your server
// 1.2a Corrected bad duplicate names. (bugreport:921) [Samuray22]
//============================================================

-	script	銀行職員::bankg	833,{
	mes "[馬尼斯]";
	mes strcharinfo(0)+", 歡迎來到普隆德拉第二銀行！";

	set @kb_int,(gettime(DT_MONTH)*31)+gettime(DT_DAYOFMONTH); //today's number
	set @income,0;
//calculate %
	if (#kafrabank<=0 || #kb_int>=@kb_int) goto L_NoIncomeToday;
	set @income,(#kafrabank/1000)*(@kb_int-#kb_int); //@income == % of the sum
	//max income constant:
	if (@income>100000) set @income,100000;
L_NoIncomeToday:
	set #kb_int,@kb_int; //reset days timer

	if(#kafrabank==0) mes "我們可以為您開立一個帳戶。";
	if(@income>0) mes "今日收益：^135445" + callfunc("F_InsertComma",@income) + "^000000 Zeny。";
	set #kafrabank,#kafrabank+@income;
	if(#kafrabank>0) mes "您的帳戶餘額：^135445" + callfunc("F_InsertComma",#kafrabank) + "^000000 Zeny。";
	mes "請問您需要什麼服務？";
	next;
	if(#kafrabank==0) menu "開立帳戶",-,"退出",B_EXIT2;
	if(#kafrabank>0) menu "存款",-,"提款",M_WITHDRAW,"退出",B_EXIT2;

	mes "[馬尼斯]";
	mes "請告訴我您想存入多少 Zeny。";
	next;
	if(input(@kafrabank) == 1) goto L_TOO_BIG_AMOUNT;
	if(@kafrabank<1000) goto L_LESS_1000;
	if(@kafrabank>zeny) goto L_NOT_ENOUGH;
	set Zeny, Zeny-@kafrabank;
	set #kafrabank,#kafrabank+@kafrabank;
	mes "[馬尼斯]";
	mes "您已成功存入 ^135445" + callfunc("F_InsertComma",@kafrabank) + " Zeny^000000。";
//we log these zeny operations into the log db
	logmes "Bank %: " + @income +"z, Deposit: "+ @kafrabank +"z, Final: "+ #kafrabank +"z";
	goto B_EXIT;

M_WITHDRAW:
	if(#kafrabank==0) goto L_ZERO_ACCOUNT;
	mes "[馬尼斯]";
	mes "您的帳戶餘額：^135445" + callfunc("F_InsertComma",#kafrabank) + "^000000 Zeny。";
	mes "請問您想提取多少 Zeny？";
	next;
	if(input(@kafrabank) == 1) goto L_TOO_BIG_AMOUNT;

	if(@kafrabank<1) goto B_EXIT2;
	if(@kafrabank>#kafrabank) goto L_NOT_ENOUGH;
	set #kafrabank,#kafrabank-@kafrabank;
	set Zeny, Zeny+@kafrabank;
	mes "[馬尼斯]";
	mes "這是您的 ^135445" + callfunc("F_InsertComma",@kafrabank) + " Zeny^000000，請在這裡簽名...";
//we log these zeny operations into the log db
	logmes "Bank %: " + @income +"z, Withdraw: "+ @kafrabank +"z, Final: "+ #kafrabank +"z";
	goto B_EXIT;

L_NOT_ENOUGH:
	mes "[馬尼斯]";
	mes "您沒有足夠的 Zeny 來進行此操作。";
	next;
	goto B_EXIT2;

L_ZERO_ACCOUNT:
	mes "[馬尼斯]";
	mes "您的帳戶裡沒有任何 Zeny！";
	next;
	goto B_EXIT2;

L_TOO_BIG_AMOUNT:
	mes "[馬尼斯]";
	mes "抱歉，您一次最多只能存入 10,000,000 Zeny。";
	next;
	goto B_EXIT2;

L_LESS_1000:
	mes "[馬尼斯]";
	mes "很抱歉，最低存款金額為 1,000 Zeny。";
	next;
	goto B_EXIT2;
B_EXIT:
	mes "非常感謝... 歡迎下次光臨！";
	next;

B_EXIT2:
	mes "[馬尼斯]";
	mes "感謝您使用我們的銀行服務。希望能很快再見到您。";
	close;
}
