function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "你的 "+getitemname($@wed_ring)+" 怎麼了？你該不會... 把弄它丟了吧？我們需要它來進行接下來的典禮！";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;
//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "你的背包似乎沒有空間可以裝戒指了... 請先清出一些空間，再回來領取你伴侶的戒指。";
		close;
	}
	mes "這是給你新娘的結婚戒指。";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "一旦你的新娘領取了戒指，婚禮就會開始。";
		close;
	}
}

function SF_StartCeremony {

	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;

		next;
		mes "["+@name$+"]";
		mes $@msg$+" 跑到哪裡去了？？如果少了一個人，我可沒辦法幫你們證婚...";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "孩子，你把你的 "+getitemname(7170)+" 弄到哪去了？";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "孩子，在典禮期間，你應該要一直穿著 "+getitemname(2338)+" 才對...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "孩子，你現在還不能脫下你的 "+getitemname(2206)+"....";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "很好，你想要以什麼身分登記？";
		next;
		set @submenu, select("新郎","新娘","取消");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "很好，你要以新郎的身分登記嗎？";
		next;
		if (select("是的","我改變心意了。")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "很好，你要以新娘的身分登記嗎？";
		next;
		if (select("是的","我改變心意了。")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你現在已登記為新郎。";
		mes "請盡快通知你的新娘來登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你現在已登記為新娘。";
		mes "請盡快通知你的新郎來登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "等你準備好再來吧。";
		close;
	}
}

function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "新郎";
	if (@bride)
		set @type$, "新娘";
		
	mes "["+@name$+"]";
	mes "在登記成為"+@type$+"之前，讓我檢查一下你是否符合所有條件...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "噢親愛的，你還太小了，不該考慮結婚的事！";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "如果你想結婚，你應該穿著 "+getitemname(@item)+"。";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "你的 "+getitemname(2206)+" 呢？那是禮服的必備配件。";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "戒指呢？親愛的，你需要準備一個 "+getitemname($@wed_ring)+" 來進行戒指交換喔。";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "很抱歉，你沒有足夠的錢支付登記費。";
		mes "等你湊齊 "+@cost+"z 後再來吧。"; 
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
