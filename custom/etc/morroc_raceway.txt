//Counts down and starts race
pvp_y_1-5,145,269,5	script	起跑員	733,{
	if ($@race != 0) goto L_Started;
	if ($@counting != 0) goto L_Started;
	if ($@racecount == 1) goto L_Started;
L_Menu:
	mes "[起跑員]";
	mes "請留在我的東側。";
	menu "開始比賽",L_Count,"取消",-;
	close;
L_Count:
	set $@counting,1;
	mes "倒數計時...";
	addtimer 1000, "起跑員::OnCount1000";
	addtimer 2000, "起跑員::OnCount2000";
	addtimer 3000, "起跑員::OnCount3000";
	addtimer 4000, "起跑員::OnCount4000";
	announce strcharinfo(0) + " 開始了倒數計時",1;
	announce "準備開始比賽！",1;
	close;

OnCount1000:
	announce "[3]",1;
	end;
OnCount2000:
	announce "[2]",1;
	end;
OnCount3000:
	announce "[1]",1;
	end;
OnCount4000:
	emotion ET_GO;
	specialeffect EF_CHIMTO;
	announce "[GO!]",1;
	set $@race,1;
	set $@position,0;
	set $@counting,0;
	set $@raceid,rand(100000,999999);
	end;
L_Started:
	if ((getmapusers("pvp_y_1-5") < 3) && ($@position > 0)) goto L_Menu;
	if ($@position > 2) goto L_Menu;
	mes "[起跑員]";
	mes "比賽進行中";
	close;

OnInit:
	set $@race,0;
	set $@position,0;
	set $@racecount,0;
	end;
}

//Checkpoint 1
pvp_y_1-5,144,262,5	script	檢查點 1	111,0,5,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_Started;
	if (@race == 6) goto L_Finished;
	if ($@race == 1) goto L_Started;
	mes "比賽尚未開始，請退後。";
	close;
L_Started:
	set @race,1;
	set @raceid,$@raceid;
	end;
L_Finished:
	mes "你已經完成比賽了。";
	close;
}

pvp_y_1-5,174,244,5	script	終點線	111,6,6,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_WrongRace;
	if (@race != 5) goto L_Miss;
	set @race,6;
	set $@position,$@position+1;
	announce "[" + strcharinfo(0) +"] 已抵達終點線！ [名次: " + $@position + "]",1;
	end;
L_Miss:

	close;

	mes "你並沒有參加這場比賽。";
	close;
}
