valkyrie,44,55,5	script	神射手#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神射手]";
			mes "恭喜你。";
			mes "榮耀歸於戰士們！";
			close;
		}
		mes "[神射手]";
		mes "一擊。";
		mes "必殺。";
		mes "這並不難，";
		mes "只要你培養出";
		mes "適合這種戰鬥風格";
		mes "的直覺。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[神射手]";
		mes "這個世界";
		mes "需要像你這樣強大的弓箭手。";
		mes "你準備好承擔";
		mes "這份重責大任了嗎？";
		next;
		mes "[神射手]";
		mes "你願意";
		mes "邁出下一步，";
		mes "成為一名神射手嗎？";
		next;
		if (select("不。:願意。") == 1) {
			mes "[神射手]";
			mes "當你準備好時，";
			mes "隨時歡迎你回來。";
			mes "榮耀歸於戰士們！";
			close;
		}
		if (SkillPoint) {
			mes "[神射手]";
			mes "你還有可以學習的技能。請使用";
			mes "你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[神射手]";
		mes "恭喜你。";
		mes "身為一名神射手，我希望你";
		mes "永遠不會錯失你的目標。";
		close;
	}
	mes "[神射手]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "英靈殿。";
	next;
	mes "[神射手]";
	mes "在這裡時";
	mes "請隨意，";
	mes "當作自己家。";
	mes "榮耀歸於戰士們！";
	close;
}
