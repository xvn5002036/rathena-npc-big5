valkyrie,44,47,5	script	超魔導師#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[超魔導師]";
			mes "恭喜你。";
			mes "榮耀歸於戰士們！";
			close;
		}
		mes "[超魔導師]";
		mes "我們超魔導師";
		mes "有責任將我們";
		mes "具破壞性的魔法";
		mes "用於正確的目的上。";
		next;
		mes "[超魔導師]";
		mes "在成為超魔導師之前，";
		mes "需要一生的訓練。";
		mes "你能想像如果我們的力量";
		mes "落入壞人手中，";
		mes "會發生什麼事嗎？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[超魔導師]";
		mes "時候到了。";
		mes "而米德加爾特";
		mes "需要那些能夠";
		mes "駕馭最強魔法的人...";
		next;
		mes "[超魔導師]";
		mes "你願意";
		mes "成為超魔導師嗎？";
		next;
		if (select("不。:願意。") == 1) {
			mes "[超魔導師]";
			mes "當你準備好時，";
			mes "隨時歡迎你回來。";
			mes "榮耀歸於戰士們！";
			close;
		}
		if (SkillPoint) {
			mes "[超魔導師]";
			mes "你還有可以學習的技能。請使用";
			mes "你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[超魔導師]";
		mes "恭喜你。";
		mes "身為一名超魔導師，";
		mes "我希望你能將";
		mes "你的魔法用於";
		mes "為正義與和平而戰。";
		close;
	}
	mes "[超魔導師]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "英靈殿。";
	next;
	mes "[超魔導師]";
	mes "在這裡時";
	mes "請隨意，";
	mes "當作自己家。";
	mes "榮耀歸於戰士們！";
	close;
}
