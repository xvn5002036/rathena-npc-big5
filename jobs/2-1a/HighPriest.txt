valkyrie,44,42,5	script	神官#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神官]";
			mes "恭喜你。";
			mes "榮耀歸於戰士們！";
			close;
		}
		mes "[神官]";
		mes "透過神聖的";
		mes "力量，願我們能";
		mes "找到平靜、力量";
		mes "與庇護。將我們";
		mes "從邪惡勢力中拯救出來...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[神官]";
		mes "我們的世界需要";
		mes "有天賦且";
		mes "信念堅定的人。";
		mes "請作為一名";
		mes "更偉大的神聖英雄";
		mes "繼續你的善行...";
		next;
		mes "[神官]";
		mes "你願意";
		mes "成為神官嗎？";
		next;
		if (select("不。:願意。") == 1) {
			mes "[神官]";
			mes "當你準備好時，";
			mes "隨時歡迎你回來。";
			mes "榮耀歸於戰士們！";
			close;
		}
		if (SkillPoint) {
			mes "[神官]";
			mes "你還有可以學習的技能。請使用";
			mes "你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[神官]";
		mes "恭喜你。";
		mes "身為一名神官，";
		mes "我希望你能繼續";
		mes "為正義";
		mes "與和平而戰。";
		close;
	}
	mes "[神官]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "英靈殿。";
	next;
	mes "[神官]";
	mes "在這裡時";
	mes "請隨意，";
	mes "當作自己家。";
	mes "榮耀歸於戰士們！";
	close;
}
