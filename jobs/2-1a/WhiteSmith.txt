	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神工匠]";
			mes "恭喜你。";
			mes "榮耀歸於戰士們！";
			close;
		}
		mes "[神工匠]";
		mes "啊...";
		mes "最頂尖的";
		mes "工藝。";
		mes "這就是";
		mes "神工匠的傑作。";
		next;
		mes "[神工匠]";
		mes "一旦你成為";
		mes "最頂尖的佼佼者，";
		mes "你就會以探索";
		mes "更新、更好的";
		mes "製作方式";
		mes "為目標來進行鍛造...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[神工匠]";
		mes "時候到了！";
		mes "我們的世界需要像你這樣";
		mes "勇敢且勤奮的";
		mes "冒險者...";
		next;
		mes "[神工匠]";
		mes "你願意";
		mes "成為神工匠嗎？";
		next;
		if (select("不。:願意。") == 1) {
			mes "[神工匠]";
			mes "當你準備好時，";
			mes "隨時歡迎你回來。";
			mes "榮耀歸於戰士們！";
			close;
		}
		if (SkillPoint) {
			mes "[神工匠]";
			mes "你還有可以學習的技能。請使用";
			mes "你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[神工匠]";
		mes "恭喜你！";
		mes "身為一名神工匠，";
		mes "我希望你能鍛造出";
		mes "最強大、最";
		mes "有韌性的武器來";
		mes "守護世界和平！";
		close;
	}
	mes "[神工匠]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "英靈殿。";
	next;
	mes "[神工匠]";
	mes "在這裡時";
	mes "請隨意，";
	mes "當作自己家。";
	mes "榮耀歸於戰士們！";
	close;
}
