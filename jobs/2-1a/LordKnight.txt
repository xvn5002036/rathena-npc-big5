valkyrie,44,39,5	script	騎士領主#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[騎士領主]";
			mes "恭喜你。";
			mes "榮耀歸於戰士們！";
			close;
		}
		mes "[騎士領主]";
		mes "我們騎士肩負著";
		mes "極大的重任...";
		mes "服務與保護。";
		next;
		mes "[騎士領主]";
		mes "即使要付出";
		mes "我們自己的生命，";
		mes "我們也必須捍衛";
		mes "同志們的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[騎士領主]";
		mes "你的時代來臨了！";
		mes "這世界依然需要你。";
		mes "請以全新的姿態";
		mes "繼續你英雄的生涯。";
		next;
		mes "[騎士領主]";
		mes "你願意";
		mes "成為騎士領主嗎？";
		next;
		if (select("不。:願意。") == 1) {
			mes "[騎士領主]";
			mes "當你準備好時，";
			mes "隨時歡迎你回來。";
			mes "榮耀歸於戰士們！";
			close;
		}
		if (SkillPoint) {
			mes "[騎士領主]";
			mes "你還有可以學習的技能。請使用";
			mes "你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[騎士領主]";
		mes "恭喜你！";
		mes "身為一名騎士領主，";
		mes "我希望你能";
		mes "身先士卒";
		mes "帶領每一場戰鬥！";
		close;
	}
	mes "[騎士領主]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "英靈殿。";
	next;
	mes "[騎士領主]";
	mes "在這裡時";
	mes "請隨意，";
	mes "當作自己家。";
	mes "榮耀歸於戰士們！";
	close;
}
