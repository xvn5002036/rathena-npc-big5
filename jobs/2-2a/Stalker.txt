valkyrie,53,58,3	script	神行太保#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神行太保]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[神行太保]";
		mes "呵...";
		mes "這很艱難";
		mes "作為一個英雄";
		mes "卻又陰險、";
		mes "不可靠、";
		mes "狡猾...";
		next;
		mes "[神行太保]";
		mes "但當情況";
		mes "變得艱難時";
		mes "我的夥伴知道他們";
		mes "可以指望我。";
		mes "我需要他們，而";
		mes "他們也需要我。";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[神行太保]";
		mes "這個世界需要";
		mes "更多願意";
		mes "遊走在秩序與不法之間";
		mes "的英雄。";
		next;
		mes "[神行太保]";
		mes "你準備好";
		mes "加入這群";
		mes "最狡猾的戰士";
		mes "行列了嗎？你準備好";
		mes "成為一名神行太保了嗎？";
		next;
		if (select("否。:是。") == 1) {
			mes "[神行太保]";
			mes "當你準備好的時候，";
			mes "隨時歡迎你回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[神行太保]";
			mes "你仍然可以學習更多的技能。請將";
			mes "所有剩餘的技能點數分配完後再來找我。";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[神行太保]";
		mes "恭喜！";
		mes "身為一名神行太保，我希望";
		mes "你能在背後";
		mes "捅對人。用惡人";
		mes "自己卑鄙的手段";
		mes "來驅逐他們吧！";
		close;
	}
	mes "[神行太保]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "榮耀的殿堂。";
	next;
	mes "[神行太保]";
	mes "請把這裡";
	mes "當作自己家一樣放輕鬆，";
	mes "當你在這裡的時候。";
	mes "向戰士們致敬！";
	close;
}
