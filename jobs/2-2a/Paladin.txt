valkyrie,53,39,3	script	聖殿十字軍#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[聖殿十字軍]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[聖殿十字軍]";
		mes "不要忘記";
		mes "聖戰";
		mes "即將到來！";
		mes "我們必須做好準備！";
		mes "願正義之光";
		mes "永遠照亮我們的道路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[聖殿十字軍]";
		mes "聖戰";
		mes "即將在我們";
		mes "不知不覺中降臨。現在比以往任何時候，";
		mes "都更需要堅強的男男女女";
		mes "來為善良與正義";
		mes "而戰。";
		next;
		mes "[聖殿十字軍]";
		mes "你願意作為一名聖殿十字軍";
		mes "站在正義的一方";
		mes "戰鬥嗎？";
		next;
		if (select("否。:是。") == 1) {
			mes "[聖殿十字軍]";
			mes "當你準備好的時候，";
			mes "隨時歡迎你回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[聖殿十字軍]";
			mes "你仍然可以學習更多的技能。請將";
			mes "所有剩餘的技能點數分配完後再來找我。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[聖殿十字軍]";
		mes "恭喜你。";
		mes "身為一名聖殿十字軍，我希望";
		mes "你能保護那些";
		mes "弱小的人，並在";
		mes "看到邪惡時將其驅逐。";
		close;
	}
	mes "[聖殿十字軍]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "榮耀的殿堂。";
	next;
	mes "[聖殿十字軍]";
	mes "請把這裡";
	mes "當作自己家一樣放輕鬆，";
	mes "當你在這裡的時候。";
	mes "向戰士們致敬！";
	close;
}
