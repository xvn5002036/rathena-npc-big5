valkyrie,53,56,3	script	冷艷舞者#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[冷艷舞者]";
			mes "恭喜妳。";
			mes "榮耀歸於戰士！";
			close;
		}
		mes "[冷艷舞者]";
		mes "向左移，";
		mes "向右移?！";
		mes "然後踏步...！";
		mes "跳舞可以";
		mes "不僅僅是個嗜好。";
		mes "對我來說，這是一種生活方式?";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[冷艷舞者]";
		mes "米德加爾特大陸";
		mes "正需要有才華的女性，";
		mes "來巧妙地改變";
		mes "善與惡之戰間的平衡。";
		next;
		mes "[冷艷舞者]";
		mes "妳準備好";
		mes "接下這個角色，";
		mes "成為一名冷艷舞者了嗎？";
		next;
		if (select("還沒。:是的。") == 1) {
			mes "[冷艷舞者]";
			mes "當妳準備好的時候，";
			mes "隨時歡迎妳回來。";
			mes "榮耀歸於戰士！";
			close;
		}
		if (SkillPoint) {
			mes "[冷艷舞者]";
			mes "妳目前還可以學習更多技能。請在";
			mes "消耗完所有剩餘的技能點數後再來找我。";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[冷艷舞者]";
		mes "恭喜妳！";
		mes "身為一名冷艷舞者，我知道";
		mes "妳的表演將會";
		mes "牽動所有觀眾的心...";
		close;
	}
	mes "[冷艷舞者]";
	mes "歡迎來到";
	mes "瓦爾哈拉，";
	mes "榮譽的殿堂。";
	next;
	mes "[冷艷舞者]";
	mes "在妳停留期間，";
	mes "請把這裡當作自己的家。";
	mes "榮耀歸於戰士！";
	close;
}
