valkyrie,53,54,3	script	搞笑藝人#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[搞笑藝人]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[搞笑藝人]";
		mes "你想和我一起";
		mes "唱首歌嗎？";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[搞笑藝人]";
		mes "沉悶的";
		mes "凡人世界需要";
		mes "更多歡快的歌曲。";
		mes "你願意將歡樂帶給他們，";
		mes "並在對抗邪惡的";
		mes "戰鬥中扭轉局勢嗎？";
		next;
		mes "[搞笑藝人]";
		mes "你願意為了";
		mes "米德加爾特這麼做...";
		mes "以搞笑藝人的身分嗎？";
		next;
		if (select("否。:是。") == 1) {
			mes "[搞笑藝人]";
			mes "當你準備好的時候，";
			mes "隨時歡迎你回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[搞笑藝人]";
			mes "你仍然可以學習更多的技能。請將";
			mes "所有剩餘的技能點數分配完後再來找我。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[搞笑藝人]";
		mes "恭喜！";
		mes "身為一名搞笑藝人，";
		mes "你的歌曲將為你的盟友";
		mes "帶來希望，並";
		mes "讓你的敵人感到絕望。";
		close;
	}
	mes "[搞笑藝人]";
	mes "歡迎";
	mes "來到瓦爾哈拉，";
	mes "榮耀的殿堂。";
	next;
	mes "[搞笑藝人]";
	mes "請把這裡";
	mes "當作自己家一樣放輕鬆，";
	mes "當你在這裡的時候。";
	mes "向戰士們致敬！";
	close;
}
