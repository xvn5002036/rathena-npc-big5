valkyrie,53,50,3	script	創造者#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[創造者]";
		mes "啊，你是來";
		mes "取回你失去的";
		mes "記憶的嗎？是的，你";
		mes "一定是為了曾經屬於你的";
		mes "生命奧秘而來...";
		next;
		if (select("是:否") == 1) {
			mes "[創造者]";
			mes "閉上你的眼睛，";
			mes "讓你的心靈平靜下來。";
			mes "我們將回到你的";
			mes "過去，收集你";
			mes "失去記憶的碎片。";
			next;
			mes "[創造者]";
			mes "當你睜開眼睛時，";
			mes "你會清楚地記得";
			mes "生命的奧秘。你也將";
			mes "記得為了正確的目的";
			mes "使用這些奧秘的責任感...";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[創造者]";
			mes "睜開你的眼睛...";
			mes "既然你已經";
			mes "想起了如何";
			mes "創造人工生命，我只";
			mes "要求你尊重你所有的";
			mes "創造物。";
			close;
		}
		mes "[創造者]";
		mes "如果你希望";
		mes "取回你失去的";
		mes "記憶，請";
		mes "再回來找我。";
		mes "創造生命的奧秘";
		mes "可不是件小事...";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[創造者]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[創造者]";
		mes "像我這樣的人";
		mes "會在這裡真是奇怪。";
		mes "但即便是精通";
		mes "科學之道的人，";
		mes "也能成為一名英雄。";
		next;
		mes "[創造者]";
		mes "在這個例子中，";
		mes "未必是我使用的方法，而是";
		mes "我所為之奮鬥的目的，";
		mes "讓我在瓦爾哈拉";
		mes "贏得了一席之地...";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[創造者]";
		mes "是的...";
		mes "是時候了。";
		mes "米德加爾特需要";
		mes "更多像你這樣的天才。";
		next;
		mes "[創造者]";
		mes "你想要";
		mes "成為一名創造者嗎？";
		next;
		if (select("否。:是。") == 1) {
			mes "[創造者]";
			mes "當你準備好的時候，";
			mes "隨時歡迎你回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[創造者]";
			mes "你仍然可以學習更多的技能。請將";
			mes "所有剩餘的技能點數分配完後再來找我。";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[創造者]";
		mes "恭喜！";
		mes "身為一名創造者，";
		mes "我希望你將你";
		mes "淵博的知識用於";
		mes "正確的目的。";
		close;
	}
	else {
		mes "[創造者]";
		mes "歡迎";
		mes "來到瓦爾哈拉，";
		mes "榮耀的殿堂。";
		next;
		mes "[創造者]";
		mes "請把這裡";
		mes "當作自己家一樣放輕鬆，";
		mes "當你在這裡的時候。";
		mes "向戰士們致敬！";
		close;
	}
}
