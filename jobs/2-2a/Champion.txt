valkyrie,53,42,3	script	武術宗師#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[武術宗師]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[武術宗師]";
		mes "技術。";
		mes "速度。";
		mes "力量。";
		mes "敏捷。";
		next;
		mes "[武術宗師]";
		mes "武術宗師能夠";
		mes "從這所有之中";
		mes "獲益。但是";
		mes "一個人一生中";
		mes "能掌握的終究有限...";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[武術宗師]";
		mes "時候到了。";
		mes "是偉大的英雄們";
		mes "挺身對抗困擾";
		mes "米德加爾特世界的";
		mes "邪惡勢力的時候了！";
		next;
		mes "[武術宗師]";
		mes "你想要";
		mes "成為一名武術宗師嗎？";
		next;
		if (select("否。:是。") == 1) {
			mes "[武術宗師]";
			mes "當你準備好的時候，";
			mes "隨時歡迎你回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[武術宗師]";
			mes "你仍然可以學習更多的技能。請將";
			mes "所有剩餘的技能點數分配完後再來找我。";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[武術宗師]";
		mes "恭喜！";
		mes "以武術宗師的身分活下去吧，";
		mes "並透過你雙拳的力量，";
		mes "為這個世界";
		mes "帶來光明。";
		close;
	}
	mes "[武術宗師]";
	mes "歡迎來到";
	mes "瓦爾哈拉，";
	mes "榮耀的殿堂。";
	next;
	mes "[武術宗師]";
	mes "當你在這裡的時候，";
	mes "請把這裡當作自己家";
	mes "一樣放輕鬆。";
	mes "向戰士們致敬！";
	close;
}
