			mes "感謝你的兌換。";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "你沒有足夠的 "+getitemname(.@cost)+"。";
		break;
	}
	close;

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[阿克塞爾·羅斯]";
			mes "受祝福的吉洛姆！";
		} else if (.@team$ == "Croix") {
			mes "[史旺德利]";
			mes "受祝福的克羅瓦！";
		}
		mes "讓我們享受這光榮的勝利吧！";
		mes strcharinfo(0)+"，這是反映著勝利的象徵。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[阿克塞爾·羅斯]";
			mes "你們輸了，但你們為這場戰鬥付出了心力。";
			mes "這是吉洛姆·馬羅洛為你們偉大的奉獻所給予的獎勵！";
			mes "把這次失敗當作一個教訓，下次你們一定會贏的。";
		} else if (.@team$ == "Croix") {
			mes "[史旺德利]";
			mes "喔，"+strcharinfo(0)+" 別難過。";
			mes "雖然我們沒有贏，但我們盡力了。";
			mes "這是來自克羅瓦的皇家禮物，請不要忘記這場戰鬥。我們下一場會贏的。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
