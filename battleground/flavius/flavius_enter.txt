bat_room,133,178,5	script	弗拉維烏斯 軍官#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 目前你身上攜帶了 -";
		mes "- 太多的物品。 -";
		mes "- 請在減輕一些重量後 -";
		mes "- 再試一次。 -";
		close;
	}
	mes "[吉洛姆軍官]";
	mes "讓那些臭克羅瓦見識一下吉洛姆軍隊的力量！";
	next;
	switch(select("我想加入你的軍隊！:結束對話")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[吉洛姆軍官]";
			mes "我很高興你想加入我們對抗克羅瓦軍隊，但很抱歉：我不能把你這種菜鳥送到殘酷的戰場上送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[吉洛姆軍官]";
				mes "我很高興你想加入我們對抗克羅瓦軍隊，但很抱歉：我不能把你這種菜鳥送到殘酷的戰場上送死。";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[吉洛姆軍官]";
					mes "你似乎才剛從戰場上回來。";
					mes "現在回去還太早了。去休息吧，把敵人交給我們！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[吉洛姆軍官]";
						mes "我接到報告，一支菁英部隊已經被派往戰場對抗克羅瓦軍隊了。";
						mes "你何不先去等一會兒？我建議你趁這段時間把武器磨利並準備好補給品。";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[吉洛姆軍官]";
							mes "一支菁英部隊已經在待命，準備前往戰場。";
							mes "你何不先去等一會兒？我建議你趁這段時間把武器磨利並準備好補給品。";
						}
						else {
							mes "[吉洛姆軍官]";
							mes "你看起來絕對準備好戰鬥了！";
							mes "去讓克羅瓦見識一下什麼叫真正的恐懼吧！";
							mes "今天，我們勝利的歡呼聲將響徹整個戰場！";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[吉洛姆軍官]";
		mes "今天，我們將會取得勝利！";
		break;
	}
	close;
}
