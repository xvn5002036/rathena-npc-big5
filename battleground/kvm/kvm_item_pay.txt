	mes "[後勤官]";

	if (!.@item_type)
		mes "你選擇了 ^ff0000"+ getitemname(.@item_id) +"^000000。";
	else if (.@item_type == 1)
		mes "只有一種 "+ callfunc("F_GetWeaponType",.@item_id) +"："+ getitemname(.@item_id) +"。";
	else if (.@item_type == 2) {
		mes "你選擇了「"+ callfunc("F_GetArmorType",.@item_id) +"」。";
		mes "供 80 級以上玩家裝備的防具為：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}
	else if (.@item_type == 3) {
		mes "你選擇了「量產型 "+ callfunc("F_GetArmorType",.@item_id) +"」。";
		mes "只有一種量產型 "+ callfunc("F_GetArmorType",.@item_id) +"：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}
	else if (.@item_type == 4) {
		mes "你選擇了「普及型 "+ callfunc("F_GetArmorType",.@item_id) +"」。";
		mes "只有一種普及型 "+ callfunc("F_GetArmorType",.@item_id) +"：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}

	if (!.@req_setting)
		mes "這需要 ^0000ff2,000^000000 點KVM積分，而 "+ .@name$ +"，你目前的積分為：^580080"+ kvm_point +"^000000 點。";
	else if (.@req_setting == 4)
		mes "這需要 ^0000ff1,200^000000 點KVM積分，而 "+ .@name$ +"，你目前的積分為：^580080"+ kvm_point +"^000000 點。";
	else
		mes "這需要 ^0000ff"+ .@prices[.@req_setting] +"^000000 點KVM積分，而 "+ .@name$ +"，你目前的積分為：^580080"+ kvm_point +"^000000 點。";
	mes "你確定要兌換這個物品嗎？";
	next;
	switch(select("不，我不想購買。:是的，我要購買。")) {
	case 1:
		mes "[後勤官]";
		mes "你選擇了「我不想購買」。";
		mes "兌換物品時請謹慎，因為無法退還。";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[後勤官]";
			mes "你成功兌換了 "+ getitemname(.@item_id) +"。";
			if (!.@req_setting)
				mes "你的KVM積分扣除了 ^0000ff"+ .@req_setting +"^000000 點，你現在剩餘 ^580080"+kvm_point+"^000000 點 KVM積分。";
			if (.@req_setting == 4)
				mes "你的KVM積分扣除了 ^0000ff1,200^000000 點，你現在剩餘 ^580080"+kvm_point+"^000000 點 KVM積分。";

		else {
			mes "[後勤官]";
			if (!.@req_setting)
				mes "你需要 ^0000ff2,000^000000 點KVM積分才能購買此物品。";
			else if (.@req_setting == 4)
				mes "你需要 ^0000ff1,200^000000 點KVM積分才能購買此物品。";
			else

			mes "然而，你目前的積分只有 ^580080"+ kvm_point +"^000000 點。";
