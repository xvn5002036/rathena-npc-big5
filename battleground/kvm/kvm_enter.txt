//============================================================
bat_room,164,178,5	script	KVM 傭兵軍官#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[克羅伊傭兵長]";
		mes "我知道你已經和克羅伊簽約了。回去加入他們的隊伍吧。我們不接受叛徒為我們戰鬥。";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[吉歐傭兵長]";
		mes "看來你最近剛參加過KVM戰場。你必須等待5分鐘才能再次報名。";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[吉歐傭兵長]";
	mes "讓他們見識吉歐真正的實力！";
	next;
	switch(select("我會與你並肩作戰。:結束對話。")) {
	case 1:
		mes "[吉歐傭兵長]";
		mes "讓他們看看我們有多強大。";
		mes "今天，所有人都能聽見吉歐勝利的歡呼聲！";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[吉歐傭兵長]";
		mes "我們一定會贏！";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	吉歐騎士#kvm	417,{
	mes "[吉歐騎士]";
	mes "你好。";
	mes "你想知道些什麼？";
	next;
	switch(select("報名KVM。:什麼是KVM？:我要如何參加KVM？:我想查詢我的KVM積分。")) {
	case 1:
		mes "[吉歐騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "如果你已經準備好戰鬥，請點擊介面上的 ^0000FF戰場按鈕^000000 來報名KVM戰場！";
		else {
			mes "目前尚未開放報名。";
			mes "若要報名，你需要去找KVM傭兵長。";
		}
		close;
	case 2:
		mes "[吉歐騎士]";
		mes "KVM是Kreiger Von Midgard（米德加爾特戰役）的縮寫。";
		mes "冒險者，你知道通往新世界的道路已經開啟了嗎？";
		next;
		mes "[吉歐騎士]";
		mes "我們吉歐政府，為了進軍新世界制定了幾項計畫。";
		mes "而KVM就是其中之一。";
		next;
		mes "[吉歐騎士]";
		mes "這意味著身為米德加爾特大陸眾多國家之一的我們，";
		mes "決定招募大量冒險者，為即將到來的新世界遠征做準備。";
		next;
		mes "[吉歐騎士]";
		mes "因此，為了選拔出最優秀的冒險者，我們舉辦了KVM戰役。";
		mes "我們非常期待能有許多冒險者響應。";
		next;
		mes "[吉歐騎士]";
		mes "我們也會給予參戰者豐厚的獎勵！";
		mes "至於獎勵的詳情，請洽詢KVM後勤官。";
		close;
	case 3:
		mes "[吉歐騎士]";
		if (getbattleflag("feature.bgqueue")) {
			mes "基本上，KVM是一場3對3的戰鬥。";
			mes "請點擊介面上的 ^0000FF戰場按鈕^000000，以團體或個人身分進行報名。";
		} else {
			mes "基本上，KVM是一場5對5的戰鬥。";
			mes "首先，你需要向KVM報名官以團體或個人身分報名。";
		}
		next;
		mes "[吉歐騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "團體報名適用於當你打算與隊伍或公會成員一起進入KVM時，";
		else
			mes "團體報名適用於當你打算與隊伍成員一起進入KVM時，";
		mes "而個人報名則適用於你打算單獨進入KVM戰場時。";
		next;
		mes "[吉歐騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "請注意，一旦你取消報名，^0000FF在1分鐘內將無法再次報名^000000。";
		else {
			mes "請向KVM傭兵長報名，稍後當你在普隆德拉時，他會與你聯絡。";
			next;
			mes "[吉歐騎士]";
			mes "然後你就可以進入戰場，並遵從戰場內的指示。";
			next;
			mes "[吉歐騎士]";
			mes "但是請注意，除非你在KVM辦公室內，否則他無法聯絡到你。";
		}
		close;
	case 4:
		mes "[吉歐騎士]";
		mes "你的KVM積分為：";
		mes ""+kvm_point+" 點。";
		close;
	}
}
