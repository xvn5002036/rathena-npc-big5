function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[世亞布連]";
	mes "你想為 "+ getitemname(.@item_id) +" 打洞嗎？";
	mes "好的，請帶給我 ^FF0000"+ .@req_amount[0] +" 個 "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "":"") +"^000000，"+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" 個 "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "":"") + "^000000 ") : "") + "以及 ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 Zeny^000000 的服務費。";
	mes "啊，別忘了帶那件 "+ getitemname(.@item_id) +"！";
	next;
	mes "[世亞布連]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "如果你有需要的物品和 Zeny，我現在就可以嘗試打洞。";
		mes "然而，你必須知道我有可能會失敗。";
		mes "因此，我必須事先警告你...";
		next;
		mes "[世亞布連]";
		mes "如果這次為武器打洞的嘗試失敗了，那麼這把^FF0000武器^000000，^FF0000以及插在上面的任何卡片都會被摧毀^000000。";
	}
	else // armor
	{
		mes "如果你備齊了所有需要的材料、我的 Zeny 服務費以及防具，那我們就可以開始嘗試打洞了。";
		mes "但在那之前，我必須警告你其中的風險。";
		next;
		mes "[世亞布連]";
		mes "如果這次為防具打洞的嘗試失敗了，那麼這件^FF0000防具^000000、^FF0000它的精煉值^000000 ^FF0000以及插在上面的任何卡片都會被摧毀^000000。";
	}
	mes "你還想要嘗試打洞嗎？";
	next;
	switch(select("嘗試打洞:取消"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[世亞布連]";
				mes "好的，那我們開始吧！";
				mes "你最好祈禱能有成功的結果。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY;
					// only S Class items have Lord of Vermillion effect
					mes "[世亞布連]";
					mes "太棒了，看起來成功了。";
					mes "看起來做得非常好。恭喜你！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[世亞布連]";
					mes "下次見，朋友！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[世亞布連]";
					mes "哇！...非常抱歉，失敗了。";
					mes "不過，我是完全無辜的。";
					mes "這是你的運氣，是上天註定的，好嗎？";
					mes "別太失望了，下次再試試吧。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[世亞布連]";
					mes "祝你下次好運！";
					close;
				}
			}
			else
			{
				mes "[世亞布連]";
				mes "我很想繼續為你嘗試打洞，但你少了一些東西。";
				mes "你確定你帶了裝備、需要的材料和 Zeny 嗎？";
				close;
			}
		case 2:
			mes "[世亞布連]";
			mes "需要一些時間考慮，是嗎？";
			mes "好吧，我可以理解。";
			mes "只要記住，如果總是打安全牌，人生就太無趣了?";
			close;
	}
}
