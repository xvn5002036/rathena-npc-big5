//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "歡迎來到";
	mes getarg(1) + "。";
	mes "有什麼我可以為您服務的嗎？";
	next;
	switch(select("儲存地點:休息一下 -> 5000 Zeny:取消")) {
	case 1:
		mes .@npc_name$;
		mes "您的重生點";
		mes "已經儲存完畢。";
		mes "謝謝您，";
		mes "歡迎下次光臨。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "很抱歉，服務費是 5,000 Zeny。下次請確保您有足夠的錢辦理入住，好嗎？";
close;
		}
		mes "謝謝您。";
		mes "希望您";
		mes "休息得愉快?";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
case 3:
		close;
	}
}
