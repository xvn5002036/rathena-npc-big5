-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "看起來你身上帶了太多東西。";
			mes "為什麼不把一些物品放進倉庫後再來呢？";
			close;
		}
		mes "你好。";
		mes "你在找什麼呢？";
		next;
		if (strnpcinfo(1) == "瑪努克商人") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("查看物品說明:瑪努克的機會:瑪努克的勇氣:瑪努克的信念:取消");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("查看物品說明:購買捕蟲堇妖果醬:購買露西歐拉蜂蜜果醬:購買獨角飛馬之淚:什麼都不做");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "這些是物品說明。";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "我可以用 ^3131FF1 枚硬幣^000000 賣給你 3 個 "+getitemname(.@Items[.@i])+"。";
			next;
			switch(select("購買:不購買")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "謝謝光臨。";
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "你的硬幣不夠。";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "如果你改變心意了再來吧。";
		close;
	}
	else {
		if (strnpcinfo(1) == "瑪努克商人") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
