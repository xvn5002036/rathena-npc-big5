function Dyes {
	setarray .@DyeName$[0],"原色","橘色","紫色","紅色","黑色","綠色","藍色","白色","黃色";
	setarray .@DyeItems[0],0,980,981,975,983,979,978,982,976;
	set .@menu$,"- 原色:";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+2)
		set .@menu$, .@menu$+"- "+.@DyeName$[getarg(.@i)]+":";
	set .@menu$, .@menu$+"- ^777777取消^000000";
	set .@s, select(.@menu$)-1;
	if (.@s == (getargcount()/2)+1) goto L_End;
	mes .n$;
	if (.@s == 0) {
		mes "我可以免費將你的衣服";
		mes "染回原色。";
		mes "你確定嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			mes .n$;
			setlook 7,0;
			mes "你的衣服已經被染成了 "+.@DyeName$[0]+"。";
			close;
		case 2:
			goto L_End;
		}
	}
	if (!countitem(.@DyeItems[getarg((.@s-1)*2)])) {
		mes "為了幫你的衣服染色，我需要對應的染料。請查看我的價目表以獲取資訊。";
		close;
	}
	if (Zeny < 10000) {
		mes "很抱歉，但你沒有足夠的錢。";
		close;
	}
	delitem .@DyeItems[getarg((.@s-1)*2)],1;
	set Zeny, Zeny-10000;
	setlook 7, ((.@s)?(getarg((.@s-1)*2+1)):0);
	mes "你的衣服已經被染成了 "+((.@s)?.@DyeName$[getarg((.@s-1)*2)]:.@DyeName$[0])+"。";
	close;
}
