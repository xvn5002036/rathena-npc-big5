//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我是武器精煉師。";
	mes "我可以精煉各種武器、防具和裝備，所以請告訴我";
	mes "你想精煉什麼。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我看我沒辦法精煉你身上的任何物品...";
		close;
	}
	.@part = .@indices[select(.@menu$)];
	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你在那裡並沒有穿戴";
		mes "任何我可以";
		mes "精煉的東西。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我完全沒辦法";
		mes "精煉這個物品...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我沒辦法再精煉這個";
		mes "物品了。這已經是";
		mes "最完美的精煉狀態了！";
		close;
	}
	.@refineitemid = getequipid(.@part);
// save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;
	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "精煉這個我需要";
		mes "一顆 ^003366"+getitemname(.@material)+"^000000 以及";
		mes "" + .@price + " Zeny 的服務費。";
		mes "你真的要繼續嗎？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "是啊...";
			mes "沒必要";
			mes "著急。慢慢來。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "糟了！如果我繼續";
			mes "精煉這個，它可能會";
			switch(.@material) {
			case 985:
				mes "損壞！這代表^FF0000這件裝備^000000，以及防具上^FF0000所有的卡片^000000或附加屬性，^FF0000都會消失^000000。";
				break;
			default:
				mes "損壞，而你將會^FF0000失去這把武器^000000、武器上^FF0000所有的卡片^000000，";
				mes "或是任何附加的特殊屬性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我沒辦法說得更清楚了。";
			mes "武器一旦損壞，";
			mes "就再也回不來了。";
			mes "你真的有極大的可能會";
			mes "永遠^FF0000失去這把武器^000000。";
			mes "你還想要精煉嗎？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意...";
				mes "我也許是個偉大的精煉師，但有時候就連我也會犯錯。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "你似乎沒有";
			mes "足夠的 Zeny 或 "+getitemname(.@material)+"...";
			mes "再去準備一些吧。如果你需要我，我一整天";
			mes "都會在這裡。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下...";
			mes "你以為我是笨蛋嗎？！";
			mes "你趁我不注意的時候把裝備掉包了！給我滾出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "噢！我的天啊！";
				mes "可惡！又失敗了！";
				mes "我非常抱歉，但你知道熟能生巧的道理。";
				mes "呃，對吧？嘿嘿...";
			} else if(.@lose == 2) {
				mes "不！！！！！";
				mes "它碎了！";
				mes "我-我很抱歉！";
			} else {
				mes "糟糕！";
				mes "它承受不住";
				mes "更多的鍛造了！";
				mes "很抱歉變成這樣...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "嘿嘿！";
			mes "再次展現了，";
			mes "大師";
			mes "完美無瑕的手藝?";
		} else if(.@win == 2) {
			mes "成功...！";
			mes "又一次，我驚人的";
			mes "才華在今天";
			mes "耀眼地綻放著。";
		} else {
			mes "嘿嘿！";
			mes "大功告成。";
			mes "毫無疑問，我的作品";
			mes "包你滿意。";
			mes "純粹、徹底的完美?";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以將它精煉到安全界限，或者你想要的次數。由你決定。";
		next;
		.@menu2 = select("請精煉到安全界限。","讓我決定要精煉幾次。","我改變主意了...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你想讓我精煉你的物品幾次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "我沒辦法精煉這件裝備那麼多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這將會嘗試精煉裝備 " + .@refinecheck + " 次，超過安全界限。你的裝備可能會損壞... 這樣好嗎？";
			next;
			if(select("好...","不好...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "你都這麼說了... 那就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了... 那就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "這將會花費你 " + .@refinecnt + " 顆 " + getitemname(.@material) + " 以及 " + .@fullprice + " Zeny。這樣可以嗎？";
	next;
	if(select("是","否...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了... 那就這樣吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "你只有這些嗎？很遺憾，我沒辦法算你便宜點。你試著站在我的立場想想吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "看這裡... 你的裝備沒穿上...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏘... 不對，你以為我有那麼笨嗎？！";
			mes "你把它掉包了...";
			mes "在我用鐵鎚敲暈你之前，給我滾出去！！";
			close;
		} 
		mes "鏘，鏘！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊啊啊！！！我很抱歉... 我警告過你可能會發生這種事的...";
			.@refinecnt = .@refinecnt - 1;

		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "全數完工... 歡迎下次再來。";
	close;
}

		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "交易";
			mes "已經取消。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "好了，你最多只能";
			mes "購買 500 個。";
			mes "不能再多了，";
			mes "懂嗎？很好。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "呃...";
		mes "你的 Zeny 不夠";
		mes "買 "+ .@input +" 個。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯...";
		mes "如果你的物品欄沒有足夠的空間，我不能把東西給你。你怎麼不把多餘的東西存進卡普拉倉庫然後再來試試呢？";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "給你！";
	mes "感謝";
	mes "你的光臨。";
	close;
}

//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "嘿！";
	mes "你需要我";
	mes "幫你修理任何物品嗎？";
	mes "你可以放心把裝備";
	mes "交給我修理！";
	next;
	switch(select("其實，我有一些裝備...:目前沒有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哦哇，這太不可思議了！";
			mes "你一定非常愛惜你的東西。你的物品沒有任何損壞！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "如果每個人都像你一樣，我就要失業啦！！哈哈?！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "嗯...";
		mes "讓我看看...";
		mes "在你所有的物品中，";
		mes "有 " + .@checkitem + " 件損壞了。";
		mes "你想要修理嗎？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每次修理需要 " + .@repairprice + " Zeny。所以修理你所有損壞的物品總共需要 " + .@totalcost + " Zeny！你想修理這些物品嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "哇喔哇喔...";
				mes "在拿到維修帳單之前先檢查一下你的錢包！因為你沒有足夠的 Zeny，我沒辦法幫你修理任何東西。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好的！全都搞定了。現在，試著再小心一點。物品也是有生命的你知道的。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？怪怪的。等一下... 把你需要修理的裝備穿上，然後再來找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "好吧，這對我來說沒什麼損失，但放任物品損壞可不是好事。你應該盡快把它們修好！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "呵呵呵...";
		mes "如果你沒有";
		mes "任何物品需要修理，";
		mes "那你就沒必要";
		mes "來找我。";
		close;
	}
}
