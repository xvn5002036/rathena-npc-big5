// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	大嘴鳥飼養員#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[大嘴鳥飼養員]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "歡迎。";
		mes "尊敬的騎士，";
		mes "您想要租借";
		mes "一隻大嘴鳥嗎？租借";
		mes "費用是 "+.@price+" Zeny。";
		next;
		switch(select("租借大嘴鳥:取消")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥飼養員]";
				mes "您沒有";
				mes "足夠的 Zeny。";
				mes "您是...";
				mes "破產了嗎？";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥飼養員]";
				mes "很抱歉，但是您";
				mes "不符合這項";
				mes "服務的資格。請先去學習";
				mes "騎乘術技能。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥飼養員]";
				mes "您已經";
				mes "騎乘著";
				mes "大嘴鳥了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥飼養員]";
				mes "請先解除您的商城坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥飼養員]";
			mes "我明白了。";
			mes "那麼，";
			mes "祝您有美好的一天。";
			close;
		}
	} else {
		mes "很抱歉，但是這些";
		mes "大嘴鳥只提供給";
		mes "騎士";
		mes "和騎士領主。";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)

prontera,232,318,3	script	大嘴鳥飼養員#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[大嘴鳥飼養員]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "歡迎，十字軍。";
		else mes "歡迎，聖殿十字軍。";
		mes "我們為您準備了";
		mes "特別的大嘴鳥。";
		mes "租借一隻";
		mes "需要花費 "+.@price+" Zeny。";
		next;
		switch(select("租借大嘴鳥:離開")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥飼養員]";
				mes "您沒有";
				mes "足夠的 Zeny。";
				mes "如果您想要";
				mes "一隻大嘴鳥，請";
				mes "帶 "+.@price+" Zeny 來...";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥飼養員]";
				mes "您必須先學習";
				mes "騎乘大嘴鳥，之後";
				mes "我才能租借給您。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥飼養員]";
				mes "您已經";
				mes "騎乘著大嘴鳥了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥飼養員]";
				mes "請先解除您的商城坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥飼養員]";
			mes "再見。";
			close;
		}
	} else {
		mes "有什麼我可以為您效勞的嗎？";
		mes "請注意，";
		mes "這項大嘴鳥租借";
		mes "服務僅限於";
		mes "十字軍和聖殿十字軍使用。";
		close;
	}
}

hu_in01,381,304,5	script	獵鷹飼養員#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[獵鷹飼養員]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "嗯...";
			mes "您無法租借獵鷹，";
			mes "除非您先解散您的狼！";
			close;
		}
		mes "您需要一隻獵鷹嗎？";
		mes "您可以租借一隻專屬於您的";
		mes "可靠猛禽，";
		mes "費用只要 "+.@price+" Zeny?";
		next;
		switch(select("租借獵鷹:取消")) {
		case 1:
			if (checkwug()) {
				mes "[獵鷹飼養員]";
				mes "嗯...";
				mes "您無法租借獵鷹，";
				mes "除非您先解散您的狼！";
				close;
			}
			if (Zeny < .@price) {
				mes "[獵鷹飼養員]";
				mes "這是怎麼回事？";
				mes "您沒有";
				mes "足夠的 Zeny 嗎？！";
				mes "您最好開始";
				mes "去打獵賺錢，";
				mes "而不是打怪物了?";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[獵鷹飼養員]";
				mes "天啊?";
				mes "去學習如何";
				mes "駕馭獵鷹";
				mes "吧！我不能把牠租給您，";
				mes "如果您不知道如何";
				mes "掌控牠的話。";
				close;
			}
			else if(checkfalcon()) {
				mes "[獵鷹飼養員]";
				mes "嗯...";
				mes "您已經有";
				mes "一隻獵鷹了。牠就在";
				mes "那裡，您沒看到嗎？";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[獵鷹飼養員]";
			mes "等、等等，您要";
			mes "去哪裡？這些";
			mes "獵鷹可是最頂級的，";
			mes "我掛保證！來吧?";
			close;
		}
	} else {
		mes "愚蠢的年輕人！";
		mes "獵鷹只能由";
		mes "獵人";
		mes "和神射手使用，懂嗎？";
		mes "...嘿嘿，嫉妒嗎？";
		close;
	}
}
