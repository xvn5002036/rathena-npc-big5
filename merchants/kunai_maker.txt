		switch(select("10個 猛毒苦無:10個 冰柱苦無:10個 朔風苦無:10個 黑土苦無:10個 爆炎苦無:取消")) {
		case 1: callfunc "Kunai_Trade",13250,20,7524,1,13259;

function	script	Kunai_Trade	{
	setarray .@item_req[0], getarg(0), getarg(2);
	setarray .@req_amount[0], getarg(1), getarg(3);
	.@item_id = getarg(4);
	mes "[卡辛]";
	mes "你可以用";
	mes .@req_amount[0] +" 個 "+ getitemname(.@item_req[0]) +" 和";
	mes .@req_amount[1] +" 個 "+ getitemname(.@item_req[1]) +" 來交換";
	mes "一組 10 個 "+ getitemname(.@item_id) +"。";
	next;
	mes "[卡辛]";
	mes "我一次最多只能給你 500 組苦無。";
	mes "如果你想取消交易，請輸入『0』。請問你";
	mes "需要幾組苦無呢？";
	next;
	input .@amount;
	if(.@amount < 1 || .@amount > 500) {
		mes "[卡辛]";
		mes "咦？很抱歉，但是";
		mes "我沒辦法給你那麼多";
		mes "苦無。請輸入";
		mes "小於 500 的數字。";
		close;
}
	if (countitem(.@item_req[0]) < .@amount*.@req_amount[0] || countitem(.@item_req[1]) < .@amount*.@req_amount[1]) {
		mes "[卡辛]";
		mes "嗯，你沒有";
		mes "足夠的物品來進行";
		mes "這次的苦無交換。請";
		mes "再檢查一下你的物品。";
		close;
 	}
	if (checkweight(.@item_id, .@amount*10) == 0) {
		mes "[卡辛]";
		mes "嗯，看來你的";
		mes "物品欄沒有";
		mes "足夠的空間來存放";
		mes "更多物品了。你最好";
		mes "先騰出一些空間。";
		close;
	}
	mes "[卡辛]";
	mes "太好了，一切都";
	mes "沒問題，那麼我們就";
	mes "來完成這筆交易吧。";
	mes "我相信你一定會對";
	mes "這些苦無感到";
	mes "非常滿意。";
	delitem .@item_req[0],.@req_amount[0]*.@amount;
delitem .@item_req[1],.@req_amount[1]*.@amount;
	getitem .@item_id,10*.@amount;
	close;
}
