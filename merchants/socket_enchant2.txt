				mes "你最好祈禱能有成功的結果。";

				set .@a, rand(1,100);

					specialeffect EF_LORD;
					mes "[雷亞布連]";

					getitem getarg(1),1;
					next;
					mes "[雷亞布連]";
					mes "下次見，朋友！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[雷亞布連]";
					mes "哇！...非常抱歉，失敗了。";
					mes "不過，我是完全無辜的。";
					mes "這是你的運氣，是上天註定的，好嗎？";
					mes "別太失望了，";
					mes "下次再試試吧。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];

						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[雷亞布連]";
					mes "下次見，朋友！";

			}
			else
			{
				mes "[雷亞布連]";
				mes "你是笨蛋還是怎樣？你根本沒有備齊所有需要的物品！";
				mes "快去把它們帶來！";
				close;
			}
		case 2:
			mes "[雷亞布連]";
			mes "下次見。";

	}
