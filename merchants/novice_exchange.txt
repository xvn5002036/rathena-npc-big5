prontera,123,102,5	script	商人#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
		mes "[商人]";
		mes "哈哈！";
		mes "你是超人嗎？";
		mes "你帶了這麼多東西！";
		mes "你最好把一些";
		mes "放進卡普拉倉庫裡?";
		close;
	}
	mes "[商人]";
	mes "你好！";
	mes "你收集到加勒結晶或柔毛";
	mes "這類的物品了嗎？";
	mes "那傑勒比結晶呢？喔對了，";
	mes "我需要那些東西來做些事情。";
	next;
	mes "[商人]";
	mes "當然，我不會要求你";
	mes "免費給我那些東西。如果我用 1 瓶紅色藥水交換 5 個加勒結晶、10 個柔毛，";
	mes "或 10 個傑勒比結晶，你覺得如何？";
	next;
	switch(select("好啊！:我覺得這是敲竹槓！:不，謝謝。")) {
	case 1:
		mes "[商人]";
		mes "那麼你想帶";
		mes "哪種物品給我？";
		mes "加勒結晶、柔毛還是傑勒比結晶？";
		next;
		switch(select("加勒結晶:傑勒比結晶:柔毛:取消")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[商人]";
			mes "好喔?";
			mes "沒問題。";
			close;
		}
	case 2:
		mes "[商人]";
		mes "敲竹槓...？";
		mes "如果你去查查這些";
		mes "交易物品的市價，你就會知道實際上被";
		mes "敲竹槓的人是我。";
		close;
	case 3:
		mes "[商人]";
		mes "好吧，";
		mes "沒問題。";
		mes "但如果你改變主意的話，";
		mes "隨時歡迎再來找我。";
		close;
	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[商人]";
		mes "嘿，你答應要給我的";
		mes "所有 "+ getitemname(.@item_req) +" 都在哪裡？";
		mes "快把 "+ getitemname(.@item_req) +" 給我！";
		close;
	}
	else {
		mes "[商人]";
		mes "好吧，讓我檢查一下";
		mes "你身上有多少個 "+ getitemname(.@item_req) +"。嗯...";
		next;
		mes "[商人]";
		mes "你總共有";
		mes ""+ countitem(.@item_req) +" 個 "+ getitemname(.@item_req) +"。";
		mes "這些我可以給你總共";
		mes ""+ (countitem(.@item_req)/.@req_amount) +" 瓶紅色藥水。";
		next;
		mes "[商人]";
		mes "你覺得如何？";
		mes "我們要交易嗎？";
		next;
		if (select("成交。:不交易。") == 1) {
			mes "[商人]";
			mes "你知道你想用那些";
			mes ""+ getitemname(.@item_req) +" 換取";
			mes "多少瓶紅色藥水，對吧？";
			next;
			mes "[商人]";
			mes "現在，我一次最少可以跟你";
			mes "交換 1 瓶紅色藥水，";
			mes "最多 100 瓶。如果你改變主意，";
			mes "只要輸入『0』就能取消。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[商人]";
					mes "什麼...？";
					mes "那你幹嘛還來找我講話？";
					mes "你還真是優柔寡斷啊，不是嗎？";
					close;
				}
				else if (.@input > 100) {
					mes "[商人]";
					mes "我一次不能給你超過";
					mes "100 瓶紅色藥水。";
					mes "我們再來一次。";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[商人]";
				mes "哎呀，你輸入的數字似乎不對。你最好再檢查一下你能交換的紅色藥水數量。";
				close;
			}
			mes "[商人]";
			mes "給你！";
			mes "檢查一下我給你的紅色藥水";
			mes "數量，應該沒錯吧。謝謝，這真是一筆好交易?";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[商人]";
		mes "蛤?";
		mes "好吧。";
		mes "不過紅色藥水";
		mes "對像你這樣的冒險者來說";
		mes "不是更有用嗎？";
		close;
	}
}
