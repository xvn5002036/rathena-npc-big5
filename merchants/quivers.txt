	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[發明家 賈克斯]";
		mes "嘿，你身上帶了";
		mes "太多東西了。你為什麼不把它們存進卡普拉倉庫呢？等你存好我們再來談，好嗎？";
		close;
	}
	mes "[發明家 賈克斯]";
	mes "我的名字是賈克斯。";
	mes "我可以毫不自誇地";
	mes "說，我也許是";
	mes "^663300我們這個時代最偉大的發明家^000000。";
	next;
	mes "[發明家 賈克斯]";
	mes "這次，我";
	mes "創造了一件";
	mes "真正非凡的作品。我稱它們為... ^663300魔法箭矢筒^000000！！這將會作為一場";
	mes "箭矢革命被載入史冊！";
	emotion ET_BEST;
	next;
	mes "[發明家 賈克斯]";
	mes "我研究魔法和箭矢筒好幾年了，日以繼夜地工作，直到我終於弄清楚如何用魔法將箭矢壓縮！有了魔法箭矢筒，你就能攜帶更多的箭矢，但重量卻更輕！";
	next;
	mes "[發明家 賈克斯]";
	mes "你想試用看看我的箭矢筒嗎？我深信像你這樣的人一定能欣賞我的才華！";
	next;
	switch(select("箭矢筒:鐵箭矢筒:鋼鐵箭矢筒:神之金屬箭矢筒:火箭矢筒:銀箭矢筒:風箭矢筒:岩石箭矢筒:水晶箭矢筒:無形箭矢筒:影子箭矢筒:生鏽箭矢筒")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;

	//This doesn't seem to be accessable...

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[發明家 賈克斯]";
		mes "太棒了！";
		mes "你有帶任何箭矢在身上嗎？我會提供你一個箭矢筒，它可以裝下 "+ .@amount +" 支你的 "+ getitemname(.@item_id) +"，而且只要 ^FF3131"+ .@zeny_req +" Zeny^000000。";
		next;
		switch(select("盡可能把箭矢裝進箭矢筒裡:購買 1 個箭矢筒:取消")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "箭矢數量：";
			mes "^3131FF"+.@arrows+" ^000000";
			mes "可購買的箭矢筒";
			mes "最大數量：^3131FF"+.@quiver+" ^000000";
			mes "需要 Zeny：^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[發明家 賈克斯]";
			mes "你想用你現在";
			mes "身上帶的箭矢，";
			mes "盡可能購買最多的";
			mes "箭矢筒嗎？";
			next;
			if (select("是:取消") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[發明家 賈克斯]";
					mes "給你！";
					mes "只要記住，^FF0000當你的負重量達到最大負重量限制的 90% 時，你將無法使用箭矢筒^000000。";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[發明家 賈克斯]";
					mes "那麼...";
					mes "只要隨時注意";
					mes "你攜帶的重量，";
					mes "應該就沒問題了。";
					close;
				}
				else {
					mes "[發明家 賈克斯]";
					mes "很抱歉，你的 Zeny 不夠。我花了好幾年開發這項革命性的技術，";
					mes "可不能就這樣白白送人！";
					close;
				}
			}
			mes "[發明家 賈克斯]";
			mes "你改變主意了？";
			mes "擁有箭矢筒的榮耀明明";
			mes "就近在眼前了？";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[發明家 賈克斯]";
				mes "給你！";
				mes "只要記住，^FF0000當你的負重量達到最大負重量限制的 90% 時，你將無法使用箭矢筒^000000。";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[發明家 賈克斯]";
				mes "那麼...";
				mes "只要隨時注意";
				mes "你攜帶的重量，";
				mes "應該就沒問題了。";
				close;
			}
			else {
				mes "[發明家 賈克斯]";
				mes "你連";
				mes ""+ .@zeny_req +" Zeny 都沒有？";
				mes "真是太遺憾了。我沒想到";
				mes "你竟然這麼...";
				mes "^333333貧窮^000000。";
				close;
			}
		case 3:
			mes "[發明家 賈克斯]";
			mes "什麼...？";
			mes "你沒看出這項發明將永遠改變攜帶箭矢的方式嗎？！未來就在現在！";
			close;
		}
	}
	else {
		mes "[發明家 賈克斯]";
		mes "這個箭矢筒最多可以裝 "+ .@amount +" 支箭矢。它是用我的獨家秘方製作的，";
		mes "所以箭矢加上箭矢筒的總重量，會比單獨攜帶箭矢還要輕。";
		next;
		mes "[發明家 賈克斯]";
		mes "這簡直是科學奇蹟！只要你帶至少 "+ .@amount +" 支箭矢和每個箭矢筒所需的 "+ .@zeny_req +" Zeny 來找我，你就能親身體驗。";
		close;
	}
}
