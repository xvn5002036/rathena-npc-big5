	next;
	mes "[美髮師 喬娃維琪]";

	else {

			mes "咦！？天啊！噢，不行不行不行不行！你的頭髮受損得太嚴重了！如果放任不管可不太好。";
			next;
			mes "[美髮師 喬娃維琪]";
			mes "能讓我為你護理頭髮嗎？拜託嘛？";
		}
		else mes "哇！如果染個顏色，你的頭髮就完美了?要不要染個頭髮換換心情？";
	}
	next;
	while(1) {
		switch(select("染髮:提示與資訊:取消")) {
		case 1:
			mes "[美髮師 喬娃維琪]";
			mes "對對，好選擇?";
			mes "那麼，這是你改頭換面的機會，也是你綻放美麗的機會！選擇你想要的顏色吧。";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[美髮師 喬娃維琪]";
					mes "你覺得怎麼樣？你想換個顏色嗎？";
					next;
					if (select("是的:不用了") == 1) {
						mes "[美髮師 喬娃維琪]";
						mes "好的！選擇你想要的顏色吧。";
						next;
					}
					else  {
						mes "[美髮師 喬娃維琪]";
						mes "嗯，我有點失望。我本來想做得更好的。但我保證下次會做得更好。歡迎下次光臨?";
						close;
					}
				}

				while(1) {
					switch(select("紅色，麻煩了。:黃色，麻煩了。:紫色，麻煩了。:橘色，麻煩了。:綠色，麻煩了。:藍色，麻煩了。:白色，麻煩了。:黑色，麻煩了。:其實，我喜歡現在的頭髮。")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[美髮師 喬娃維琪]";
							mes "你一定很喜歡你的髮色?";
							close;
						}
						mes "[美髮師 喬娃維琪]";
						mes "咦?？你不打算染髮嗎？我有點難過......";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[美髮師 喬娃維琪]";
						mes "咦？但這就是你現在的髮色呀。請選擇一個不同的顏色。";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"yellow"; break;
						case 2: callsub S_NoDye,978,"violet"; break;
						case 3: callsub S_NoDye,980,"orange"; break;
						case 4: callsub S_NoDye,979,"green"; break;
						case 5: callsub S_NoDye,981,"blue"; break;
						case 6: callsub S_NoDye,982,"white"; break;
						case 7: callsub S_NoDye,983,"black"; break;
						case 8: callsub S_NoDye,975,"red"; break;
						}
						if (Zeny < 1000) {
							mes "[美髮師 喬娃維琪]";
							mes "費用是 1000 Zeny。你的錢不夠嗎......？";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[美髮師 喬娃維琪]";
			mes "當你心情低落時，當你被甩時，當你想給某人留下深刻印象時，甚至當你只是想引人注目時......";
			next;
			mes "[美髮師 喬娃維琪]";
			mes "為了那個特別的地點和時間，你難道不想要一個專屬於你的髮型嗎？只要你有合適的染料，我就會讓你的頭髮看起來光彩照人。";
			next;
			mes "[美髮師 喬娃維琪]";
			mes "而且別太擔心費用的問題。獲得美麗就如同獲得一切一樣。呵呵?";
			next;
			mes "[美髮師 喬娃維琪]";
			mes "* 費用與資訊 *";
			mes "- 1 個你想要染的髮色的染料。";
			mes "- 1000 Zeny 費用。";
			close;

		case 3:
			mes "[美髮師 喬娃維琪]";
			mes "無論是男人還是女人......";
			mes "每個人都有追求美麗的權利與義務。";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[美髮師 喬娃維琪]";
		mes "咦？！但你需要物品『"+getitemname(getarg(0))+"』才能把頭髮染成 "+getarg(1)+" 色......";

	}
