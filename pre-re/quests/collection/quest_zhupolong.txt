lou_dun02,161,264,4	script	屠龍者	732,{

	if (checkquest(60182) != -1) {
		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[屠龍者]";
			mes "你已經狩獵完 50 隻土龍了嗎？";
			next;
			switch(select("是:否:我想要放棄")) {
			case 1:
				mes "[屠龍者]";
				mes "嗯... 我覺得你應該還沒狩獵完 50 隻喔。";
				close;
			case 2:
				mes "[屠龍者]";
				mes "請記住，我需要狩獵土龍的幫忙，所以請去狩獵 50 隻吧。";
				close;
			case 3:
				mes "[屠龍者]";
				mes "你確定要停止狩獵嗎？";
				mes "你目前所有的進度將會被清除。";
				next;
				switch(select("是:否")) {
				case 1:
					mes "[屠龍者]";
					mes "好吧，如果你改變主意了再回來。";
					erasequest 60182;
					close;
				case 2:
					mes "[屠龍者]";
					mes "為了讓這裡成為人們可以前來獲取經驗值的地方，請擊敗 50 隻土龍。";
					close;
				}
			}
		}
		mes "[屠龍者]";
		mes "太棒了，你的速度真快。";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[屠龍者]";
		mes "哦，太好了！";
		mes "你是來幫我的，對吧？";
		mes "如果不是，我真不知道該拿這個地方的怪物怎麼辦！";
		next;
		switch(select("當然:不要")) {
		case 1:
			mes "[屠龍者]";
			mes "太棒了！我看到你的那一刻就知道我可以指望你！";
			mes "我需要你減少目前在這張地圖上肆虐的怪物數量！";
			mes "這附近有太多的土龍，導致低等級的玩家無法在這裡好好獲得經驗值。";
			next;
			mes "[屠龍者]";
			mes "那麼，你還願意幫我嗎？";
			next;
			switch(select("當然:取消")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[屠龍者]";
					mes "非常感謝你幫忙對付土龍！";
					mes "目前的狀況已經穩定下來，我不再需要你幫忙殺牠們了。";
					close;
				}
				mes "[屠龍者]";
				mes "去擊敗 50 隻土龍，完成後讓我知道。我會為你的努力提供獎勵。";
				setquest 60182;
				close;
			case 2:
				mes "[屠龍者]";
				mes "如果你改變主意了，請再回來。";
				close;
			}
		case 2:
			mes "[屠龍者]";
			mes "如果你改變主意了，請再回來。";
			close;
		}
	}
	mes "[屠龍者]";
	mes "天啊，天啊！";
	next;
	mes "[屠龍者]";
	mes "抱歉，我現在沒辦法說話。我正處於極度的痛苦之中，這些該死的土龍快把我殺了。";
	close;
}
