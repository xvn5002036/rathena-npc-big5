				next;

					if (.@input == 8472)
						set .@magic_powder,1;
					else if (.@input == 3735)
						set .@magic_powder,2;
					else if (.@input == 2750)
						set .@magic_powder,3;
					else if (.@input == 5429)
						set .@magic_powder,4;
					else
						set .@magic_powder,5;
				}
				break;
			}
		}
		mes "[混合機]";
		mes "選擇一顆";
		mes "觸媒礦石。";
		next;
		switch(select("黃色魔力礦石。:紅色魔力礦石。:藍色魔力礦石。:1克拉鑽石。:跳過。")) {
		case 1: set .@mixitem3,1; break;
		case 2: set .@mixitem3,2; break;
		case 3: set .@mixitem3,3; break;
		case 4: set .@mixitem3,4; break;
		case 5: break;
		}
		mes "[混合機]";
		mes "設定完成。";
		mes "啟動";
		mes "混合程序。";
		mes "請稍候。";
		next;
		mes "[混合機]";
		mes "- 每日一句 -";
		switch(rand(1,5)) {
		case 1:
			mes "以眼還眼：當你從別人那裡拿走東西時，你必須以某種方式替換或償還。";
			break;
		case 2:
			mes "信譽是人的貨幣：真正的信任具有無法估量的價值。";
			break;
		case 3:
			mes "善有善報，惡有惡報：最終，你對待別人的方式將會回報到自己身上。";
			mes "這意味著「當你傷害別人時，在不可避免的情況下你也會被他傷害」。";
			break;
		case 4:
			mes "遠親不如近鄰：當你需要幫助時，你可以指望身邊的人。";
			break;
		default:
			mes "物以類聚：你可以從一個人的朋友看出他們的性格。";
		}
		next;
		if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 1 && .@mixitem2 == 1 && .@mixitem3 == 1 && .@magic_powder == 1) {
			mes "[混合機]";
			mes "魔法師測試1號混合液。";
			getitem 1071,1; //Mage_Test_1
			next;
		}
		else if (.@mixitem1_1 == 3 && .@mixitem1_2 == 1 && .@mixitem1_3 == 1 && .@mixitem2 == 0 && .@mixitem3 == 2 && .@magic_powder == 2) {
			mes "[混合機]";
			mes "魔法師測試2號混合液。";
			getitem 1085,1; //Mage_Test_2
			next;
		}
		else if (.@mixitem1_1 == 6 && .@mixitem1_2 == 1 && .@mixitem1_3 == 0 && .@mixitem2 == 1 && .@mixitem3 == 3 && .@magic_powder == 3) {
			mes "[混合機]";
			mes "魔法師測試3號混合液。";
			getitem 1086,1; //Mage_Test_3
			next;
		}
		else if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 0 && .@mixitem2 == 2 && .@mixitem3 == 4 && .@magic_powder == 4) {
			mes "[混合機]";
			mes "魔法師測試4號混合液。";
			getitem 1087,1; //Mage_Test_4
			next;
		}
		else {
			mes "[混合機]";
			mes "發生";
			mes "未知的錯誤。";
			getitem 1090,1;
			//Mage_Test_Etc
			next;
		}
		mes "[混合機]";
		mes "混合完成。";
		mes "謝謝你。";
		close;
	}
	close;
}
