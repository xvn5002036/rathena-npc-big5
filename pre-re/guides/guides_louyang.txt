louyang,213,213,3	script	士兵#BA::LouGuide	825,{
	mes "[士兵]";
	mes "歡迎來到洛陽，";
	mes "這是一座擁有悠久";
	mes "且豐富歷史的城市。";
	next;
	mes "[士兵]";
	mes "最近我們開通了";
	mes "一條海上航線，";
	mes "以促進與外國的";
	mes "積極交流。";
	next;
	mes "[士兵]";
	mes "除了豐富的歷史之外，洛陽也以各種特產聞名。在這裡你可以找到許多我們這片土地獨有的東西。";
	next;
	mes "[士兵]";
	mes "請慢慢參觀，";
	mes "我們誠摯地邀請您享受";
	mes "您在洛陽的旅程。";
	next;
	switch(select("詢問建築物位置。:消除小地圖上的所有標記。:取消。")) {
	case 1:
		mes "[士兵]";
		mes "請問您想去哪裡呢？";
		next;
		switch(select("龍之城:醫館:市政廳:武器店:道具店:酒館:取消")) {
		case 1:
			mes "[士兵]";
			mes "龍之城位在 ^FF3355+^000000 的位置。那是包含我們領主在內，所有貴族居住的地方。";
			next;
			mes "[士兵]";
			mes "既然您是外地人，我想您最好先去拜訪一下我們的領主。";
			viewpoint 1,218,255,2,0xFF3355;
			break;
		case 2:
			mes "[士兵]";
			mes "我們有一位醫術非常高明的醫生。";
			mes "您可以在 ^CE6300+^000000 找到她的醫館。";
			next;
			mes "[士兵]";
			mes "據說沒有";
			mes "她治不好的病。不過，我也不能保證那是不是真的。";
			viewpoint 1,263,94,3,0xCE6300;
			break;
		case 3:
			mes "[士兵]";
			mes "我們有一座負責聯邦政府運作的市政廳。";
			mes "它位在 ^A5BAAD+^000000 的位置。";
			next;
			mes "[士兵]";
			mes "如果您有任何問題，";
			mes "您應該去和市政廳裡的";
			mes "官員們談談。";
			viewpoint 1,309,80,4,0xA5BAAD;
			break;
		case 4:
			mes "[士兵]";
			mes "武器店位在 ^55FF33+^000000 的位置。";
			next;
			mes "[士兵]";
			mes "您將會看到";
			mes "由洛陽經驗豐富的";
			mes "鐵匠所鍛造出的";
			mes "令人驚嘆的武器。";
			viewpoint 1,145,174,5,0x55FF33;
			break;
		case 5:
			mes "[士兵]";
			mes "道具店位在 ^3355FF+^000000 的位置。";
			next;
			mes "[士兵]";
			mes "知己知彼，";
			mes "百戰百勝！";
			mes "有備無患總比事後後悔來得安全。您何不去看看他們賣些什麼呢？";
			viewpoint 1,135,98,6,0x3355FF;
			break;
		case 6:
			mes "[士兵]";
			mes "當您在旅途中感到疲倦時，我建議您去酒館坐坐。它位在 ^00FF00+^000000 的位置。";
			next;
			mes "[士兵]";
			mes "酒館是個結交";
			mes "其他遊客的好地方，同時也能打聽到一些可能有用的消息。";
			viewpoint 1,280,167,7,0xFF00FF;
			break;
		case 7:
			mes "[士兵]";
			mes "如果您想消除小地圖上的";
			mes "所有標記，請從選單中選擇『消除小地圖上的所有標記。』。";
		}
		break;
	case 2:
		viewpoint 2,218,255,2,0xFF00FF;
		viewpoint 2,263,94,3,0xFF00FF;
		viewpoint 2,309,80,4,0xFF00FF;
		viewpoint 2,145,174,5,0xFF00FF;
		viewpoint 2,135,98,6,0xFF00FF;
		viewpoint 2,280,167,7,0xFF00FF;
		mes "[士兵]";
		mes "好了，我已經把您小地圖上的標記都消除了。如果您需要，隨時可以來問我建築物的位置。";
		break;
	case 3:
		mes "[士兵]";
		mes "我想有時候";
		mes "自己去探索";
		mes "也是挺有趣的。請多保重。";
	}
	close;
}
