alberta,212,202,4	script	德瓦他水手#alberta	536,{
	mes "[德瓦他水手]";
	mes "你想去^8B4513德瓦他島^000000嗎？";
	mes "那裡有著耀眼的海浪和迷人的景色，是個放鬆的好地方。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "搭船費用是 10,000 Zeny，VIP 則是 1,000 Zeny。";
	} else {
		set .@cost,10000;
		mes "搭船費用是 10,000 Zeny。";
	}

	switch(select("好！:不用了。")) {
	case 1:
		if (Zeny >= .@cost) {
			mes "[德瓦他水手]";
			mes "^8B4513德瓦他^000000是個美麗又和平的島國。";
			mes "祝你旅途愉快?";
			close2;
			set Zeny, Zeny - .@cost;

			end;

			mes "[德瓦他水手]";
			mes "你的錢不夠哦，孩子。";
			mes "等你有錢再來吧。";
			close;
		}
	case 2:
		mes "[德瓦他水手]";
		mes "讓我知道";
		mes "你什麼時候準備好要去旅行?";
		close;
	}
}
