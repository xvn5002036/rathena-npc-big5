ecl_in02,164,56,3	script	接待員#Laphine	835,{
	mes "[接待員]";
	mes "歡迎來到埃克拉珠診所。";
	next;
	switch (select("儲存位置:休息:結束對話")) {
	case 1:
		mes "[接待員]";
		mes "診所已經設定為你的儲存點。不過路上要小心喔?";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[接待員]";
		mes "在我們的診所恢復你的身心吧。這裡充滿了埃克拉珠純淨的能量。";
		mes "我們通常收費 1 個斯普蘭迪迪硬幣，但我們為旅行者提供特別優惠。總共是 5,000 Zeny。 ";
		next;
		switch (select("用斯普蘭迪迪硬幣支付:用 Zeny 支付:算了")) {
		case 1:
			if (!countitem(6081)) {
				mes "[接待員]";
				mes "看來你沒有足夠的硬幣。麻煩你再確認一下好嗎？";
				close;
			}
			mes "[接待員]";
			mes "就把這裡當自己家吧?";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[接待員]";
				mes "看來你沒有足夠的財產。麻煩你再確認一下好嗎？";
				close;
			}
			mes "[接待員]";
			mes "就把這裡當自己家吧?";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
