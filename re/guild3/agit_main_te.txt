function	script	F_flag_woe_TE	{
	.@castle$ = getarg(1);
	.@guild_id = getcastledata( .@castle$,CD_GUILD_ID );
	if (.@guild_id == 0) {
		mes "[ 神聖盧恩米德加茲王國詔書 ]";
		mes " ";
		mes "1. 遵照神聖盧恩米德加茲王國的法令，";
		mes "我們宣布";
		mes "這座城池目前沒有正式的主人。";
		mes " ";
		mes "2. 對於能夠";
		mes "克服所有考驗";
		mes "並破壞華麗金屬的人，";
		mes "國王將賦予該名勇士";
		mes "這座城池的擁有權。";
		close;
	}
	if (.@guild_id == getcharid(2) && getarg(0)) {
		mes "[ 迴盪的聲音 ]";
		mes "勇敢的人們啊……";
		mes "你們希望回到專屬你們的榮耀之地嗎？";
		next;
		if (select( "返回公會城池。","離開。" ) == 1) {
			if (jobcanentermap(.@castle$) == 0) {
				mes "三轉職業和擴充職業的玩家不允許參加訓練型攻城戰 (TE)。";
				close;
			}
			if (getcastledata( getarg(1),CD_GUILD_ID ) == getcharid(2))
				warp getarg(1),getarg(2),getarg(3);
			end;
		}
	}
	.@guildname$ = getguildname(.@guild_id);
	.@guildmaster$ = getguildmaster(.@guild_id);
	mes "[ 神聖盧恩米德加茲王國詔書 ]";
	mes " ";
	mes "1. 遵照神聖盧恩米德加茲王國的法令，";
	mes "我們認可這個地方屬於";
	mes "^ff0000"+.@guildname$+"^000000 公會的私人財產。";
	mes " ";
	mes "2. ^ff0000"+.@guildname$+"^000000 公會的會長是";
	mes "^ff0000"+.@guildmaster$+"^000000";
	mes "如果有人對此有異議，";
	mes "請用你手中的鋼刃來證明你的力量與榮耀。";
	close;
}
