dali02,51,130,4	script	次元旅行者	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- 由於你攜帶太多物品，無法繼續進行。 -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[次元旅行者]";
	mes "冒險者朋友，我需要的是 " + mesitemlink( 6672 ) + "。要不要和我這裡的東西交換？";
	next;
	.@type = select( "讓我看看你的武器。", "讓我看看你的防具。", "有什麼特別的嗎？", "我不需要。" ) - 1;
	mes "[次元旅行者]";
	switch(.@type) {
	case 0:
		mes "喔！你對武器感興趣嗎？我們擁有能從幽靈皇宮獲得的所有武器。為何不挑選一把呢？";
		break;
	case 1:
		mes "防具？我們擁有能從幽靈皇宮獲得的所有防具。為何不挑選一件呢？";
		break;
	case 2:
		mes "哈哈！看來許多人都已經聽說我有最棒的東西了。隨便看看我的收藏吧。";
		break;
	case 3:
		mes "再見了朋友?歡迎再來！";
		close;
	}
	next;
	mes "[次元旅行者]";
	mes "啊！寫在 " + mesitemlink( 6672 ) + " 旁邊的數字就是需要的數量。";
	mes "請牢記這一點。";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200; 	// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100; 	// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50; 			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";
	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[次元旅行者]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "你是指 " + mesitemlink( .@items[.@s] ) + " 嗎？";
		mes "需要 " + .@items[.@s+1] + " 個 ^006400" + .@item_name_req$ + "^000000 來進行交換。";
		next;
		switch( select( "我們交換吧。", "讓我看看其他東西。", "我晚點再來。" ) ) {
		case 1:
			break;
		case 2:
			mes "[次元旅行者]";
			switch(.@type) {
			case 0:
				mes "我有非常多的武器。";
				break;
			case 1:
				mes "我們有許多防具供你選擇。為何不挑選一件呢？";
				break;
			case 2:
				mes "如果你能帶給我 " + .@item_name_req$ + "，我可以和你交換任何你想要的東西。";
				break;
			}
			continue;
		case 3:
			mes "[次元旅行者]";
			mes "保重了朋友?下次見！";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[次元旅行者]";
			mes "朋友，你讓我有些難辦啊。";
			mes "我需要的是 " + .@item_name_req$ + "。需要用 ^006400" + .@items[.@s+1] + " 個 " + .@item_name_req$ + "^000000 才能交換。";
			close;
		}
		mes "[次元旅行者]";
		mes "謝謝你。如果還需要什麼，隨時回來找我。";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
