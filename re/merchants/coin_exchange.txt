malangdo,220,167,5	script	硬幣兌換商 CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "你的物品種類太多了。請減少一些物品種類後再來。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你的負重太高了，無法繼續。請減輕負重後再來。";
		close;
	}
	mes "[硬幣兌換商 CX-1]";
	mes "歡迎使用 CX-1。如果你有硬幣，可以將它們兌換成較高或較低等級的硬幣。";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "取消:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (無)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[硬幣兌換商 CX-1]";
		mes "謝謝你的光臨。";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[硬幣兌換商 CX-1]";
		mes "你選擇了你沒有的硬幣。";
		mes "謝謝你的光臨。";
		close;
	}
	mes "[硬幣兌換商 CX-1]";
	mes "----兌換比例----";
	mes "兌換高階 - 30 換 10";
	mes "兌換低階 - 10 換 30";
	mes "------------------------";
	mes "^ff3333兌換時，我們會根據當前市場價格收取手續費。^000000";
	next;
	setarray .@exchange_name$[0],"高階","低階";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0;
// Cannot exchange for Silvervine.
	.@menu$ = "取消:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333無法兌換成"+.@exchange_name$[.@i]+"硬幣^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaa兌換成"+.@exchange_name$[.@i]+"硬幣 (不足)^000000:";
		else
			.@menu$ = .@menu$ + "兌換成"+.@exchange_name$[.@i]+"硬幣 - "+getitemname(.@coin)+" (需要 "+.@exchange_rate[.@i]+" 個):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[硬幣兌換商 CX-1]";
		mes "謝謝你的光臨。";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[硬幣兌換商 CX-1]";
		mes "此硬幣無法兌換成"+.@exchange_name$[.@i]+"硬幣。";
		mes "謝謝你的光臨。";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[硬幣兌換商 CX-1]";
		mes "你沒有足夠的硬幣來兌換"+.@exchange_name$[.@i]+"硬幣。";
		mes "謝謝你的光臨。";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[硬幣兌換商 CX-1]";
	mes "收到了 ^003399"+.@exchange_rate[.@i]+"^000000 個 ^003399 "+getitemname(.@coin)+"^000000。";
	mes "我已為你兌換了 ^003399"+.@exchange_name$[.@i]+"硬幣^000000，扣除 ^003399"+.@exchange_loss[(!.@i)]+"^000000 個手續費後，還剩下 ^003399"+.@exchange_total+"^000000 個。";

}
