//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "我是史上最棒的武器工匠！";
	mes "我不精鍊那些普通無聊的物品。";
	mes "我只精鍊精鍊值+10或以上的物品。";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "總之，如果你的物品精鍊值在+10以上，你就可以使用我的服務。";
	mes "你想要我精鍊什麼？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我無法精鍊你身上的任何物品...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你那裡沒有裝備";
		mes "任何我可以";
		mes "精鍊的物品。";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我根本";
		mes "無法精鍊這件物品...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我說過我不處理精鍊值低於+10的物品。";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "我無法再精鍊";
		mes "這個了。這已經是";
		mes "極限了！";
		close;
	}
	.@refineitemid = getequipid(.@part);
// save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "";
		.@type$ = "武器";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "";
		.@type$ = "防具";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "嗯，是一件" + .@type$ + "，確定嗎？";
	mes "如果你想精鍊這件" + .@type$ + "，";
	mes "我需要 1個 ^003366" + getitemname(.@material) + "^000000 和 " + callfunc("F_InsertComma",.@price) + " Zeny。";
	mes "你確定要繼續嗎？";
	next;
	if (select("是:否") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯... 如果你介意的話... 那就算了...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "這件"+.@type$+"已經被精鍊過好幾次了。";
			mes "如果我再嘗試，它可能會被摧毀。";
			mes "它不一定會壞，但還是有小機率會壞。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你可能會^FF0000降低"+.@type$+"的精鍊值^000000，";
			mes "或者如果它壞了，你將會失去附加在上面的^FF0000所有卡片^000000或特殊屬性。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你仍然要我精鍊它嗎？";
			mes "我想我已經給你足夠的警告了。";
			next;
			if(select("是。:否。") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "好吧，不挑戰也是一種選擇...";
				mes "不冒險... 這可能是明智的。";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "嗯。你似乎沒有足夠的錢或"+getitemname(.@material)+"。";
			mes "請等你準備好再來。";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下...";
			mes "你以為我傻嗎？！";
			mes "你趁我不注意時把物品換了！滾出去！";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏗！鏘！鏗！鏘！";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "太好了！成功了！！！";
			mes "我是最棒的武器工匠。";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏘！鏗！鏘！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！";
				mes "精鍊值下降了...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏘！鏗！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！我精鍊失敗了...";
				mes "我不是故意的！";
			}
			mes "即使我是史上最棒的武器工匠，我也會犯錯。";
			mes "這只是運氣不好。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "我下次會做得更好！別擔心！";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以將它精鍊到安定值，或是你想要的次數。由你決定。";
		next;
		.@menu2 = select("請精鍊到安定值。","我要自己決定次數。","我改變主意了...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "你想要我將你的物品精鍊幾次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "我無法將這件物品精鍊這麼多次。";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這將嘗試將裝備精鍊超過安定值 " + .@refinecheck + " 次。你的裝備可能會被摧毀... 這樣可以嗎？";
			next;
			if(select("是的...","不要...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "既然你這麼說... 那就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "既然你這麼說... 那就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "這將花費你 " + .@refinecnt + "個 " + getitemname(.@material) + " 和 " + .@fullprice + " Zeny。可以嗎？";
	next;
	if(select("是:不要...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "既然你這麼說... 那就這樣吧。";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯。你似乎沒有足夠的錢或"+getitemname(.@material)+"。";
		mes "請等你準備好再來。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "看這裡... 你身上沒有裝備任何物品...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;
		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏗... 不，你真的以為我這麼蠢嗎？！";
			mes "你把它換掉了...";
			mes "在我用鐵鎚敲暈你之前滾出去！！";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏗！鏘！鏗！鏘！";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏘！鏗！鏘！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！";
				mes "精鍊值下降了...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏘！鏗！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！我精鍊失敗了...";
				mes "我不是故意的！";
			}
			mes "即使我是史上最棒的武器工匠，我也會犯錯。";
			mes "這只是運氣不好。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "我下次會做得更好！別擔心！";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "太好了！成功了！！！";
	mes "我是最棒的鐵匠。";
	close;
}

	switch(select("神之金屬兌換鈤礦石。:鋁兌換鈣礦石。:高密度鈤礦石兌換鈣礦石。:不用了，謝謝。")) {
	case 1:
		setarray .@i[0],984,3,6224;  //Oridecon -> Bradium
		break;
	case 2:
		setarray .@i[0],985,3,6223;
//Elunium -> Carnium
		break;
	case 3:
		setarray .@i[0],6090,1,6223; //Purified_Bradium -> Carnium
		break;
	case 4:
		mes "[奧斯特里]";
		mes "嗯...";
		close;
	}
	if (countitem(.@i[0]) >= .@i[1] && Zeny >= 50000) {
		delitem .@i[0],.@i[1];
		Zeny = Zeny - 50000;
		getitem .@i[2],1;
		mes "[奧斯特里]";
		if (.@i[0] == 6090) {
			mes "使用高密度鈤礦石精鍊";
			mes "可能會有點貴。";
			mes "我可以將它兌換成一些鈣礦石。";
		} else
			mes "好！這是你的 "+getitemname(.@i[2])+"。";
		mes "拿去好好使用吧。";
		close;
	}
	mes "[奧斯特里]";
	mes "你最好不要因為";
	mes "沒有足夠的錢或";
	mes ""+getitemname(.@i[0])+"";
	mes "而試圖欺騙我。";
	close;
}

	next;
	if(select("是的，我有！！:那就算了吧！！") == 2) {

	if (getequippercentrefinery(.@part) < 100) {
