		mes "很神奇吧？哈哈哈！";

		mes "那是因為我完全不知道它們是怎麼運作的。";
		mes "哈哈哈！";
		next;
		mes "[獵鷹笛商人]";
		mes "當然，你需要具備^ff0000馴鷹術^000000技能。";
		mes "我沒有這個技能，所以我不知道怎麼駕馭獵鷹。";
		mes "照目前來看，我可能永遠都不會有一隻獵鷹。";
		next;
		mes "[獵鷹笛商人]";
		mes "但我不能免費送給你！";
		mes "我也要吃飯，所以我必須賺錢！";
		next;
		mes "[獵鷹笛商人]";
		mes "如果你需要獵鷹笛，我可以用 12,500 zeny 賣給你。還是覺得太貴了？";
		next;
		if (eaclass()&EAJL_THIRD) {
			mes "[獵鷹笛商人]";
			mes "不過，如果是遊俠的話，我算你特價 10,000 zeny！";
			set .@price,10000;
			next;
		} else
			set .@price,12500;
		if(select("不買:購買獵鷹笛") == 1) {
			mes "[獵鷹笛商人]";
			mes "真可惜。如果你需要的話，隨時回來找我！";
			close;
		}
		if (Zeny >= .@price) {
			if (countitem(12848) >= 1) {
				mes "[獵鷹笛商人]";
				mes "我看你已經有一把獵鷹笛了！";
				mes "別擔心，年輕的朋友，獵鷹笛買一次就夠了，不需要再買，";
				mes "因為獵鷹笛使用後並不會消失，這不是很棒嗎？";
				close;
			}
			mes "[獵鷹笛商人]";
			mes "來，拿著這把笛子。";
			mes "有了它，你就能在任何地方吹哨呼喚你的獵鷹。";
			set Zeny, Zeny - .@price;
			getitem 12848,1; //Falcon_Flute
			next;
			mes "[獵鷹笛商人]";
			mes "提醒你一下。";
			mes "你需要具備^ff0000馴鷹術^000000技能才能使用它！";
			close;
		}
		mes "[獵鷹笛商人]";
		mes "很抱歉，你沒有足夠的錢購買獵鷹笛，請等你有更多錢的時候再來吧！";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "你好，年輕人！";
		mes "目前我幫不上你的忙，";
		mes "不如等你成為一名偉大的獵人，並學會如何正確駕馭獵鷹後再來找我吧？";
		close;
	}
	mes "今天真是個美好的一天，不是嗎？";
	close;
}
