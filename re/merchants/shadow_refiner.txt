//===== rAthena Script =======================================
//= Shadow Blacksmith
//===== Description: =========================================
//= [Official Conversion]
//= Shadow equipments refining NPC.
//===== Changelog: ===========================================
//= 1.0 First version [Aleos]
//= 1.1 Removed re-roll behavior and fetch materials from db
//=     [Secret]
//= 1.2 Added Refine UI [Atemo, Lemongrass]
//= 1.3 Moved to Cash Mall [Lemongrass]
//============================================================
-	script	::ShadowBlacksmith	-1,{
	if( getbattleflag( "feature.refineui" ) ){
		mes "[影子鐵匠]";
		mes "現在巴斯塔負責影子道具的精煉。";
		mes "如果你想精煉影子道具，請去找他。";
		close;
	}

	.@zeny_cost = 200000;
	// Zeny cost is 200,000 according to official script [Secret]
	disable_items;
	mes "[影子鐵匠]";
	mes "你想精煉影子裝備嗎？挑一個吧！";
	next;
	setarray .@indices[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
	.@indlen = getarraysize(.@indices) - 1;
	for(.@i = 1; .@i <= .@indlen; .@i++)
		.@menu$ = .@menu$ + (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[未裝備]") +":";
	.@menu$ = .@menu$ + "精煉資訊";
	.@choice = select(.@menu$);
	if (.@choice == .@indlen + 1) { // Refine info
		mes "[影子鐵匠]";
		mes "當影子道具被精煉時，它會獲得額外的加成，非常類似於一般道具。";
		next;
		mes "[影子鐵匠]";
		mes "武器：每成功精煉+1，ATK、MATK 就會提升 1。";
		mes "其他：每成功精煉+1，HP 就會提升 10。";
		next;
		mes "[影子鐵匠]";
		mes "各種類型的奧利哈鋼和鋁都可以用來精煉影子裝備。每次嘗試還將花費 20,000 zeny。";
		next;
		mes "[影子鐵匠]";
		mes "高密度礦石可用於精煉值至少達到+7的裝備，只要你一直跟我對話，它就能防止裝備損壞。";
		close;
	}

	.@part = .@indices[.@choice];

	if (!getequipisequiped(.@part)) {
		mes "[影子鐵匠]";
		mes "這裡什麼都沒有！";
		close;
	}
	.@equip_id = getequipid(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	while(1) {
		.@equip_refine = getequiprefinerycnt(.@part);
		mes "[影子鐵匠]";
		mes "每次精煉嘗試我需要收取 " + callfunc("F_InsertComma", .@zeny_cost) + " zeny 作為手續費。";
		mes "選擇你的礦石並開始精煉。";
		next;

		.@material[0] = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@material[1] = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
		.@material[2] = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);
		.@is_enriched = false;

		if (countitem(.@material[0]))
			.@mate$[0] = getitemname(.@material[0]);
		else
			.@mate$[0] = "^8C8C8C"+ getitemname(.@material[0]) +"^000000";
		if (countitem(.@material[1]))
			.@mate$[1] = getitemname(.@material[1]);
		else
			.@mate$[1] = "^8C8C8C"+ getitemname(.@material[1]) +"^000000";
		if (.@equip_refine > 6 && countitem(.@material[2]))
			.@mate$[2] = getitemname(.@material[2]);
		else
			.@mate$[2] = "^8C8C8C"+ getitemname(.@material[2]) +"^000000";

		switch( select( .@mate$[0], .@mate$[1], .@mate$[2], "取消" ) ) {
		case 1:// NORMAL
			.@choose = .@material[0];
			break;
		case 2:// ENRICHED
			.@is_enriched = true;
			.@choose = .@material[1];
			break;
		case 3:// HD
			if (.@equip_refine < 7) {
				mes "[影子鐵匠]";
				mes "高密度礦石只能用於精煉值為+7或更高的裝備。";
				close;
			}
			.@hoihoi = 1;
			.@choose = .@material[2];
			break;
		case 4:
			mes "[影子鐵匠]";
			mes "你已取消精煉。";
			close;
		}
		if (!countitem(.@choose)) {
			mes "[影子鐵匠]";
			mes "你的 "+ getitemname(.@choose) +" 不夠。";
			close;
		}
		if (Zeny < .@zeny_cost) {
			mes "[影子鐵匠]";
			mes "你的 zeny 不夠。";
			close;
		}
		if (.@equip_refine > 9) {
			mes "[影子鐵匠]";
			mes "影子裝備最高只能精煉到 10...";
			close;
		}
		if (!getequipisenableref(.@part)) {
			mes "[影子鐵匠]";
			mes "此裝備無法精煉。";
			close;
		}
		if (getequippercentrefinery(.@part) < 100) {
			mes "[影子鐵匠]";
			mes "影子裝備的安全精煉值為+4。";
			if (!.@hoihoi)
				mes "在隨後的精煉中，影子裝備可能會被摧毀。想要開始嗎？";
			else
				mes "下一次精煉如果失敗，精煉值將會下降。你想要精煉嗎？";
			next;
			if (select("繼續","取消") == 2) {
				mes "[影子鐵匠]";
				mes "嘿，我就知道！";
				close;
			}
		}
		mes "[影子鐵匠]";
		mes "那麼.. 祝好運了！";
		next;
		delitem .@choose,1;
		Zeny -= .@zeny_cost;
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
			close;
		if (getequippercentrefinery(.@part, .@is_enriched) > rand(100)) {
			successrefitem .@part;
			mes "[影子鐵匠]";
			mes "成功了！成功了！";
			next;
		} else {
			if (.@hoihoi)
				downrefitem .@part;
			else
				failedrefitem .@part;
			mes "[影子鐵匠]";
			mes "喔，奧丁不啊！";
			close;
		}
	}
}
