// Warp Scroll Seller
cmd_fild07,375,167,1	script	遺跡黑市商人#pa0829_01	4_F_JOB_HUNTER,{
	mes "[遺跡黑市商人]";
	mes "嘿，每次都要從這裡走進走出是不是很辛苦？只要 ^0000ff20,000^000000 Zeny，你就可以買到一張直接傳送到遺跡入口的卷軸，你覺得怎麼樣？";
Purchase:
	next;
	switch( select( "給我一張", "我不需要" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "你攜帶的物品太多，無法繼續對話。";
				mes "請整理你的物品後再試一次。";
				close;
			}
			else if (Zeny < 20000) {
				mes "[遺跡黑市商人]";
				mes "你好像沒有足夠的錢了。一張卷軸需要 20,000 Zeny。";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[遺跡黑市商人]";
			mes "這是一筆好交易。還需要更多嗎？";
			goto Purchase;
		case 2:
			mes "[遺跡黑市商人]";
			mes "隨時歡迎再來";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	遺物鑑定商#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你攜帶的物品太多，無法繼續對話。";
		mes "請整理你的物品後再試一次。";
		close;
	}
	.@stone_id = 6905;
	mes "[遺物鑑定商]";
	mes "選擇你想要購買的裝備類型。只要你有 " + getitemname(.@stone_id) + "，想買多少都可以。";
	next;
	switch (select("取消:武器:防具")) {
		case 1:
			mes "[遺物鑑定商]";
			mes "隨時歡迎再來?";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "取消:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[遺物鑑定商]";
			mes "隨時歡迎再來?";
			close;

		default:
			.@s--;
			mes "[遺物鑑定商]";
			mes "你需要 ^0000FF" + .@price + "^000000 個 " + getitemname(.@stone_id) + " 才能購買 " + getitemname(.@equip_id[.@s]) + "?";
			next;
			if (select("取消:購買") == 1) {
				mes "[遺物鑑定商]";
				mes "隨時歡迎再來?";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[遺物鑑定商]";
				mes "你沒有足夠的 " + getitemname(.@stone_id) + " 來購買這個物品。";
				close;
			}
			mes "[遺物鑑定商]";
			mes "感謝你的信任。下次請再來。";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	遺物強化師#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你攜帶的物品太多，無法繼續對話。";
		mes "請整理你的物品後再試一次。";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[遺物強化師]";
	mes "如果你想強化在無限空間獲得的裝備，那你來對地方了。";
	mes "你有 " + getitemname(.@stone_id) + " 嗎？";
	next;
	switch (select("我要怎麼附魔裝備？:裝備附魔。:重置裝備附魔。")) {
		case 1:
			mes "[遺物強化師]";
			mes "如果你去探索發樂斯燈塔底下的空間，就能獲得 " + getitemname(.@stone_id) + "。";
			next;
			mes "[遺物強化師]";
			mes "只要有那個材料，我就可以幫你強化從無限空間獲得的裝備。";
			next;
			mes "[遺物強化師]";
			mes "我只能附魔第3跟第4個插槽。";
			next;
			mes "[遺物強化師]";
			mes "在附魔過程中裝備不會有損壞的風險，但在重置附魔時，有很高的機率會損壞。";
			next;
			mes "[遺物強化師]";
			mes "那麼，當你想附魔裝備時再來找我吧。";
			break;
		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[遺物強化師]";
			mes "請選擇你想要附魔的裝備。";
			next;
			switch (select("取消:武器:鎧甲:鞋子:披肩:頭飾")) {
				case 1:
					mes "[遺物強化師]";
					mes "那麼，當你想附魔裝備時再來找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;
				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[遺物強化師]";
				mes "我最多只能附魔到第三個插槽。你的裝備無法再進一步附魔了。";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[遺物強化師]";
			mes "你可以選擇3種類型的附魔，附魔費用是 " + .@fee + " 個 " + getitemname(.@stone_id) + "。";
			mes "我會確保附魔過程順利進行。";
			next;
			.@type = select("退出:物理:魔法:遠距離") - 2;
			mes "[遺物強化師]";
			if (.@slot == 3)
				mes "好的，讓我們開始第一次附魔。";
			else
				mes "好的，讓我們進行第二次附魔。";
			next;
			if (select("我晚點再來。:請繼續。") == 1) {
				mes "[遺物強化師]";
				mes "如果你改變主意，隨時回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[遺物強化師]";
				mes "嗯。順帶一提，我覺得你根本沒在聽，你沒有足夠的 " + getitemname(.@stone_id) + "。";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[遺物強化師]";
			mes "嗯。做得很好。趕快檢查你的裝備吧。";
			break;
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[遺物強化師]";
			mes "請選擇你想要附魔的裝備。";
			next;
			switch (select("取消:武器:鎧甲:鞋子:披肩:頭飾")) {
				case 1:
					mes "[遺物強化師]";
					mes "那麼，當你想附魔裝備時再來找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;
				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[遺物強化師]";
				mes "你的裝備上沒有任何附魔。";
				close;
			}
			mes "[遺物強化師]";
			mes "在重置過程中，你的無限空間裝備有可能會損壞。你仍願意繼續嗎？";
			next;
			if (select("退出:繼續") == 1) {
				mes "[遺物強化師]";
				mes "如果你改變主意，隨時回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[遺物強化師]";
				mes "嗯。順帶一提，我覺得你根本沒在聽，你沒有足夠的 " + getitemname(.@stone_id) + "。";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[遺物強化師]";
				mes "嗯。做得很好。趕快檢查你的裝備吧。";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[遺物強化師]";
				mes "唉，我警告過你了。你的運氣不太好，對吧？";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[遺物強化師]";
			mes "你把裝備脫下來了嗎？";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[遺物強化師]";
			mes "這個裝備不適合附魔。別忘了，只有無限裝備系列才能附魔。";
			close;
		}
		return;
	}
