//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	服裝效果解除服務	4_EP16_COOK2,{
	mes "[德瑞克·肖特]";
	mes "你是否因為服裝狀態而無法參與戰鬥？這一定很困擾吧。";
	next;
	mes "[德瑞克·肖特]";
	mes "雖然穿在你身上很好看，但也可能會變得非常麻煩。我只需要一個手勢就能把它消除！！我會幫你整理好的。如果你準備好了，請選擇你的選項！";
	next;
	switch( select( "保留它", "立刻解除" ) ){
		case 1:
			mes "[德瑞克·肖特]";
			mes "如果你改變心意，隨時回來找我。";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[德瑞克·肖特]";
			mes "解除已完成。確認一下你現在是不是能自由活動了。";
			atcommand "@changedress";
			close;
	}
}
