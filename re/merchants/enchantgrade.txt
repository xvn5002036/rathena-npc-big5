grademk,34,184,4	script	史拉托斯#sratos	4_JP_GARM_H,{
	mes "[史拉托斯]";
	mes "你好啊，走進來的貴客，希望今天能有奇蹟發生。";
	mes "有什麼我可以幫忙的嗎？";
	next;
	switch( select( "強化裝備階級。", "交換埃特爾道具", "什麼是階級強化？" ) ){
		case 1:
			mes "[史拉托斯]";
			mes "我們的客人。你想要強化裝備的階級。";
			mes "要轉移珠寶的魔力可不是一件容易的事。";
			next;
			mes "[史拉托斯]";
			mes "魔力可能會當場爆炸。";
			mes "那麼將要繼承魔力的武器也會跟著爆炸！";
			next;
			switch( select( "我還是要做！", "我再考慮看看。" ) ){
				case 1:
					mes "[史拉托斯]";
					mes "祝我們勇敢的客人好運！";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[史拉托斯]";
					mes "無論何時，當你有勇氣挑戰時，請再回來。";
					close;
			}
		case 2:
			mes "[史拉托斯]";
			mes "這些是用來強化階級的珠寶。";
			mes "將埃特爾星塵和作為基底的珠寶交給我，並支付少許手續費，我就能幫你兌換成埃特爾珠寶。";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[史拉托斯]";
			mes "有時候會發現一些含有大量純粹魔力、狀態不穩定的珠寶。";
			mes "我們把這些稱為埃特爾寶石，對吧？";
			next;
			mes "[史拉托斯]";
			mes "你可以將埃特爾寶石的魔力轉移到其他武器上，但這當然伴隨著一些風險。";
			mes "這過程很微妙，如果一不小心，砰！它就會爆炸。";
			next;
			mes "[史拉托斯]";
			mes "總之，如果你使用這種埃特爾寶石，就可以強化你正在使用的武器。";
			mes "性能會變得更好。我們稱之為階級強化。";
			next;
			mes "[史拉托斯]";
			mes "如果是你，看到心愛的武器更進一步成長，不也會感到開心嗎？";
			mes "我們就是從旁協助的人。雖然有點風險啦。哈哈哈。";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<階級強化師>");
	end;
}
