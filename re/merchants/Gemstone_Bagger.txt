-	script	#GemstoneBagger	HIDDEN_NPC,{
	mes "[魔石打包師]";
	mes "我們會把魔石打包成袋，方便您尋找。";
	mes "您想要做什麼呢？";
	next;
	switch(select("兌換魔石袋:取消")) {
	case 1:
		mes "[魔石打包師]";
		mes "我來準備魔石袋。";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[魔石打包師]";
		mes "如果需要幫助，請再回來。";
		close;
	}
}
