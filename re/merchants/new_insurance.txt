prontera,82,108,5	script	愛心商人#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "你身上攜帶的物品太多，無法進行對話。";
		mes "請整理你的物品後再來找我。";
		close;
	}
	mes "[愛心商人]";
	mes "你好！看得出來你是一位熱愛旅行、搭船橫渡大陸，並享受危險刺激的冒險家！";
	next;
	if(!countitem(.@old_item)){
		mes "[愛心商人]";
		mes "你的倉庫裡有任何" + mesitemlink( .@old_item ) + "嗎？";
		next;
		mes "[愛心商人]";
		mes "如果你帶 " + .@old_item_name$ + " 給我，我會幫你換成 " + mesitemlink( .@new_item ) + "。你覺得如何？你知道這是什麼物品嗎？";
	} else {
		mes "[愛心商人]";
		mes "你有沒有想過把你的 " + mesitemlink( .@old_item ) + " 換成更好的東西？例如... " + mesitemlink( .@new_item ) + "。你覺得如何？";
	}
	next;
	switch(select("這是什麼？",(countitem(.@old_item)?"請幫我兌換！":""),"我沒有興趣")) {
	case 1:
		mes "[愛心商人]";
		mes mesitemlink( .@old_item ) + " 是一種使用後30分鐘內，死亡時不會掉落經驗值的物品。";
		next;
		mes "[愛心商人]";
		mes "但是很多人對這個時間限制感到遺憾，對吧？所以我研究了一下，準備了 " + mesitemlink( .@new_item ) + "。";
		next;
		mes "[愛心商人]";
		mes mesitemlink( .@new_item ) + " 是一種只要帶在身上，^EE0000無論時間長短^000000，死亡時就能防止掉落一次經驗值的物品。當然，死亡後會消耗掉一個 " + .@new_item_name$ + "。";
		next;
		select("你為什麼要開放兌換呢？");
		mes "[愛心商人]";
		mes "你很好奇嗎？我可是個商人。無論是關於生或死，買賣物品就是我的宿命。你覺得我會錯過這個賺錢的機會嗎？我賺到了錢，而活著的人能減輕死亡帶來的痛苦。這不是一筆很棒的交易嗎？";
		next;
		mes "[愛心商人]";
		mes "兌換比例是 5個 " + mesitemlink( .@old_item ) + " 換 1個 " + mesitemlink( .@new_item ) + "。你有什麼問題嗎？";
		next;
		mes "[愛心商人]";
		mes "嗯?你覺得如何？這是一筆很公平的交易吧？你想要一直擔心時間限制嗎？這可是熱愛驚險刺激的冒險家必備的物品喔。";
		close;
		
	case 2:	
		mes "[愛心商人]";
		mes "對、對。明智的決定。換成更好的東西不是比較好嗎？";
		next;
		mes "[愛心商人]";
		mes "我可以將你的 ^0000cd5個 " + .@old_item_name$ + "^000000 換成 1個 " + .@new_item_name$ + "。";
		next;
		mes "[愛心商人]";
		mes "不過，^EE0000[非賣品] " + .@old_item_name$ + " 是無法進行兌換的^000000，所以拿那個來要求兌換是沒有用的。";
		next;
		mes "[愛心商人]";
		mes "我們現在要進行兌換嗎？";
		next;
		if (select("請幫我兌換！:我不想兌換。") == 2)
			break;
		mes "[愛心商人]";
		mes "你想要兌換幾個呢？^0000cd每次最多可以兌換 10個 " + .@new_item_name$ + "^000000。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[愛心商人]";
			mes "嗯？你沒興趣嗎？這可是個好東西，對你這樣的人來說非常實用。不過如果你不想保護自己的性命，我也不會勉強你。";
			close;
		}
		if (.@amount > 10) {
			mes "[愛心商人]";
			mes "就像我說的，^0000cd每次最多只能兌換 10個^000000。請再次確認數量。";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[愛心商人]";
			mes "呃，不好意思... 你的 " + .@old_item_name$ + " 數量不夠喔？是 5個 " + mesitemlink( .@old_item ) + " 換 1個 " + mesitemlink( .@new_item ) + "。請確認一下數量。";
			close;
		}
		mes "[愛心商人]";
		mes "那麼，我已經將你的 ^0000cd" + .@old_item_name$ + "^000000 兌換成 ^EE0000" + .@new_item_name$ + "^000000 了！！這是一筆很划算的交易，對吧？下次如果你又拿到了 " + .@old_item_name$ + "，記得再來找我喔！";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[愛心商人]";
	mes "這絕對是筆好交易！冒險家，對於像你這樣在危險地方冒險的人來說，這可是必備物品啊！";
	next;
	mes "[愛心商人]";
	mes "如果你對它產生了興趣，一定要回來兌換喔。好嗎？";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "你有好東西呢。想不想把你身上的 " + getitemname(12209) + " 換成更好的？", "", bc_self;
	end;
}
