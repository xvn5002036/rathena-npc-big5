job3_rune01,88,62,5	script	龍培育員	105,{
	mes "[龍培育員]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "歡迎。你想要租借一隻龍嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[龍培育員]";
				mes "請先學習如何騎乘龍。";
				close;
			}
			else if (checkdragon()) {
				mes "[龍培育員]";
				mes "你已經有一隻龍了。";
				close;
			}
			else if (ismounting()) {
				mes "[龍培育員]";
				mes "請先解除你的商城坐騎。";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[龍培育員]";
			mes "我明白了。祝你有個美好的一天。";
			close;
		}
	}
	mes "你在這裡做什麼？";
	mes "只有盧恩騎士才能租借龍。";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[魔導戰甲技師]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "歡迎，機械工匠。";
		mes "你想要租借一輛手推車或是";
		mes "搭乘魔導戰甲嗎？";
		next;
		switch(select("租借手推車:搭乘魔導戰甲:購買緊急魔導戰甲:升級冷卻裝置:取消")) {
		case 1:
			if (checkcart()) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你已經";
				mes "有一輛手推車了。";
				close;
			}
			setcart;
			mes "[魔導戰甲技師]";
			mes "給你！";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你已經";
				mes "在搭乘魔導戰甲了。";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[魔導戰甲技師]";
				mes "請先學習獲得魔導戰甲執照的技能。";
				close;
			}
			else if(ismounting()) {
				mes "[魔導戰甲技師]";
				mes "請先解除你的商城坐騎。";
				close;
			}
			setmadogear;
			mes "[魔導戰甲技師]";
			mes "祝你玩得開心，歡迎下次再來！";
			close;
		case 3:
			mes "[魔導戰甲技師]";
			mes "緊急魔導戰甲在緊急情況下非常有用，售價為 1,000,000 Zeny。";
			next;
			if (select("購買:取消") == 2) {
				mes "[魔導戰甲技師]";
				mes "我明白了。如果你改變主意，";
				mes "請隨時來找我。";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你已經有一個緊急魔導戰甲了。";
				close;
			}
			if (Zeny < 1000000) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你沒有足夠的 Zeny 來購買緊急魔導戰甲。";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[魔導戰甲技師]";
			mes "給你！";
			close;
		case 4:
			mes "[魔導戰甲技師]";
			mes "你想要升級哪個裝置？";
			next;
			if (select("冷卻裝置:高品質冷卻器") == 1) {
				mes "[魔導戰甲技師]";
				mes "將冷卻裝置升級為高品質冷卻器需要 1個 冷卻裝置 和 2,000,000 Zeny。";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[魔導戰甲技師]";
				mes "將高品質冷卻器升級為特殊冷卻器需要 1個 高品質冷卻器 和 4,000,000 Zeny。";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("升級:取消") == 2) {
				mes "[魔導戰甲技師]";
				mes "我明白了。如果你改變主意，";
				mes "請隨時來找我。";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你沒有 " + getitemname(.@itemid) + "。";
				close;
			}
			if (Zeny < .@cost) {
				mes "[魔導戰甲技師]";
				mes "很抱歉，但是你沒有足夠的 Zeny 來升級該裝置。";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[魔導戰甲技師]";
			mes "給你！這是你專屬的 " + getitemname(.@itemid) + "。";
			close;
		case 5:
			close;
		}
	}
	mes "有什麼我可以幫你的嗎？";
	mes "魔導戰甲只有機械工匠可以使用。";
	close;
}
