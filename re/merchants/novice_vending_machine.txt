-	script	初學者自動販賣機#main	-1,{
	.@storage_ticket = 7059;	// Cargo_Free_Ticket
	.@storage_ticket_name$ = getitemname(.@storage_ticket);

	mes "初學者專用自動販賣機。";
	mes "上面有許多按鈕。";
	next;
	switch( select( "使用說明", "兌換 " + .@storage_ticket_name$, "兌換初學者裝備" ) ) {
	case 1:
		mes "[兌換 " + .@storage_ticket_name$ + "]";
		mes "4張 " + .@storage_ticket_name$ + " 可以兌換 1張 免費空間移動服務券、免費飛空艇搭乘券或免費手推車使用券，或者是 1張 免費大嘴鳥騎乘券、15個 初學者蒼蠅翅膀、5個 初學者蝴蝶翅膀、30個 初學者專用紅色藥水。";
		next;
		mes "[兌換初學者裝備]";
		mes "從克里圖拉學院獲得的初學者裝甲、初學者兜帽、初學者皮靴、初學者短劍和初學者圓盾，可以兌換成 4張 " + .@storage_ticket_name$ + "。";
		next;
		mes "[兌換初學者裝備]";
		mes "初學者蛋殼帽可以兌換成克里圖拉髮型設計券，用來改變你的髮型。";
		close;
	case 2:
		setarray .@exchange[0],
			7060,1,		// Warp_Free_Ticket
			7311,1,		// Free_Flying_Ship_Ticket
			7061,1,		// Cart_Free_Ticket
			7310,1,		// Free_Peco_Ticket
			12323,15,	// N_Fly_Wing
			12324,5,	// N_Butterfly_Wing
			569,30;		// Novice_Potion
		.@size = getarraysize(.@exchange);
		
		for ( .@i = 0; .@i < .@size; .@i += 2 ) {
			if (.@exchange[.@i+1] > 1)
				.@menu$ += .@exchange[.@i+1] + "個 ";
			.@menu$ += getitemname(.@exchange[.@i]) + ":";
		}
		.@s = (select(.@menu$) - 1) * 2;
		if (countitem(.@storage_ticket) < 4) {
			mes "你的 " + .@storage_ticket_name$ + " 數量不足，無法進行兌換。";
			close;
		}
		if (.@exchange[.@s+1] > 1)
			.@amount$ = .@exchange[.@s+1] + "個 ";
		mes "已將 4張 " + .@storage_ticket_name$ + " 兌換成 " + .@amount$ + getitemname(.@exchange[.@s]) + "。";
		delitem .@storage_ticket,4;
		getitem .@exchange[.@s],.@exchange[.@s+1];
		close;
	case 3:
		mes "將所有初學者裝備兌換成使用券。";
		mes "請選擇繼續以進行兌換。";
		next;
		if (select( "繼續", "取消" ) == 2) {
			mes "兌換已取消。";
			close;
		}
		mes "正在將所有裝備兌換成使用券。";
		setarray .@list[0],2352,2510,2414,1243,2112,5055;
		.@size = getarraysize(.@list);
		
		for ( .@i = 0, .@i < .@size; .@i++ ) {
			.@count = countitem(.@list[.@i]); // note: maybe check if equipped items are skipped
			if (.@count < 1)
				continue;
			switch(.@list[.@i]) {
			case 2352:	// Novice_Plate
			case 2510:	// Novice_Hood
			case 2414:	// Novice_Boots
			case 1243:	// Novice_Knife
			case 2112:	// Novice_Guard
				delitem .@list[.@i], .@count;
				getitem .@storage_ticket,(4*.@count);	// Cargo_Free_Ticket
				break;
			case 5055:	// Novice_Egg_Cap
				delitem .@list[.@i], .@count;
				getitem 6593,.@count;	// Cryptura_Hair_Coupon
				break;
			}
		}
		close;
	}
}
