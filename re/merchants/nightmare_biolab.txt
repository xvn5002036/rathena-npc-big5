function	script	F_Cursed_Fragment	{	// 23016
	.@r = rand(1,12);
	if (.@r == 1) {
		showscript "我絕對不會原諒你...！";
		sc_start SC_ATKPOTION,600000,50;
		sc_start SC_MATKPOTION,600000,50;
	}
	else if (.@r == 2) {
		showscript "誰來救救我...";
		sc_start SC_CURSE,10000,0;
	}
	else if (.@r == 3) {
		showscript "住手... 好痛...";
		sc_start SC_BLEEDING,5000,0;
	}
	else if (.@r == 4) {
		showscript "這裡好冷...";
		sc_start SC_FREEZE,5000,0;
	}
	else if (.@r == 5) {
		showscript "拜託... 留下來陪我...";
		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 6) {
		showscript "留在... 這裡陪我...";
		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 7) {
		showscript "你不該來這裡的...";
		sc_start SC_STUN,5000,0;
	}
	else if (.@r == 8) {
		showscript "哈囉... 有人在嗎...？";
		sc_start SC_BLIND,5000,0;
	}
	else if (.@r == 9) {
		showscript "我好怕... 好怕...";
		sc_start SC_CONFUSION,10000,0;
	}
	else if (.@r == 10) {
		showscript "親愛的奧丁...";
		sc_start SC_ASSUMPTIO,100000,5;
		specialeffect2 EF_ASSUMPTIO;
	}
	else if (.@r == 11) {
		showscript "快醒醒！";
		percentheal 0,20;
	}
	else {
		showscript "對不起... 我要離開了... 謝謝你...";
		getitem 6820,1;// Energy_Fragment
	}
	return;
}

S_Make:
	.@reward_id = getarg(0);
	.@equipment_req = getarg(1);
	.@item_misc_req = getarg(2);
	if (getarg(3,0)) {
		setarray .@string$[0], "寒氣", "斗篷";

	}

		setarray .@material[0],1,20; // soul, energy
	}
	next;
	mes "[悲傷的意念]";
