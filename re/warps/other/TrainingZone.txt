prontera,246,62,3	script	訓練教官#sol1	8W_SOLDIER,{
	mes "[訓練教官]";
	mes "如果你想在更廣闊的區域進行訓練，我可以讓你進入。";
	next;
	mes "[訓練教官]";
	mes "訓練場主要分為兩個區域。";
	next;
	mes "[訓練教官]";
	mes "上半部區域，你可以測試根據魔物體型所造成的傷害。";
	next;
	mes "[訓練教官]";
	mes "下半部區域，你可以測試根據魔物屬性與種族所造成的傷害。";
	next;
	mes "[訓練教官]";
	mes "你想進入哪一個區域？";
	next;
	switch( select( "上半部區域", "下半部區域", "不進入" ) ) {
	case 1:
		mes "[訓練教官]";
		mes "那麼我就為你帶路。";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "[訓練教官]";
		mes "那麼我就為你帶路。";
		close2;
		warp "tra_fild",111,74;
		end;
	case 3:
		mes "[訓練教官]";
		mes "如果你需要引導，再來找我吧。";
		close;
	}
}
