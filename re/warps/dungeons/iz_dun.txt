iz_dun04,130,234,5	script	守門人#iz_dun	413,{
	mes "[守門人]";
	mes "站住！";
	mes "冒險家，你來這裡做什麼？";
	mes "我們發現了這座海底城市的新樓層。";
	next;
	mes "[守門人]";
	mes "你認為你能應付這個挑戰嗎？";
	mes "這裡的怪物等級介於 110 到 124 之間。";
	mes "而且你必須小心克拉肯！";
	next;
	switch(select("釋放克拉肯！:我要留在這裡。")) {
	case 1:
		mes "[守門人]";
		mes "萬事小心，祝你好運。";
		mes "如果你看到克拉肯，一定要找朋友幫忙，或者像我們一樣——轉身快跑。";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[守門人]";
		mes "在探索新區域之前，最好先做好萬全的準備。";
		mes "等你準備好的時候再來吧。";
		close;
	}
}
