tra_fild01,204,110,2	script	訓練場倉庫士兵#fild01_1	8W_SOLDIER,{
	mes "[訓練場倉庫士兵]";
	mes "你需要使用倉庫嗎？在訓練場使用倉庫需要 1000 Zeny。請選擇你要使用的倉庫。";
	next;
	// switch( select( "? 倉庫 1", "? 倉庫 2", "? 倉庫 3", "? 公會倉庫", "? 取消" ) ) {
	switch( select( "? 倉庫 1", "", "", "? 公會倉庫", "? 取消" ) ) {
	case 1:
		.@storage_num = 1;
		break;
	case 2:
		.@storage_num = 2;
		break;
	case 3:
		.@storage_num = 3;
		break;
	case 4:
		// copied from npc/re/merchants/guild_warehouse.txt
		.@guildid = getcharid( 2 );
		if( .@guildid == 0 ){
			mes "[訓練場倉庫士兵]";
			mes "公會倉庫僅限公會成員使用。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[訓練場倉庫士兵]";
			mes "看來你的公會還無法使用公會倉庫，或者你沒有存取倉庫的權限。";
			mes "請確認公會技能和存取權限後再來。";
			close;
		}

		if( Zeny < 1000 ){
			mes "[訓練場倉庫士兵]";
			mes "恐怕你的錢不夠付手續費。";
			mes "手續費是 1000 Zeny。";
			close;
		}

		mes "[訓練場倉庫士兵]";
		mes "那我就為你開啟公會倉庫。祝你有段難忘的時光！";
		close2;
		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[訓練場倉庫士兵]";
			mes "恐怕現在有其他人正在使用倉庫。";
			mes "請稍後再試。";
			close;
		}
	case 5:
		end;
	}
	// TODO (custom text)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[訓練場倉庫士兵]";
		mes "很抱歉，但你";
		mes "需要初心者的";
		mes "基本技能等級 6 才能";
		mes "使用倉庫服務。";
		close2;
		cutin "",255;
		end;
	}

	if (Zeny < 1000) {
		mes "[訓練場倉庫士兵]";
		mes "^4d4dff在訓練場使用倉庫需要 1000 Zeny。^000000";
		close;
	}
	Zeny -= 1000;
	openstorage; // TODO: open correct storage 1-3
	end;
}

// Cleaner
tra_fild01,142,113,6	script	清潔機器人#fild01	EP17_2_OMEGA_CLEANER,{
OnStart:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	if (.is_moving[.@num])
		end;
	.is_moving[.@num] = true;
	stopnpctimer;

	setarray .@emotions[0], ET_SWEAT, ET_PROFUSELY_SWEAT, ET_HUNGRY, ET_OHNO;
	.@r = rand( getarraysize(.@emotions) );
	emotion .@emotions[.@r];
	setarray .@speed[0], 300, 400, 500, 600;
	.@r = rand( getarraysize(.@speed) );
	npcspeed .@speed[.@r];

	getmapxy .@map$,.@x,.@y, BL_NPC;
	switch( .@num ) {
	case 1:
		if (.@x <= 126)
			.sens[.@num] = 1;
		else if (.@x >= 198)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 2:
		if (.@x <= 21)
			.sens[.@num] = 1;
		else if (.@x >= 85)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 3:
		if (.@y <= 33)
			.sens[.@num] = 1;
		else if (.@y >= 89)
			.sens[.@num] = -1;
		.@y += 8 * .sens[.@num];
		break;
	}
	unitwalk getnpcid(0),.@x,.@y, strnpcinfo(0) + "::OnMoveEnd";
	end;
OnMoveEnd:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.is_moving[.@num] = false;
	initnpctimer;
	end;
OnTimer30000:
	donpcevent strnpcinfo(0) + "::OnStart";
	end;

OnInit:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.sens[.@num] = 1;
	initnpctimer;
	end;
}
