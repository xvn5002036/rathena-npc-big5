function	script	F_AvenBoard	{
	.@mo_board = select( "依樓層查詢怪物",
					"怪物資訊檢索系統",
					( countitem(909) == 30000 ? "請求" : "" ),
					"取消" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[怪物資訊檢索系統]";
			mes "這是怪物資訊檢索系統。可以查詢出現在此迷宮的怪物資訊。搜尋時請輸入包含空格的完整怪物名稱。";
			next;
			input .@inputstr$;
			setarray .@element$[0],"無","水","地","火","風","毒","聖","暗","念","不死";
			setarray .@size$[0],"中","小","大";
			setarray .@race$[0],"無形","不死","動物","植物","昆蟲","魚貝","惡魔","人型","天使","龍";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[怪物資訊檢索系統]";
			mes "搜尋中";
			next;
			mes "[怪物資訊檢索系統]";
			mes "搜尋已完成。";
			mes "現在顯示搜尋結果。";
			next;
			mes "[怪物資訊檢索系統]";
			if (.@i >= .@size) {
				mes "沒有該怪物的可用資訊。";
				mes "請檢查您的怪物名稱。";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- 名稱 : "+ getarg(.@i+1);
				mes "- 屬性 : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- 體型 : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- 種族 : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[怪物資訊檢索系統]";
			mes "您想搜尋其他怪物資訊嗎？";
			next;
			if (select( "搜尋","取消" ) == 1)
				continue;
			mes "[怪物資訊檢索系統]";
			mes "怪物搜尋系統現在將關閉。";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "要求的資訊將出現 - 如果出現也沒有問題";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "操作錯誤。";
		close;
	}
}

	mes "如需更多關於怪物的資訊";
	mes "請務必使用";

	next;
	return;
}
