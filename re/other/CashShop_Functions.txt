function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[不知名處傳來的聲音]";
		mes "只有擁有特定數量重置石的人，才能使用 VIP 專屬服務。";
		next;
		if (countitem(6320) < 1) {
			mes "[不知名處傳來的聲音]";
			mes "為了轉職為三轉職業，你必須攜帶 1 個 ^0000CC重置石^000000，或擁有 VIP 帳號。";
			//mes "Please visit our iShop to purchase the ^0000CCReset Stone^000000.";
			close;
		} else {
			mes "[不知名處傳來的聲音]";
			mes "你願意支付 1 個 ^0000CC重置石^000000 來進行這項任務嗎？";
			next;
			switch(select("是:否")) {
			case 1:
				mes "[不知名處傳來的聲音]";
				mes "祝你好運，冒險者。";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[不知名處傳來的聲音]";
				mes "那好吧。";
				close;
			}
		}
	}
	return;
}
