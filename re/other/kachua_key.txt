//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	卡翠亞的秘密箱子#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "一個沒人知道裡面有什麼的秘密箱子。";
	mes "^4d4dff你可以透過消耗 " + mesitemlink( .@key$ ) + " 來打開這個箱子。^000000";
	next;
	switch(select("^4d4dff開啟箱子 1 次 (消耗 1 個" + .@keyname$ + ")^000000","^4d4dff開啟箱子 10 次 (消耗 10 個" + .@keyname$ + ")^000000")) {
		case 1:
			mes "^FF0000[注意]^000000";
			mes "^FF0000使用 1 個" + .@keyname$ + "交換一項隨機道具。^000000";
			mes "^FF0000上述交換的道具無法退回，也無法換回" + .@keyname$ + "。^000000";
			next;
			if(select("繼續。","結束對話。") == 2) {
				mes "你決定不打開箱子。";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "你的 " + mesitemlink( .@key$ ) + " 數量不足。";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dff請確認你的物品欄有足夠的空間。^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " 被打開了！";
				mes "你運氣好嗎？";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[注意]^000000";
			mes "^FF0000使用 10 個" + .@keyname$ + "交換 10 項隨機道具。^000000";
			mes "^FF0000上述交換的道具無法退回，也無法換回" + .@keyname$ + "。^000000";
			next;
			if(select("繼續。","結束對話。") == 2) {
				mes "你決定不打開箱子。";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "你的 " + mesitemlink( .@key$ ) + " 數量不足。";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dff請確認你的物品欄有足夠的空間。^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " 已經打開了 " + .@i + " 次。正在打開下一個。",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " 已經打開了 10 次！";
				mes "你運氣好嗎？";
				close;
			}
	}
}
