que_ng,21,72,6	script	防具製作大師	4_M_RASWORD,{

	mes "絕對不要讓自己倒下！";

	mes "為此，你必須更好地保護自己。";
	mes "當你忽視『自己』的防護時，你是無法保護『他人』的。";
	next;

		mes "謹慎也是戰鬥中一項重要的美德。";

	}
	switch(select("野狼護臂[1]:弦月護臂[1]:忍者鱗片鎧甲[1]:潛影鎧甲[1]:退出。")) {
	case 1:
		mes "[喬]";
		mes "野狼護臂……一件充滿野性的防具。";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[喬]";
			mes "你感受到狼的本性了嗎？";
			mes "既狂野又迷人……";
			close;
		}
		break;
	case 2:
		mes "[喬]";
		mes "弦月護臂……一件兼具美麗與力量的物品……兩者毫不遜色。";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[喬]";
			mes "如此的美麗……";
			mes "彷彿全身都洋溢著魔力。";

		}
		break;
	case 3:
		mes "[喬]";
		mes "忍者鱗片鎧甲……據說曾經有一位傳奇的忍者穿過它。";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[喬]";
			mes "這真是一件傳奇的鎧甲……";
			mes "即使我不相信傳說……";
			close;
		}
		break;
	case 4:
		mes "[喬]";
		mes "潛影鎧甲……它是為了保護他人而製作的。";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness

	mes "你沒有足夠的材料。";
	mes "再檢查看看……";
	close;

L_Select:
	.@string1$ = "確認材料。";
	.@string2$ = "確認能力。";
	while(1) {
		switch(select(.@string1$,.@string2$,"合成。","退出。")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[喬]";
			mes "看來你需要更多時間來決定。";
			close;
		}
	}

L_Ingredients:
	mes "[喬]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "製作^ff0000野狼護臂，你需要 1 個狐狸護臂和 10 個狼血^000000。";
		mes "你能感覺到嗎？那狼血的悸動……";
		break;
	case 2173:	//Crescent_Armguard
		mes "製作^ff0000弦月護臂，你需要 1 個狐狸護臂和 100 個水晶碎片^000000。";
		mes "它將會被刻上如同水晶般美麗清澈的新月。";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "製作^ff0000忍者鱗片鎧甲，你需要 30 個火龍鱗片、30 個冰鱗片和 30 個暗紅色鱗片碎片^000000。";
		mes "你能感覺到它傳奇般的力量。";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "製作^ff0000潛影鎧甲，你需要 1 件有洞的特殊忍者服和 10 個黑暗碎片^000000。";
		mes "它以黑暗抹去了自身的存在……";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "物理近戰攻擊時，有機率在 5 秒內 ATK+100，FLEE-50。";
		mes "盾牌類型，防禦 45，插槽 1";
		mes "限制等級 100，僅限忍者系列";
		break;
	case 2173:	//Crescent_Armguard
		mes "精練每 +1，技能後延遲減少 2%。";
		mes "盾牌類型，防禦 70，插槽 1";
		mes "限制等級 100，僅限忍者系列";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%，MSP-30%";
		mes "鎧甲類型，防禦 90，插槽 1";
		mes "限制等級 100，僅限忍者系列";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "遭受物理近戰攻擊時，有機率發動幻影-幻象 1 級。";
		mes "鎧甲類型，防禦 60，插槽 1";
		mes "限制等級 100，僅限忍者系列";
		break;
	}
	next;
	return;

L_Combine:
	mes "[喬]";
	mes "請確認^ff0000你手邊有正確的材料和裝備^000000。";
	mes "事後再哭是沒有用的。";
	next;
	switch(select("合成。:退出。")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- 稍等！！ -";
			mes "- 你無法獲得物品 -";
			mes "- 因為你攜帶了太多東西。 -";
			mes "- 請再試一次 -";
			mes "- 在減輕你的負擔之後。 -";
			close;
		}
		return;
	case 2:
		mes "[喬]";
		mes "看來你需要更多時間來決定。";
		close;
	}
	end;
}
