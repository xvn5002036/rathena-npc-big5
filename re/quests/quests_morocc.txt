		close;
	mes "[可疑的貓]";
	mes "門都沒有！我花了好長時間才找到那條路的！！";
	next;
	mes "[可疑的貓]";
	mes "嗯……但如果你堅持的話，只要給我 5000 Zeny。我就帶你去。";
	mes "怎樣?？別用那種眼神看我！你以為貓在米德加爾特生存有多難嗎！";
	next;
	mes "[可疑的貓]";
	mes "^FF0000不過話說回來，像你這樣的弱者在那裡是活不下去的。如果你夠幸運的話，或許能活個一秒鐘？^000000";
	mes "好吧，這不關我的事。但我已經警告過你了！";
	next;
	switch(select("支付 5,000 Zeny 並出發。:算了吧。")) {
	case 1:
		if (Zeny < 5000) {
			mes "[可疑的貓]";
			mes "你以為你要去哪裡！你這個窮光蛋！";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[可疑的貓]";
		mes "呸，這舊錢是怎麼回事……你就沒有什麼閃閃發亮的東西嗎？！";
		mes "啊，跟著我來就是了！";
		close2;
		warp "moc_prydn1",93,96;
		end;
	case 2:
		mes "[可疑的貓]";
		mes "天啊，你連 5000 Zeny 都沒有，對吧？";
		close;
	}
}
