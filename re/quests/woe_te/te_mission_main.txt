	end;
}

						mes "就像"+ .@npc_name$ +"說的，你翻找了前面的箱子，然後...";
						delitem .@item_quest, countitem(.@item_quest);
						erasequest .@main_quest;
						getitem .@reward[0],.@reward[1];
						getexp .@exp[0],.@exp[1];
						callsub( S_Rentbox, .@num_tool_box );
					}
					callsub( S_Cancelquest, .@main_quest, .@npc_name$, "這個任務是在"+ .@fortress$ +"的城池中^4d4dff收集"+ .@string$[2] +"^000000。" );
				}
				mes "時間到了。";
				mes "很遺憾，任務失敗！";
				mes "下次再努力吧！";
				erasequest .@main_quest;
				close;
			}
			if (.@time_check != 1 && .@siege_check != 1) {
				mes "你目前沒有進行這個任務。";
				mes "你能再確認一下嗎？";
				close;
			}
			if (.@mission02 == 1 || .@mission03 == 1) {
				mes "你已經在進行相同內容的任務了，";
				mes "儘管等級不同";
				mes "除非你想取消，否則我希望你先處理好手邊的任務。";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "這樣才合理，不是嗎？";
				mes "你不覺得嗎？";
				close;
			}
			mes "嗯，你可能不相信，但這個任務跟一宗竊案有關。";
			mes "他們告訴我，"+ .@fortress$ +"的城池裡遺失了一堆"+ .@string$[2] +"。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "小偷可能是趁著攻城戰的時間下手的。";
			mes "這是一個委託，要求找回^4d4dff "+ .@num_items +" 個以上的"+ .@string$[2] +"^000000。";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "既然TE區域的氣氛比較輕鬆，";
		mes "只有打攻城戰也不太好玩，對吧？";
		mes "所以我準備了^4d4dff一些你在攻城戰期間可以做的工作^000000?";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你可以在^4d4dff攻城戰TE開始前1小時直到攻城戰結束為止^000000接受任務，";
		mes "而且每個任務都有^4d4dff 4小時的時間限制^000000來完成。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "簡單來說，你在攻城戰期間向我接受任務，然後";
		mes "^4d4dff在時間限制內回來向我報告^000000。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "試試看，當作是找點樂子吧。";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "我明白了。";
		mes "如果你改變心意，隨時可以回來。";
		mes "不過我可能不一定有空就是了。呵呵呵。";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "看來你還沒有完成任務？";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你難道是想取消嗎？";
		next;
		switch( select( "沒有。","是的。" ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "祝你好運！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "如你所願，這個任務已經取消了。";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "嗯，你還有一些時間，但現在還不是攻城戰時間";
		mes "我覺得你不應該繼續下去。";
		mes "你應該取消這個任務。我會說任務失敗了。";
		next;
		switch( select( "我想繼續。","是的，我要取消。" ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "我不認為這值得。好吧，祝你好運！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "如你所願，這個任務已經取消了。";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "接受。","不接受。" ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dff時間限制是4小時。^000000";
		mes "在那段時間內完成任務並向我報告。";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "那就試試別的任務吧。";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "你什麼都沒找到。";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "你找到了「"+ getitemname(.@item_id[ .@miss_item01 ]) +"」！";
		}
	}
	close;
S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dff確認結果^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
