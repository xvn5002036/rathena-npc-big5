function	script	Magic_Book_Essence	{
	if (MaxWeight - Weight < 1000) {
		mes "請減輕你背包的重量。";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "你使用了 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 創造出 ^999900"+ getarg(3) +"精華^000000。";
			next;
			mes "^008800除非發生問題，否則"+getarg(3)+"精華將永久存在。^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888你未能使用 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 提煉出精華。^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "你已經在這個區域創造出"+getarg(3)+"精華了。";
		close;
	}
	mes "已經有人使用了這個區域。";
	close;
}
