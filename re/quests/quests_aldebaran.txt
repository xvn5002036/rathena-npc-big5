c_tower1,232,222,5	script	貝爾傑明 巴頓#ct	968,{
	mes "[貝爾傑明 巴頓]";
	mes "這裡的魔物擁有控制時間的能力。";
	if( BaseLevel < 150 ){
		mes "等你達到 ^0000FF150 級^000000 之後再來吧。";
		close;
	}
	mes "時間、記憶、環境... 在這裡都會被扭曲。";
	next;
	mes "[貝爾傑明 巴頓]";
	mes "我求你別再問了。這可能會讓你有生命危險。";
	next;
	.@menu = select("不問了:前往扭曲的鐘塔 2 樓:前往扭曲的鐘塔 3 樓");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[貝爾傑明 巴頓]";
			mes "請慎重選擇。";
			next;
			mes "[貝爾傑明 巴頓]";
			mes "我需要 ^0000FF10,000z^000000 才能讓你進去。那裡可是個非常危險的地方。";
			next;
			switch( select( "退出:支付 10,000z:你為什麼需要錢？" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[貝爾傑明 巴頓]";
						mes "真的嗎？你沒錢還叫我帶你去？";
						close;
					}
					Zeny -= 10000;
					mes "[貝爾傑明 巴頓]";
					mes "我不確定你在那裡面要怎麼計算時間，但請盡可能小心。";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[貝爾傑明 巴頓]";
					mes "我整天待在這裡是為了冒險者們的安全。我總是會警告他們這裡的危險。這就是為什麼我需要一些報酬。";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[貝爾傑明 巴頓]";
	mes "呼... 你做了一個明智的選擇。這個地方非常殘酷，所以你最好回到安全的地方。";
	close;
}

	end;
