glast_01,195,273,3	script	福金的追隨者#1	2_F_MAGICMASTER,{
	mes "[福金的追隨者]";
	mes "我雖然無法像福金大人那樣在時間與次元之間穿梭，但我可以模仿一下。";
	next;
	mes "[福金的追隨者]";
	mes "是什麼原因讓這座克雷斯特漢姆古城充滿了如此強大的黑暗力量呢？";
	next;
	if( select( "我不在乎:那很有趣，請告訴我更多" ) == 2 ){
		mes "[福金的追隨者]";
		mes "多虧了福金大人，我找到了進入這座被強大黑暗力量充斥的古老克雷斯特漢姆古城的方法。";
		next;
		mes "[福金的追隨者]";
		mes "你覺得如何？你想去那裡探險嗎？";
		next;
		if( select( "我不想探險:我真的很想去探險！" ) == 2 ){
			mes "[福金的追隨者]";
			mes "沒問題，但這可不是免費的服務！我必須得到一些回報。呵呵...";
			next;
			mes "[福金的追隨者]";
			mes "我可以送你去兩個地方。你想讓我送你去哪裡？";
			next;
			switch( select( "取消:修道院墓地 (Lv100~130) - 5000z:2樓迴廊 (Lv140以上) - 10000z" ) ){
				case 1:
					mes "[福金的追隨者]";
					mes "好吧，如果你改變主意隨時可以回來找我。我想你會有很多話想跟我說...";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[福金的追隨者]";
				mes "什麼？你以為這項服務是免費的嗎？";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[福金的追隨者]";
	mes "好吧，如果你不感興趣，那就別管了。";
	close;
}
