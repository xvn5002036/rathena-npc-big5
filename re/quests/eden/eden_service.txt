-	script	物品櫃#00::pggc	-1,{
	mes "這是一個物品儲藏櫃。";
	mes "上面貼著一張紙條寫著：";
	next;
	mes "+ 儲藏櫃僅限 +";
	mes "+ 伊甸園成員使用 +";
	mes "如果您想使用";
	mes "此儲藏櫃，請確認";
	mes "以下事項：";
	next;
	mes "1.您是伊甸園的";
	mes "   成員嗎？";
	mes "2.您是否具備";
	mes "   足夠的基本技能？";
	mes "3.儲藏櫃使用費為";
	mes "  ^4d4dff500 zeny^000000！";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "使用儲藏櫃需要投入 Zeny。";
		mes "費用：500 Zeny ";
		mes "您想要使用嗎？";
		next;
		switch(select("使用儲藏櫃。:取消。")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "這個儲藏櫃對我沒有反應。";
						mes "或許我還沒有資格使用它。";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "我的 Zeny 不夠。";
				mes "我至少需要 500 Zeny 才能使用這個儲藏櫃。";
				close;
			case 2:
				mes "我下次再使用吧。";
				close;
		}
	}
	mes "我需要伊甸園標誌才能使用這個儲藏櫃。";
	close;
}
moc_para01,173,120,0	duplicate(pggc)	物品櫃#01	111
