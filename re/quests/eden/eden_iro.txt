function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "不過，我全部的力量只對 VIP 開放。";
		mes "但我還是可以提供傳送點將你送到"+getarg(1)+"。";
		next;
		switch(select(getarg(0)+" -> 1800z:取消")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "很抱歉，但是你沒有";
				mes "足夠的 Zeny 來支付傳送點";
				mes "服務。傳送到";
				mes ""+getarg(1)+" 的費用是 1,800 Zeny。";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}

//callsub L_Exchange,<Insert Item>,<Amount>,<Return Item>,<Amount>
L_Exchange:
	mes "[兌換機]";
	mes "你想要多少個 "+getitemname(getarg(2))+"？";
	mes "我最多可以給你 60 個。";
	mes "輸入 0 來取消。";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[兌換機]";
		mes "已取消。";
		close;
	} else if (.@amount < 0 || .@amount > 60) {
		mes "[兌換機]";
		mes "我說過最多只能 60 個。";
		close;
	}
	mes "[兌換機]";
	mes "總共是 "+.@amount+" 個 "+getitemname(getarg(2))+"。";
	mes "這樣正確嗎？";
	next;
	if(select("是的，給我吧！:不，等等先不要。") == 2)
		close;
	set .@cost, getarg(1) * .@amount;
	set .@total_amount, getarg(3) * .@amount;
	if (countitem(getarg(0)) < .@cost) {
		mes "[兌換機]";
		mes "你沒有足夠的 "+getitemname(getarg(0))+" 來進行這項交易。";
		close;
	}
	delitem getarg(0),.@cost;
	getitem getarg(2),.@total_amount;
	close;
}
