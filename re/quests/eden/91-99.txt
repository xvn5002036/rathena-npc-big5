L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[91-99 任務佈告欄]";
		mes "你需要等待 3 小時才能再次接受這個任務。";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[91-99 任務佈告欄]";
		mes "你已經狩獵完 30 隻"+getarg(1)+getarg(5,"")+"了嗎？";
		close;
	} else if (.@hunting == 2) {
		mes "[91-99 任務佈告欄]";
		mes "你已經完成了狩獵。";
		mes "請收下這份獎勵作為報酬。";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;

		// Job Experience calculation.
		if (JobLevel > 50 && JobLevel < 56) set .@jexp,67000;
		else if (JobLevel > 55 && JobLevel < 61) set .@jexp,94000;
		else if (JobLevel > 60 && JobLevel < 66) set .@jexp,141000;
		else if (JobLevel > 65 && JobLevel < 69) set .@jexp,212000;
		else if (JobLevel == 69) set .@jexp,311000;

		getexp getarg(2),.@jexp;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1
		erasequest .@quest1;
		mes "[91-99 任務佈告欄]";
		mes "你已重新接受了狩獵 "+getarg(1)+" 的任務。";
		close;
	}*/ else {
		mes "[91-99 任務佈告欄]";
		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[91-99 任務佈告欄]";
		mes "在那裡狩獵 30 隻"+getarg(1)+"，並將你的成果登記在這個佈告欄上。";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("我要去狩獵。:不用了，謝謝。") == 1) {
			mes "[91-99 任務佈告欄]";
			mes "當你完成任務後，在這個佈告欄上登記並領取你的獎勵。";
			if (.@playtime > -1) erasequest .@quest2;
			setquest .@quest1;
		}
		close;
	}
