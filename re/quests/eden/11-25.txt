moc_para01,36,38,3	script	任務 [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- 你必須擁有 -";
		mes "- ^4d4dff'伊甸園標誌'^000000 -";
		mes "- 才能接取這些任務。 -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes " - 你必須先領取你的 - ";
			mes " - 獎勵，才能開始 - ";
			mes " - 新的任務。 - ";
			mes " ";
			mes " - 管理員 : 斯派克 - ";
			close;
		}
		else if (.@j >= 0) {
			mes " - 你一次只能接取 - ";
			mes " - 一個任務。 - ";
			mes " ";
			mes " - 管理員 : 斯派克 - ";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes " - 目前沒有適合 - ";
		mes " - 你的任務。 - ";
		close;
	}
	if (BaseLevel > 25) {
		mes " - 這些任務對你的實力 - ";
		mes " - 來說太簡單了， - ";
		mes " - 請尋找更符合你 - ";
		mes " - 等級的任務。 - ";
		close;
	}
	mes " - 初學者 -";
	mes " - 任務公佈欄。 -";
	mes " ";
	mes " - 管理員 : 斯派克 -";
	next;
	switch(select("狩獵黃蜂:狩獵禿鷹:蝗蟲的後腿:狩獵蚯蚓:狩獵蘑菇:驅除害蟲:狩獵摩卡:狩獵吸血蝙蝠:收集羽毛:收集毒蘑菇孢子:取消")) {
		case 1: callsub L_Quest,11114,"狩獵黃蜂"," - 狩獵10隻^4d4dff'黃蜂'^000000 - "," - (普隆德拉西北方)。 - ";
		case 2: callsub L_Quest,11115,"狩獵禿鷹"," - 狩獵20隻^4d4dff'禿鷹'^000000 - "," - (夢羅克南方)。 - ";
		case 3: callsub L_Quest,11116,"蝗蟲的後腿"," - 狩獵10隻^4d4dff'蝗蟲'^000000 - "," - (普隆德拉西南方)，並 - "," - 帶來10個^4d4dff蝗蟲的後腿^000000。 - ";
		case 4: callsub L_Quest,11117,"狩獵蚯蚓"," - 狩獵20隻^4d4dff'蚯蚓'^000000 - "," - (斐揚東南方)。- ";
		case 5: callsub L_Quest,11118,"狩獵蘑菇"," - 狩獵30隻^4d4dff'蘑菇'^000000 - "," - (斐揚周遭)。 - ";
		case 6: callsub L_Quest,11119,"驅除害蟲"," - 狩獵20個^4d4dff盜蟲卵^000000 - "," - (普隆德拉地下水道)， - "," - 並帶來10個^4d4dff蛹^000000。 - ";
		case 7: callsub L_Quest,11120,"狩獵摩卡"," - 狩獵20隻^4d4dff'摩卡'^000000 - "," - (普隆德拉南方)。 - ";
		case 8: callsub L_Quest,11121,"狩獵吸血蝙蝠"," - 狩獵20隻^4d4dff吸血蝙蝠^000000 - "," - (地下水道 - "," - 普隆德拉西方)。 - ";
		case 9: callsub L_Quest,11122,"收集羽毛"," - 狩獵30隻^4d4dff'大嘴鳥'^000000 - "," - (普隆德拉南方)。 - ";
		case 10: callsub L_Quest,11123,"收集毒蘑菇孢子"," - 狩獵30隻^4d4dff毒蘑菇^000000 - "," - (吉芬塔內)，並 - "," - 帶來5個^4d4dff毒蘑菇孢子^000000。 - ";
		case 11:
			mes " - 停止閱讀公佈欄。 - ";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes " - 任務 ^4d4dff"+ getarg(1) +"^000000 - ";
		mes " - 今天的進度已經完成了。 - ";
		mes " ";
		mes " - 管理員 : 斯派克 - ";
		close;
	}
	mes " - 任務 '"+ getarg(1) +"' - ";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes " ";
	mes " - 管理員 : 斯派克 - ";
	next;
	if (select("我要接取這個任務。:查看其他任務。") == 2) {
		mes " - 讓我們尋找 -";
		mes " - 其他任務吧。 - ";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	斯派克	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"狩獵黃蜂",900;
				case 11115: callsub L_Quest,.@i,"狩獵禿鷹",1800;
				case 11116: callsub L_Quest,.@i,"蝗蟲的後腿",1400,940,10;
				case 11117: callsub L_Quest,.@i,"狩獵蚯蚓",2600;
				case 11118: callsub L_Quest,.@i,"狩獵蘑菇",3900;
				case 11119: callsub L_Quest,.@i,"驅除害蟲",2900,915,10;
				case 11120: callsub L_Quest,.@i,"狩獵摩卡",3200;
				case 11121: callsub L_Quest,.@i,"狩獵吸血蝙蝠",3200;
				case 11122: callsub L_Quest,.@i,"收集羽毛",5400;
				case 11123: callsub L_Quest,.@i,"收集毒蘑菇孢子",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[斯派克]";
			mes "你還沒完成任務喔。";
			mes "繼續努力，你就快完成了。";
			close;
		}
	}
	mes "[斯派克]";
	mes "你在找工作嗎？";
	mes "我們總是有很多工作要做。";
	next;
	mes "[斯派克]";
	mes "為什麼不四處看看，找些有趣的任務呢？";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[斯派克]";
				mes "你的"+ getitemname(.@item_req) +"數量不足。";
				mes "你需要帶來 "+ .@req_amount +" 個 "+ getitemname(.@item_req) +"。";
				close;
			}
		}
		mes "[斯派克]";
		mes "嗯... 讓我看看?";
		mes "你已經完成了「"+ getarg(1) +"」任務。";
		next;
		mes "[斯派克]";
		mes "做得好?！！";
		mes "我們的委託人對你的表現非常滿意。他非常感謝你的幫忙。";
		mes "請下次再幫我們的忙。謝謝！";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}
