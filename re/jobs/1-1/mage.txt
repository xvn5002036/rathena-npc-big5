	if (Upper == 1) {
		if (ADVJOB == Job_High_Wizard || ADVJOB == Job_Professor) {
			if (Class == Job_Novice_High) {
				mes "[魔法師公會關係者]";
				mes "哇，好久不見！但你不是應該死了嗎？";
				next;
				mes "[魔法師公會關係者]";
				mes "啊，你一定已經重生了。太好了，很高興你回來。";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[魔法師公會關係者]";
					mes "很抱歉，但我認為你還沒準備好學習魔法。為什麼不先去完成基本技能的學習呢？";
					next;
					mes "[魔法師公會關係者]";
					mes "慢慢來。你學得越多，再次學習魔法的準備就越充分。";
					close;
				}
				mes "[魔法師公會關係者]";
				mes "嗯，既然你已經通過過一次魔法師的測驗，我不會質疑你的資格。你想立刻恢復你的魔法技能，對吧？";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Mage_High;
				skill "MG_ENERGYCOAT",1,SKILL_PERM;
				mes "[魔法師公會關係者]";
				mes "哇，不知道為什麼，你看起來比以前更棒了。總之，我相信你這次也能成為一名出色的魔法師。";
				close;
			}else{
				mes "[魔法師公會關係者]";
				mes "還有什麼我可以幫你的嗎？如果沒有的話，何不去測試一下你的技能呢？這個世界正在等著你?！";
				close;
			}
		}else{
			mes "[魔法師公會關係者]";
			mes "怎麼，你對魔法師公會感興趣嗎？我本來不想告訴你，但你不屬於這裡。";
			next;
			mes "[魔法師公會關係者]";
			mes "我不知道你為什麼還站在我面前，但我看得出來你注定成不了魔法師。";
			close;
		}
	}
	mes "[魔法師公會關係者]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Mage) {
			mes "嘿，你還沒發現嗎？你已經是個魔法師了，傻瓜！";
			mes "總有一天，當你能用意識生火時，你就會意識到你體內蘊藏的力量！";
		} else
			mes "嘿?拜託。別鬧了。你不能成為魔法師，因為你已經有其他職業了。";
		close;
	}else{
		mes "嘿？";
		next;
		switch(select("我想成為魔法師。:請問成為魔法師的條件是什麼？:沒事，謝謝。")) {
		case 1:
			mes "[魔法師公會關係者]";
			mes "想成為魔法師？呃...";
			next;
			mes "[魔法師公會關係者]";
			if (Sex == SEX_MALE) {
				mes "嘿，看看你！你還挺可愛的?！不過不是我的菜...";
			}else{
				mes "喔喔，妳真是個辣妹?！";
				mes "我喜歡像妳這樣的女孩?";
			}
			mes "對了，你說你想成為魔法師？";
			next;
			if(select("我想成為魔法師。:沒事，謝謝。") == 2) {
				mes "[魔法師公會關係者]";
				mes "什麼??？！才剛跟我說想成為魔法師，你就改變主意了？！果斷一點好嗎！";
				close;
			}
			mes "[魔法師公會關係者]";
			if (!callfunc("F_CanChangeJob")) {
				mes "喔，天啊，你的基本技能等級還不夠成為魔法師。";
				mes "回去把你的基本技能練高一點吧。";
				close;
			}
			mes "好吧.. 簽一下申請表.... 哇?你的字寫得真漂亮！啊，好了。這樣就夠了。";
			mes "你是... "+strcharinfo(0)+"。";
			next;
			mes "[魔法師公會關係者]";
			mes "嗯，我看得出你用自己的方式付出了很多努力。雖然看起來有點笨拙，不過我想沒關係的！";
			mes "很好！永遠堅持基本功才是最棒的！我馬上為你進行轉職。";
			next;
			mes "[魔法師公會關係者]";
			mes "哈..！你現在是一名魔法師了，我們的一份子！";
			mes "歡迎來到魔法師公會，我們的新朋友！";
			next;
			mes "[魔法師公會關係者]";
			mes "『歡迎來到魔法師公會?』";
			mes "恭喜你成為魔法師公會的一員！加油！";
			callfunc "Job_Change",Job_Mage;
			getitem 1639,1;	// N_Rod
			close;
		case 2:
			mes "[魔法師公會關係者]";
			mes "想成為魔法師，是嗎？";
			mes "我很樂意為像你這麼可愛的人說明條件！";
			next;
			mes "[魔法師公會關係者]";
			mes "首先，你必須達到初心者職業等級 10 並且學會所有的基本技能。";
			next;
			mes "[魔法師公會關係者]";
			mes "過去，有一項複雜的藥水製作測驗。因為那樣，我們的申請者慢慢流失了。";
			mes "所以，我們決定接受所有符合基本條件的申請者。";
			next;
			mes "[魔法師公會關係者]";
			mes "別猶豫了。直接成為魔法師吧！";
			close;
		case 3:
			mes "[魔法師公會關係者]";
			mes "沒事...？";
			close;
		}
	}
}
