gonryun,163,60,4	script	崑崙嚮導#01gonryun	780,{
	mes "[何淵哲]";
	mes "你好！";
	mes "歡迎來到^8B4513崑崙^000000?";
	mes "到處走走並體驗一下";
	mes "我們這座令人驚嘆的城市其";
	mes "悠久的歷史與傳統吧。";
	next;
	mes "[何淵哲]";
	mes "我負責協助你解答";
	mes "任何你可能有的疑問。";
	mes "請隨時向我提問。";
	F_Navi("[何淵哲]");
	while (1) {
		next;
		switch(select("[ 卡普拉服務人員 ]:[ 村長家 ]:[ 道具商人 ]:[ 武器商人 ]:[ 防具商人 ]:[ 釀酒師 ]:[ 崑崙使節 ]:清除小地圖上的標記:取消")) {
		case 1:
			callsub L_Mark, "^0000FF卡普拉服務人員^000000";
			viewpoint 1,159,122,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("村長家","gonryun,110,131","^006400");
			viewpoint 1,110,131,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("道具商人","gonryun,147,84","^D8BFD8");
			viewpoint 1,147,84,2,0xD8BFD8;
			break;
		case 4:
			callsub L_Mark, F_Navi("武器商人","gonryun,174,101","^FF1493");
			viewpoint 1,174,101,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("防具商人","gonryun,173,84","^8B4513");
			viewpoint 1,173,84,4,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("釀酒師","gonryun,213,115","^9400D3");
			viewpoint 1,213,115,5,0x9400D3;
			break;
		case 7:
			callsub L_Mark, F_Navi("崑崙使節","gonryun,161,11","^00BFFF");
			viewpoint 1,161,11,6,0x00BFFF;
			break;
		case 8:
			mes "[何淵哲]";
			mes "我已經把小地圖上的標記都清除了。";
			mes "還有什麼我可以幫忙的嗎？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			break;
		case 9:
			mes "[何淵哲]";
			mes "祝你旅途愉快！";
			close;
		}
	}
	end;
L_Mark:
	mes "[何淵哲]";
	mes getarg(0)+" 的位置";
	mes "已經在小地圖上標記了。";
	mes "請確認你的小地圖。";
	mes "還有什麼我可以幫忙的嗎？";
	return;
}
