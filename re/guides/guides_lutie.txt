xmas,140,137,3	script	露蒂嚮導#xmas	717,{
	mes "[露蒂嚮導]";
	mes "歡迎來到^8B4513露蒂^000000，";
	mes "這座白雪之村。";
	mes "你需要幫忙指引村莊的方向嗎？";
	F_Navi("[露蒂嚮導]");
	while (1) {
		next;
		switch(select("[ 禮品店 ]:[ 教堂 ]:[ 武器防具店 ]:[ 道具店 ]:[ 聖誕老人的家 ]:[ 玩具工廠 ]:清除小地圖上的標記:取消")) {
		case 1:
			callsub L_Mark, F_Navi("禮品店","xmas,172,131","^008080");
			viewpoint 1,172,131,2,0x008080;
			break;
		case 2:
			callsub L_Mark, F_Navi("教堂","xmas,104,287","^006400");
			viewpoint 1,104,287,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("武器防具店","xmas,171,158","^FF1493");
			viewpoint 1,171,158,3,0xFF1493;
			break;
		case 4:
			callsub L_Mark, F_Navi("道具店","xmas,122,131","^8B4513");
			viewpoint 1,122,131,4,0x8B4513;
			break;
		case 5:
			callsub L_Mark, F_Navi("聖誕老人的家","xmas,149,237","^B9062F"),1;
			viewpoint 1,149,237,5,0x9400D3;
			break;
		case 6:
			callsub L_Mark, F_Navi("玩具工廠","xmas,143,312","^9400D3");
			viewpoint 1,143,312,6,0xFF0000;
			break;
		case 7:
			mes "[露蒂嚮導]";
			mes "沒問題，我已經把小地圖上的標記都清除了。";
			mes "還有什麼我可以幫忙的嗎？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			break;
		case 8:
			mes "[露蒂嚮導]";
			mes "祝你旅途平安。";
			close;
		}
	}
	end;
L_Mark:
	mes "[露蒂嚮導]";
	mes "讓我在小地圖上標記"+((getarg(1,0))?"":"");
	mes getarg(0)+"^000000";
	mes "的位置給你。";
	mes "你還想確認其他地點嗎？";
	return;
}
