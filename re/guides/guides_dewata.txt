dewata,202,106,4	script	德瓦他嚮導#dewata01	4_M_DEWWOMAN,{
	mes "[德瓦他嚮導]";
	mes "歡迎來到^8B4513德瓦他島^000000，";
	mes "這裡有著耀眼的波浪和迷人的景色，是個放鬆的好地方。";
	mes "如果你有任何問題，請隨時問我。";
	next;
	mes "[德瓦他嚮導]";
	mes "你需要我為你指引去哪裡嗎？";
	F_Navi("[德瓦他嚮導]");
	while (1) {
		next;
		switch(select("[ 卡普拉服務人員 ]:[ 武器商人 ]:[ 防具商人 ]:[ 道具商人 ]:[ 餐廳老闆 ]:[ 喀拉喀托火山守門員 ]:[ 艾爾貝塔 水手 ]:清除小地圖上的標記:取消")) {
		case 1:
			callsub L_Mark, "^0000FF卡普拉服務人員^000000";
			viewpoint 1,202,184,0,0x000FFF;
			break;
		case 2:
			callsub L_Mark, F_Navi("武器商人","dewata,218,164","^008080");
			viewpoint 1,218,163,1,0x008080;
			break;
		case 3:
			callsub L_Mark, F_Navi("防具商人","dewata,158,182","^008080");
			viewpoint 1,158,182,2,0x008080;
			break;
		case 4:
			callsub L_Mark, F_Navi("道具商人","dewata,182,164","^FF1493");
			viewpoint 1,182,164,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("餐廳老闆","dewata,146,109","^FF1493");
			viewpoint 1,146,109,4,0xFF1493;
			break;
		case 6:
			callsub L_Mark, F_Navi("喀拉喀托火山守門員","dewata,235,56","^8A2BE2");
			viewpoint 1,235,56,5,0x8A2BE2;
			break;
		case 7:
			callsub L_Mark, F_Navi("艾爾貝塔 水手","dewata,229,49","^00BFFF");
			viewpoint 1,229,49,6,0x00BFFF;
			break;
		case 8:
			mes "[德瓦他嚮導]";
			mes "我已經把小地圖上的標記都清除了。";
			mes "還有什麼我可以幫忙的嗎？";
			viewpoint 2,202,184,0,0xFFFFFF;
			viewpoint 2,218,163,1,0xFFFFFF;
			viewpoint 2,158,182,2,0xFFFFFF;
			viewpoint 2,182,164,3,0xFFFFFF;
			viewpoint 2,232,59,4,0xFFFFFF;
			viewpoint 2,195,235,5,0xFFFFFF;
			viewpoint 2,230,52,6,0xFFFFFF;
			break;
		case 9:
			mes "[德瓦他嚮導]";
			mes "祝你旅途愉快！！";
			close;
		}
	}

L_Mark:
	mes "[德瓦他嚮導]";
	mes getarg(0)+" 的位置";
	mes "已經在小地圖上標記了。";
	mes "請確認你的小地圖。";
	mes "還有什麼我可以幫忙的嗎？";
	return;
}
