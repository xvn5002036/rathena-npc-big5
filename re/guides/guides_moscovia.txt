	mes "海上的樂園，";
	mes "美麗的波浪和溫暖的陽光。";
	mes "^8B4513莫斯科比亞^000000 正等著您。";
	mes "我負責為您提供這裡的導覽資訊。";
	mes "如果您有任何問題，請隨時告訴我。";
	F_Navi("[莫斯科比亞嚮導]");
	set .@str$,"還有什麼我可以為您效勞的嗎？";
	while (1) {
		next;
		switch(select("[ 卡普拉服務人員 ]:[ 宮殿 ]:[ 禮拜堂 ]:[ 武器店 ]:[ 防具店 ]:[ 道具店 ]:[ 旅館 ]:[ 返回艾爾貝塔 ]:清除小地圖上的標記:取消")) {

			mes "[莫斯科比亞嚮導]";
			mes "^0000FF卡普拉服務人員^000000 的位置已經";
			mes "標記在您的小地圖上了";
			mes "請查看您的小地圖。";
			mes "您還需要其他的嗎？";
			viewpoint 1,223,191,0,0x0A82FF;
			break;
		case 2:
			mes "[莫斯科比亞嚮導]";
			mes "在"+F_Navi("宮殿","moscovia,255,140","^006400")+"裡您可以找到我們的沙皇陛下與其他官員。";
			mes .@str$;
			viewpoint 1,255,140,1,0x0A82FF;
			break;
		case 3:
			mes "[莫斯科比亞嚮導]";
			mes "這就是"+F_Navi("禮拜堂","moscovia,253,179","^008080")+"所在的位置。";
			mes .@str$;
			viewpoint 1,253,179,2,0xAAFF00;
			break;
		case 4:
			mes "[莫斯科比亞嚮導]";
			mes "如果您前往"+F_Navi("武器店","moscovia,186,191","^FF1493")+"，可以從莫斯科比亞最頂尖的工匠那裡購買武器。";
			mes .@str$;
			viewpoint 1,186,191,3,0x008080;
			break;
		case 5:
			mes "[莫斯科比亞嚮導]";
			mes "您只有在莫斯科比亞的"+F_Navi("防具店","moscovia,204,169","^8B4513")+"才能找到最棒的裝備。";
			mes .@str$;
			viewpoint 1,204,169,4,0xB200FF;
			break;
		case 6:
			mes "[莫斯科比亞嚮導]";
			mes ""+F_Navi("道具店","moscovia,221,176","^FF1493")+"就位於城鎮中心的南邊。";
			mes "您可以在那裡找到旅行所需的各種物品。";
			mes .@str$;
			viewpoint 1,221,176,5,0xFF1493;
			break;
		case 7:
			mes "[莫斯科比亞嚮導]";
			mes "旅館「"+F_Navi("黏草樹","moscovia,227,206")+"」就在城鎮中心的北邊。";
			mes "如果您需要休息，那裡是最好的去處。";
			mes .@str$;
			viewpoint 1,227,206,6,0x8B4513;
			break;
		case 8:
			mes "[莫斯科比亞嚮導]";
			mes "您可以在碼頭找到";
			mes F_Navi("莫斯科比亞公關人員","moscovia,166,53")+"，如果";
			mes "您想返回艾爾貝塔的話。";
			viewpoint 1,166,53,7,0x00BFFF;
			break;
		case 9:
			mes "[莫斯科比亞嚮導]";
			mes "我會清除您小地圖上的所有標記。";
			mes .@str$;
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 10:
			mes "[莫斯科比亞嚮導]";
			mes "祝您旅途愉快?！";
			close;
		}
	}
}
