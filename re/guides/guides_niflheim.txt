niflheim,201,187,3	script	尼芙菲姆嚮導#01nif	849,{
	mes "[尼芙菲姆嚮導]";
	mes "歡迎來到.. ^8B4513尼芙菲姆^000000...";
	mes "死者的國度...";
	mes "你需要幫忙在這國度裡導覽嗎...?";
	if (F_Navi() == 0) {
		next;
		mes "[尼芙菲姆嚮導]";
		mes "當你點擊.. ^B9062F[地點名稱]^000000... 你就能獲得最先進的^B9062F導航^000000服務...！";
	}
	while (1) {
		next;
		switch(select("[ 道具店 ]:[ 酒店 ]:[ 武器店 ]:[ 魔女之城 ]:清除小地圖上的標記:取消")) {
		case 1:
			callsub L_Mark, F_Navi("道具店","niflheim,218,196","^0000FF");
			viewpoint 1,218,196,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("酒店","niflheim,189,208","^006400");
			viewpoint 1,189,208,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("武器店","niflheim,218,170","^800080");
			viewpoint 1,218,170,2,0xDA70D6;
			break;
		case 4:
			callsub L_Mark, F_Navi("魔女之城","niflheim,255,195","^FF1493");
			viewpoint 1,255,195,3,0xFF1493;
			break;
		case 5:
			mes "[尼芙菲姆嚮導]";
			mes "好的，我會清除你小地圖上的所有標記...";
			mes "還有什麼我可以為你效勞的嗎...?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			break;
		case 6:
			mes "[尼芙菲姆嚮導]";
			mes "......再見~!..";
			close;
		}
	}
	end;
L_Mark:
	mes "[尼芙菲姆嚮導]";
	mes "讓我把";
	mes getarg(0)+"^000000...";
	mes "標記在你的小地圖上。";
	mes "你還想查看其他地點嗎？";
	return;
}
