// Illusion of Luanda
//==================================================
// Todo: the count should start after the end of the cooldown?
-	script	illusion_luanda	-1,{
	end;
OnMob20270: callsub( S_Mob, 20270, 5 );	// ILL_TRI_JOINT
OnMob20271: callsub( S_Mob, 20271, 5 );
// ILL_STALACTIC_GOLEM
OnMob20272: callsub( S_Mob, 20272, 5 );	// ILL_MEGALITH
OnMob20274: callsub( S_Mob, 20274, 4 );
// ILL_STONE_SHOOTER
OnMob20275: callsub( S_Mob, 20275, 4 );	// ILL_WOOTAN_SHOOTER
OnMob20276: callsub( S_Mob, 20276, 4 );	// ILL_WOOTAN_FIGHTER
S_Mob:
	.@group = getarg(1);
	if (.@group != .group)
		end;
	if (playerattached()) {
		if (.count < 1500)
			.count++;
		if (.count == 1500 && !.boss_spawn && !.cooldown) {
			getunitdata( killedgid, .@data );
			callsub( S_Boss, (20273 + .swap), .@data[UMOB_X], .@data[UMOB_Y] );
		}
	}
	.@mob_id = getarg(0);
	sleep 5000;
	if (.@group == .group)
		monster "com_d02_i",0,0,"--ja--",.@mob_id,1,"illusion_luanda::OnMob" + .@mob_id;
	end;

S_Boss:
	killmonster "com_d02_i", "illusion_luanda::OnBossDeath";
	.boss_spawn = true;
	monster "com_d02_i",getarg(1),getarg(2),"--ja--",getarg(0),1,"illusion_luanda::OnBossDeath";
	if (.swap == 0)	// ILL_TAO_GUNKA
		mapannounce "com_d02_i", "古老塔奧群卡：是時候展現巨石怪的力量了嗎？", bc_map, 0x70DBDB;
	else	// ILL_WOOTAN_DEFENDER
		mapannounce "com_d02_i", "古老烏坦防禦者：看來在這些沒用的傢伙之後，我必須親自出馬了。", bc_map, 0x70DBDB;
	end;

OnBossDeath:
	killmonster "com_d02_i", "illusion_luanda::OnBossDeath";
	.boss_spawn = false;
	.cooldown = true;
	if (.swap == 0) {	// ILL_TAO_GUNKA
		mapannounce "com_d02_i", "古老塔奧群卡：巨石怪們，是時候蹲下並屏住呼吸了。", bc_map, 0x70DBDB;
		sleep 2000;
		mapannounce "com_d02_i", "古老塔奧群卡：當我們解開束縛並踏入這個世界時，我們將變得更強大。", bc_map, 0x70DBDB;
		sleep 2000;
		mapannounce "com_d02_i", "古老塔奧群卡：蹲下。當你蹲下時，你就是巨石怪。", bc_map, 0x70DBDB;
	}
	else {	// ILL_WOOTAN_DEFENDER
		mapannounce "com_d02_i", "古老烏坦防禦者：呃.. 真是相當危險的傷口..", bc_map, 0x70DBDB;
		sleep 2000;
		mapannounce "com_d02_i", "古老烏坦防禦者：能把我這烏坦族的戰士逼到這種地步。算你厲害。很遺憾，但我現在要先退下了。", bc_map, 0x70DBDB;
		sleep 2000;
		mapannounce "com_d02_i", "古老烏坦防禦者：所有的烏坦族聽令！現在撤退，將一切託付給下一個吧。", bc_map, 0x70DBDB;
	}
	sleep 1000;
	.group = (.group == 4 ? 5 : 4);
	// all the mobs are killed and immediately replaced with new mobs
	for ( .@i = 0; .@i < 3; ++.@i ) {
		.@mob_id = 20270 + .@i + .swap;
		killmonster "com_d02_i", "illusion_luanda::OnMob" + .@mob_id;
	}

	.swap = .swap ^ 4;
	// swap 0 -> 4, 4 -> 0 (used for mob ID)

	callsub( S_SpawnInit,
		(20270 + .swap),
		(20271 + .swap),
		(20272 + .swap)
	);
	initnpctimer;
	end;

OnTimer7200000:
	.cooldown = false;
	stopnpctimer;
	end;

S_SpawnInit:
	.count = 0;
	monster "com_d02_i",0,0,"--ja--",getarg(0),50,"illusion_luanda::OnMob" + getarg(0);
	monster "com_d02_i",0,0,"--ja--",getarg(1),50,"illusion_luanda::OnMob" + getarg(1);
	monster "com_d02_i",0,0,"--ja--",getarg(2),50,"illusion_luanda::OnMob" + getarg(2);
	return;

OnInit:
	.group = 5;	// UMOB_GROUP_ID

	callsub( S_SpawnInit,
		20270,	// ILL_TRI_JOINT
		20271,	// ILL_STALACTIC_GOLEM
		20272	// ILL_MEGALITH
	);
	end;
}
