2@mir,1,1,0	script	Intruder_main	-1,{
	end;
OnEvent:
	enablenpc instance_npcname("Intruder_main");
	.@label$ = instance_npcname("Intruder_main") + "::OnMobDead";
	monster 'map_2mir$,107,73,"可疑的入侵者",3454,1, .@label$;	// HEART_HUNTER_N
	setunitdata $@mobid[0], UMOB_HP, 10000;
	monster 'map_2mir$, 98,73,"可疑的入侵者",3454,1, .@label$;
	setunitdata $@mobid[0], UMOB_HP, 10000;
	monster 'map_2mir$,101,83,"可疑的入侵者",3454,1, .@label$;
	setunitdata $@mobid[0], UMOB_HP, 10000;
	'intruder_count = 0;
	donpcevent instance_npcname("入侵者#hunter02") + "::OnStart";
	donpcevent instance_npcname("入侵者#hunter03") + "::OnStart";
	donpcevent instance_npcname("入侵者#hunter04") + "::OnStart";
	donpcevent instance_npcname("入侵者#hunter05") + "::OnStart";
	donpcevent instance_npcname("入侵者#hunter06") + "::OnStart";
	end;
OnMobDead:
	'intruder_count++;
	switch( 'intruder_count ) {
	case 1:
		donpcevent instance_npcname("入侵者#hunter02") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter05") + "::OnLeave";
		break;
	case 2:
		donpcevent instance_npcname("入侵者#hunter03") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter06") + "::OnLeave";
		break;
	case 3:
		donpcevent instance_npcname("入侵者#hunter04") + "::OnLeave";
		break;
	}
	sleep 100;
	// it seems there is a slight delay

	switch( 'intruder_count ) {
	case 3:
		mapannounce 'map_2mir$, "兄弟們！我們離開這裡！", bc_map,0xDDDDDD;
		// display the effect a second time
		donpcevent instance_npcname("入侵者#hunter02") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter03") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter04") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter05") + "::OnLeave";
		donpcevent instance_npcname("入侵者#hunter06") + "::OnLeave";

		disablenpc instance_npcname("入侵者#hunter02");
		disablenpc instance_npcname("入侵者#hunter03");
		disablenpc instance_npcname("入侵者#hunter04");
		disablenpc instance_npcname("入侵者#hunter05");
		disablenpc instance_npcname("入侵者#hunter06");
		break;
	default:
		break;
	}
	end;
}
