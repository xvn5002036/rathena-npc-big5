1@mir,103,80,3	script	芬里爾#boss1a	4_F_FENRIR,2,2,{
	end;

	'iris_name$ = instance_npcname( "伊雷絲#" + strnpcinfo(2) );
	enablenpc 'fenrir_name$;
	enablenpc 'iris_name$;
	end;

	if ('touch_mob == 0) {
		npctalk "芬里爾：做得好！現在看妳的了，伊雷絲！", 'fenrir_name$;
		'touch_mob = 1;
		initnpctimer;
	}
	end;

	end;

	npctalk "伊雷絲：麻痺護符！去吧！！", 'iris_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "呃啊啊啊---！！";
	else
		unittalk 'boss_id, "畢尤：什麼...！！";
	end;
OnTimer5000:
	npctalk "伊雷絲：成功了！", 'iris_name$;
	end;
OnTimer7000:
	npctalk "芬里爾：那麼換我了。~大喝~", 'fenrir_name$;
	specialeffect EF_BEGINSPELL,AREA, 'fenrir_name$;
	specialeffect EF_TETRACASTING,AREA, 'fenrir_name$;
	progressbar_npc "000000",3;
	end;
OnTimer10000:
	npctalk "芬里爾：接招！！", 'fenrir_name$;
	specialeffect EF_SUI_EXPLOSION,AREA, 'fenrir_name$;
	specialeffect EF_LORD,AREA, 'fenrir_name$;
	specialeffect EF_FLAMELAUNCHER,AREA, 'fenrir_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "~尖叫~";
	else
		unittalk 'boss_id, "畢尤：~呻吟~ 比想像中還痛...！";
	getunitdata 'boss_id, .@data;
	if (.@data[UMOB_HP] > 0) {
		.@damage = rand(700,1300) * 1000;
		if (.@damage >= .@data[UMOB_HP])
			.@mob_hp = 0;
		else
			.@mob_hp = .@data[UMOB_HP] - .@damage;
		setunitdata 'boss_id, UMOB_HP, .@mob_hp;
	}
	end;
OnTimer13000:
	npctalk "芬里爾：雖然不完美，但我認為我們已經造成了相當大的傷害！", 'fenrir_name$;
	end;
OnTimer14500:
	npctalk "芬里爾：我消耗了太多魔力。我先稍微休息一下。接下來就交給你了！", 'fenrir_name$;
	end;
OnTimer16500:
	donpcevent instance_npcname("eq#mir3") + "::OnStart";
OnStop:
	stopnpctimer;
	disablenpc();
	disablenpc instance_npcname( "伊雷絲#" + strnpcinfo(2) );
	'touch_mob = 0;
	'random_letter$ = "";
	end;
}
