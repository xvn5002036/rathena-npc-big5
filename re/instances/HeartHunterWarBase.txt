	else
		.@md_name$ = "Heart Hunter War Base 2";

		close;

		next;
		mes "^ff0000當您的進度與隊長或隊員不同時會發生這種情況。在這種情況下，請尋找其他能和您前往相同地點的隊員。^000000";
		next;
		mes "^ff0000如果副本尚未建立為作戰區域，也是無法進入的。^000000";
		close;
	case IE_OTHER:
		mes "發生了未知的錯誤。";
		close;
	case IE_OK:
		mapannounce "ein_fild04", "" + getpartyname( getcharid(1) ) + " 隊伍的 " + strcharinfo(0) + " ，正在進入 " + .@md_name$ + "。", bc_map, 0xFF99;

			mes "[尤里安]";
			mes "聽起來不錯。你就像是外籍兵團，所以在行動時待在我身邊吧。";
		}
		else {
			mes "[尤里安]";
			mes "嗯，你還沒有收到簡報嗎？因為我們沒有足夠的時間，讓我簡單說明一下。";
			next;
			mes "[尤里安]";
			mes "我們沒有足夠的兵力來攻擊基地。";
			next;
			mes "[尤里安]";
			mes "為了保持機動性，我們放棄了重型槍械。既然我們的首要任務是在不造成太大破壞的情況下佔領基地，大部分的爆炸性武器也都被排除了。";
			next;
			mes "[尤里安]";
			mes "如果有必要，我們可以利用戰場的地形特徵來補強火力...但必須盡可能降到最低。";
			next;
			mes "[尤里安]";
			mes "無論如何，大部分的行動必須是以殺傷人員為主，並盡可能保留基地設施。";
			next;
			mes "[尤里安]";
			mes "我們的先遣部隊已經確保了南部基地的邊界地帶。我們將走那條路線前往中心區域。每一支突擊隊都會沿著各自的路線前進。";
			next;
			mes "[尤里安]";
			mes "你就像是外籍兵團，所以在行動時待在我身邊。還有問題嗎？當然，就算你有我也不會回答的。";
		}
		next;
		setquest 12374;// Heart Hunter Base occupation
		mes "[尤里安]";
		mes "在那之後，聽從你的部隊長官指示。一旦通往基地的門開啟，去和那邊的突擊隊指揮官交談並開始行動。";

			next;
			mes "[尤里安]";
			mes "如果你準備好了，我會通知你發起攻擊的時間。";
			next;
			if (select( "我準備好了。", "我還需要一點時間。" ) == 2) {

			mes "在那之後，聽從你的部隊長官指示。一旦通往基地的門開啟，去和那邊的突擊隊指揮官交談並開始行動。";
			close;
		}
		mes "[尤里安]";
		mes "在我們佔領的基地外圍有一大片空地。我們在那裡建了一個訓練場，但設施還不夠完善。我需要能檢驗它的菁英人員。";
		next;
		mes "[尤里安]";
		mes "哦，這不是我第一次跟你說這件事嗎？如果你能去檢驗一下設施，我會很高興的。你覺得如何？";
		next;
		if (select( "交給我吧。", "下次再說..." ) == 2) {
			mes "[尤里安]";
			mes "我明白了。是我的要求太多了。";
			close;
		}
		break;
	}
	if (terra_gloria_main == 19)
		instance_create("Heart Hunter War Base 1");
	else
		instance_create("Heart Hunter War Base 2");
	mes "[尤里安]";
	mes "門很快就會開啟。一旦通往基地的門開啟，去和那邊的突擊隊指揮官交談並開始行動。";
	close;
}
