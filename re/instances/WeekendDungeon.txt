		case 1:

		mes "Zeny 不足，請帶夠 Zeny 再來。";
		mes "具體來說需要 " + F_InsertComma(.@zeny) + " Zeny。";
		close;
	}
	if (checkweight(23144,1) == 0) {
		mes "你攜帶了太多種類的物品。請先清理一下你的物品欄。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你的負重太高了，無法繼續。請清理物品欄以減輕負重。";
		close;
	}
	mes "確定要花費 " + F_InsertComma(.@zeny) + " Zeny 將 5 個 <ITEM>" + getitemname(.@item) + "<INFO>" + .@item + "</INFO></ITEM> 包裝成 <ITEM>" + getitemname(.@box) + "<INFO>" + .@box + "</INFO></ITEM> 嗎？";
	next;
	if (select( "是", "否" ) == 2) {
		mes "[禮品包裝員]";
		mes "需要包裝服務時隨時歡迎再來！";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
}
