moro_cav,50,64,5	script	黃色種子#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000等級低於 160 級的角色無法進入此副本。^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000你必須先見過流浪的老人才能進入這個副本。^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000你必須先建立或加入一個至少 1 人的隊伍才能進入此副本。^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 mins to re-enter
			mes "^ff0000在上次進入後的 23 小時內，無法再次進入此記憶迷宮。^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ff在上次進入後的 23 小時內，無法再次進入此記憶迷宮。^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Memorial Dungeon: Isle of Bios
		erasequest 15007;// Memorial Dungeon: Isle of Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Memorial Dungeon: Isle of Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "建立記憶迷宮。";
	switch( select( .@create$, "進入拜歐斯島。", "取消。" ) ) {
	case 1:
		instance_create("Isle of Bios");
		mes "^ff0000已建立記憶迷宮。^000000";
		mes "^ff0000請點擊並再次按下進入。^000000";
		close;
	case 2:
		switch( instance_enter("Isle of Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000請注意，在上次進入後的 23 小時內，無法再次進入此記憶迷宮。^000000";
				next;
			}
			mes "^ff0000你的隊長尚未建立時空裂縫。^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", "隊伍 " + getpartyname( getcharid(1) ) + " 的隊員 " + strcharinfo(0) + " 正在進入拜歐斯島。",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Memorial Dungeon: Isle of Bios
				setquest 15007;// Memorial Dungeon: Isle of Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;
OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
