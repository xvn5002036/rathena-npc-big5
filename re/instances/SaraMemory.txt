dali,139,118,4	script	次元移動機#sara	10007,{
	if (BaseLevel < 99) {
		mes "[冒險家 雷昂]";
		mes "嘿！你還太小，不能玩那個！等你至少 ^0000FF99 級^000000 再來吧！";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[冒險家 雷昂]";
		mes "在你開始玩那個之前；為什麼不過來跟我談談呢？";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Sara's Memories";
		if (!.@party_id) {
			mes "[冒險家 雷昂]";
			mes "在進入之前，你需要組織一個隊伍！不知道怎麼做嗎？輸入 /organize 你的隊伍名稱。";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "啟動次元移動機。:使用次元移動機。:取消。";
		else
			set .@menu$, ":使用次元移動機。:取消。";
		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "隊伍名稱: "+.@p_name$;
				mes "隊長: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- 預約失敗！";
				close;
			}
			mes "^FF0000次元移動機順利啟動。使用該裝置進入時空裂縫。^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "發生未知錯誤。";
				close;
			case IE_NOINSTANCE:
				mes "記憶迷宮 "+.@md_name$+" 不存在。";
				mes "隊長尚未建立副本。";
				close;
			case IE_NOMEMBER:
				mes "只有註冊的成員才能進入記憶迷宮 "+.@md_name$+"。";
				close;
			case IE_OK:
				mapannounce "dali", "隊伍 "+.@p_name$+" 的成員 "+strcharinfo(0)+" 進入了記憶迷宮 "+.@md_name$+"。",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000隊長尚未啟動次元移動機。^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[冒險家 雷昂]";
		mes "我想在距離你上次進入裂縫大約 20 小時後，它就會足夠穩定，讓你再次穿越。";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ 莎拉的記憶 ] 冷卻時間已結束。現在可以再次使用次元移動機。^000000";
		close;
	}
}
