ma_scene01,174,179,4	script	塔浩	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你攜帶太多物品了。請整理一下再來吧。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你攜帶的重量太重了。請減輕重量後再來吧。";
		close;
	}
	if (BaseLevel < 140) {
		mes "[塔浩]";
		mes "嘿。這個地方對你來說太危險了。你無法保護自己。快點離開這裡，回到村莊去吧！";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[塔浩]";
		mes "我們都會被那隻巴庫暴龍殺死的。啊，我們必須消滅牠，但可惜我們還沒準備好。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "巴庫暴龍之湖";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[塔浩]";
		mes "沒有同伴可以和你一起去嗎？即使你想單獨前往，也請先組建隊伍再來。";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[塔浩]";
			mes "你是討伐巴庫暴龍的隊長嗎？路況不好，所以我們必須用繩子爬下懸崖，這樣可以嗎？";
			next;
			set .@i, select("請編織一條繩子。:我現在要下去了。:取消。");
		} else {
			mes "[塔浩]";
			mes "你是來討伐巴庫暴龍的。你們的隊長必須幫我編織下去用的繩子。";
			next;
			set .@i, select(":我現在要下去了。:取消。");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[塔浩]";
				mes "噢，繩子掉下去了。我必須重新做一條。";
				close;
			}
			mes "[塔浩]";
			mes "我現在正在編織，等我完成了你就可以下去了。";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[塔浩]";
				mes "發生了未知的錯誤。";
				close;
			case IE_NOINSTANCE:
				mes "[塔浩]";
				mes "繩子還沒編織好。請稍等一下。";
				close;
			case IE_NOMEMBER:
				mes "[塔浩]";
				mes "平民無法進入。";
				close;
			case IE_OK:
				mapannounce "ma_scene01","隊伍 "+getpartyname(.@party_id)+" 的成員 "+strcharinfo(0)+" 正在進入副本 "+.@md_name$+"。",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278; //warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[塔浩]";
		mes "用來前往湖泊的繩子已經斷了。我們現在正在清理周圍環境，請過一段時間再來。";
		close;
	} else if (.@baku_time == 2) {
		mes "[塔浩]";
		mes "這裡已經清理完畢。現在取得繩子後，你可以再次下去了。";
		erasequest 12278;
		close;
	}
}
