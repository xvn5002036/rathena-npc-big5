1@slw,155,191,3	script	保全裝置#switch01	CLEAR_NPC,{
	if (is_party_leader() == false)
		end;
	.@number = atoi( replacestr( strnpcinfo(2), "switch", "" ) );
	.@type = (.@number < 10) ? 0 : 1;
	.@npc_num = (.@number < 10) ? .@number : (.@number-10);
	if ('device_state[.@type] & .@npc_num) {
		mes "已經啟動了。";
		close;
	}
	callsub S_Code, .@string$, .@answer;
	mes "上面寫有安全代碼。";
	mes "[" + .@string$ + "]";
	mes "你要輸入什麼？";
	next;
	if (select( "^ff0000" + .@string$ + "^000000", "^4d4dff" + .@string$ + "^000000" ) != .@answer) {
		'device_state[.@type] = 0;
		mes "裝置已初始化。";
		close;
	}
	if (is_party_leader() == false)
		end;
	'device_state[.@type] = 'device_state[.@type] | .@npc_num;
	if ('device_state[.@type] != 3) {
		mes "操作了保全裝置。";
		mes "似乎還有一個保全裝置。";
		close;
	}
	mes "操作了保全裝置。";
	mes "門現在啟動了。";
	switch( .@type ) {
	case 0:
		enablenpc instance_npcname("#slwwarp01");
		if (instance_live_info(ILI_NAME) == "Werner Laboratory central room#1")	// terra_gloria_main == 21
			enablenpc instance_npcname("艾森·魏格納#werner03");
		else {
			enablenpc instance_npcname("守衛#gard05");
			enablenpc instance_npcname("守衛#gard06");
			enablenpc instance_npcname("守衛#gard07");
		}
		break;
	case 1:
		if (instance_live_info(ILI_NAME) == "Werner Laboratory central room#1")
			enablenpc instance_npcname("艾森·魏格納#werner04");
		else
			enablenpc instance_npcname("中央入口#door");
		enablenpc instance_npcname("#slwwarp02");
		break;
	}
	close;
S_Code:
	setarray .@sample$[0],
		"純粹", 2,
		"永恆", 2,
		"黎明", 2,
		"玫瑰", 1,
		"矛盾", 1,
		"喜悅", 1,
		"海洋", 1,
		"歸途", 1,
		"孤獨", 1,
		"光輝", 1,
		"黃昏", 1;
	.@size = getarraysize(.@sample$) / 2;
	.@r = rand(.@size) * 2;
	set getarg(0), .@sample$[.@r];
	set getarg(1), .@sample$[.@r+1];
	return;
