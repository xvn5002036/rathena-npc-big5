ma_fild02,312,317,5	script	守衛#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[守衛]";
		mes "等級未滿 ^ff0000130 級^000000 的人";
		mes "無法進入此地。";
		mes "這個地方很危險。請回去吧。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"布瓦亞洞穴";

	if (!.@party_id) {
		mes "[守衛]";
		mes "布瓦亞非常危險。";
		mes "請在與其他人組成隊伍後";
		mes "再來吧。";
		close;
	}
	if (is_party_leader() == true) {
		mes "[守衛]";
		mes "這個地方很危險。請回去吧。";
		mes "如果你在這裡沒有任何事情，請回去吧。";
		next;
		while(1) {
			switch(select("為什麼這麼危險？:我是來討伐布瓦亞的。:結束對話。")) {
			case 1:
				mes "[守衛]";
				mes "最近，發生了一起";
				mes "村民失蹤的案件。";
				mes "在調查了痕跡之後，";
				mes "我們發現巨大的怪物布瓦亞";
				mes "就是罪魁禍首。";
				next;
				mes "[守衛]";
				mes "我們已經派遣士兵和傭兵去";
				mes "討伐布瓦亞，但從未";
				mes "完全成功過。";
				next;
				mes "[守衛]";
				mes "時至今日，布瓦亞仍然在綁架";
				mes "村民。";
				mes "請召集一些人來除掉布瓦亞。";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[守衛]";
					mes "布瓦亞還躲藏著。";
					mes "即使你現在進去，布瓦亞也不會出來。";
					mes "請稍後再來。";
					close;
				} else {
					erasequest 4229; // fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[守衛]";
					mes "隊伍名稱是... "+getpartyname(.@party_id)+"。";
					mes "隊長是... "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 現在無法開啟。";
					mes "請稍後再試。";
					close;
				}
				mes "[守衛]";
				mes "我將為你們開啟通往 ^0000ff"+.@md_name$+"^000000 的通道。";
				mes "祝你們好運。";
				close;
			case 3:
				mes "[守衛]";
				mes "這個地方很危險。你不能進入。";
				close;
			}
		}
	} else {
		mes "[守衛]";
		mes "讓我跟隊長說話。";
		mes "請叫隊長過來。";
		close;
	}
}

ma_fild02,315,323,0	script	洞穴入口#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[守衛]";
		mes "等級未滿 ^ff0000130 級^000000 的人";
		mes "無法進入此地。";
		mes "這個地方很危險。請回去吧。";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[守衛]";
		mes "布瓦亞還躲藏著。";
		mes "即使你現在進去，布瓦亞也不會出來。";
		mes "請稍後再來。";
		close;
	} else {
		erasequest 4229; // fall through
	}
	switch(select("進入。:回頭。")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[守衛]";
			mes "噢，現在不是個好時機。";
			mes "請稍後再試。";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[守衛]";
			mes "這個地方很危險。請不要進入。";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+" 隊伍的 "+strcharinfo(0)+" 已經開始在布瓦亞洞穴討伐布瓦亞了。",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229; //warp "1@ma_c",35,57;
			end;
		default:
			mes "- 由於不明力量的影響而無法進入。 -";
			close;
		}
	case 2:
		mes "[守衛]";
		mes "你必須珍惜你的生命。";
		mes "行動前最好三思而後行。";
		close;
	}
}
