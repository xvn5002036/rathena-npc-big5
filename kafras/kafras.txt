aldeba_in,96,181,4	script	卡普拉服務人員	113,{
	cutin "kafra_05",2;
	mes "[卡普拉 蕾拉]";
	mes "嗯...？";
	mes "哦，歡迎來到";
	mes "卡普拉公司總部。";
	mes "請問有什麼需要幫忙的嗎？";
	next;
	switch(select("儲存位置:使用倉庫:租用手推車:取消")) {
	case 1:
		mes "[卡普拉 蕾拉]";
		mes "您的重生點";
		mes "已經儲存在這裡，";
		mes "卡普拉公司總部內。謝謝您。";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[卡普拉 蕾拉]";
		mes "請多加利用在";
		mes "米德加爾特各地提供的";
		mes "卡普拉服務。感謝您";
		mes "參訪卡普拉總部。";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[卡普拉 蕾拉]";
			mes "不好意思，看來";
			mes "您沒有 20 zeny 來支付";
			mes "倉庫的使用費...";
			close3;
		}
		mes "[卡普拉 蕾拉]";
		mes "雖然這個設施是";
		mes "專門為卡普拉員工的";
		mes "訓練和行政功能而設的，";
		mes "但我還是為您開啟倉庫吧。";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[卡普拉 蕾拉]";
		mes "未來如果您希望";
		mes "使用任何卡普拉服務，";
		mes "請向值班的卡普拉服務人員詢問。";
		mes "感謝您的惠顧。";
		callfunc("F_CheckKafCode");
		//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "[卡普拉 蕾拉]";
		mes "非常抱歉，但我現在";
		mes "沒有值班。如果可以的話我很樂意";
		mes "協助您，但實際上我並沒有";
		mes "可用的手推車。";
		mes "您要不要請其他卡普拉服務人員協助呢？";
		next;
		mes "[卡普拉 蕾拉]";
		mes "卡普拉服務人員";
		mes "駐紮在米德加爾特大陸";
		mes "各地，";
		mes "在艾爾帕蘭外面應該可以找到很多。";
		close2;
		break;
	case 4:
		mes "[卡普拉 蕾拉]";
		mes "^666666*呼...*^000000";
		mes "太好了，因為我";
		mes "現在其實在休息。";
		mes "選擇「取消」是";
		mes "您明智的決定。";
		close2;
		break;
	}
	cutin "",255;
	end;
}
