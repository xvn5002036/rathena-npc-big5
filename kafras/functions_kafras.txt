//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage";
// Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);
// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[卡普拉服務人員]";
		mes "歡迎來到";
		mes "卡普拉公司。";
		mes "卡普拉服務";
		mes "永遠與您同在。";
		mes "請問需要什麼服務？";
		break;
	case 1:// Niflheim specific message
		mes "[卡普拉服務人員]";
		mes "^666666歡...歡迎";
		mes "來...來到卡...卡普拉";
		mes "公...公司...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[卡普拉服務人員]";
		mes "歡迎。^ff0000" + getguildname( getcharid(2) ) + "^000000 公會成員。";
		mes "卡普拉公司將永遠陪伴著您。";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[卡普拉服務人員]";
		mes "那麼，您是從遙遠的地方來研究我們的文化，還是只是來觀光的呢？";
		mes "無論如何，何不稍作停留？";
		mes "這裡的空氣永遠充滿著";
		mes "令人愉悅的野花香。";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[卡普拉服務人員]";
		mes "我們有許多卡普拉";
		mes "服務據點，讓您永遠";
		mes "感覺賓至如歸。";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "使用倉庫", "使用公會倉庫", "租用手推車", "使用傳送服務", "取消";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"儲存位置","使用倉庫","取消"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"使用倉庫","取消"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"儲存位置","使用倉庫","租用手推車","查看其他資訊","取消"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport.
			// (Common)
			case 5:	setarray .@K_Menu0$[0],"使用倉庫","租用手推車","查看其他資訊","取消"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"使用倉庫","查看其他資訊","取消"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"儲存位置","使用倉庫","租用手推車","取消";
			break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"儲存位置","使用倉庫","查看其他資訊","取消"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"使用倉庫","租用手推車","使用傳送服務","查看其他資訊","取消"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"使用倉庫","儲存位置","租用手推車","取消"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"儲存位置","使用倉庫","使用傳送服務","租用手推車","查看其他資訊","取消"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "儲存位置")
			return;
		else if (.@K_Menu0$[.@j] == "使用倉庫") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "使用傳送服務") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[卡普拉服務人員]";
				mes "因為^FF0000限制";
				mes "運輸協議^000000的關係，";
				mes "卡普拉公司無法";
				mes "在秀發茲發德共和國";
				mes "提供傳送服務。";
				next;
				mes "[卡普拉服務人員]";
				mes "請您改搭";
				mes "飛空艇服務。";
				mes "感謝您的";
				mes "體諒與合作。";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "租用手推車") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "查看其他資訊") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "取消") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "使用公會倉庫") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[卡普拉服務人員]";
			mes "很抱歉，但是目前有其他公會成員正在使用公會倉庫";
			mes "請等那個人使用完畢後再來。";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[卡普拉服務人員]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666抱...抱...抱歉，";
			mes "你...你是一名";
			mes "初...初心者...需要";
			mes "基...基本技能...等級 6...^000000";
			return;
		}
		mes "很抱歉，但是您";
		mes "需要初心者的";
		mes "基本技能等級 6 才能";
		mes "使用倉庫服務。";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[卡普拉服務人員]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Z...Zeny...";
					mes "更...更多的 Zeny...！";
					mes "需...需要 150... zeny...";
					mes "呃！拿...拿血來換?！^000000";
					return;
				}
				// Standard Message
				mes "很抱歉，但是您沒有";
				mes "足夠的 Zeny 來使用";
				mes "倉庫服務。我們的";
				mes "倉庫使用費是 " + .@fee + " zeny。";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[卡普拉服務人員]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666感...謝...您的...使用...^000000";
		}
		// Normal message
		else {
			mes "[卡普拉服務人員]";
			mes "來，讓我為您";
			mes "開啟倉庫。";
			mes "感謝您使用";
			mes "卡普拉服務。";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[卡普拉服務人員]";
	mes "請選擇";
	mes "您的目的地。";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "取消")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[卡普拉服務人員]";
			mes "很抱歉，但是您沒有";
			mes "足夠的 Zeny 使用傳送";
			mes "服務。傳送";
			mes "到 " + @wrpD$[.@j] + " 的費用是 " + @wrpP[.@j] + " zeny。";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "艾爾帕蘭") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "艾爾貝塔") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "克魔島") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "發樂斯燈塔") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "吉芬") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "伊斯魯得島") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "朱諾") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "廢棄礦坑") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "夢羅克") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "獸人地下洞穴") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "斐揚") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "普隆德拉") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "汶巴拉") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[卡普拉服務人員]";
		mes "很抱歉，但是";
		mes "租用手推車服務";
		if (checkre(0) == true)
			mes "只提供給商人系職業。";
		else {
			mes "只提供給商人、";
			mes "鐵匠、神工匠、";
			mes "鍊金術師和創造者。";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[卡普拉服務人員]";
		mes "您已經裝備了";
		mes "一台手推車。";
		mes "很遺憾，我們不能";
		mes "同時租借超過一台";
		mes "給同一個顧客。";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[卡普拉服務人員]";
		mes "您必須在學會「手推車使用」技能後才能租用手推車。";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[卡普拉服務人員]";
		mes "租用手推車的";
		mes "費用是 " + .@rental_fee + " zeny。您";
		mes "想要租用一台手推車嗎？";
		next;
		if (select("租用手推車。:取消") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[卡普拉服務人員]";
			mes "很抱歉，但是您";
			mes "沒有足夠的";
			mes "zeny 來支付 " + .@rental_fee + " zeny";
			mes "的手推車租用費。";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

			mes "[卡普拉服務人員]";
			mes "您可以在艾爾帕蘭的卡普拉總部";

			mes "獎勵。請多加利用我們";
			mes "便利的服務，以體驗我們獎勵計畫的優惠。";
			if (getarg(0) == 1)
				return;
			next;
			break;
		case 1:// Storage Password Service
			callfunc("F_SetKafCode","[卡普拉服務人員]","卡普拉服務");
		case 2:// Kafra Employee Locations
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 1,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			next;
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 2,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			break;
		default:
			return;
		}
	}
}

		mes "卡普拉服務。";
	} else if (.@type == 0 || .@type == 5) {// Generic End
		mes "我們卡普拉公司，";
		mes "一直致力於為您提供最好的服務。我們希望我們能滿足您的冒險需求和卓越的標準。";
	} else if (.@type == 1) {// Niflheim End

		mes "^666666卡...卡普拉永...不...";
		mes "死...死死死。與...";
		mes "與...您同同同在?^000000";
	} else if (.@type == 2) {// RockRidge Save
		mes "已儲存。";
		mes "感謝您的惠顧。";
	}
	close3;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "請輸入您的倉庫密碼：";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "倉庫密碼錯誤。";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " 很榮幸為您推出一項新服務：";
		mes "額外的密碼倉庫保護。";
		next;
		if (select( "設定新密碼 -> 5000z", "取消" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "您的倉庫已受密碼保護。您現在想做什麼？";
	next;
	switch( select( "更改舊密碼 -> 5000z", "移除倉庫密碼 -> 1000z", "取消" ) ) {
	case 1:
		mes .@npc_name$;
		mes "首先，請輸入您的^0000FF舊密碼^000000。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "密碼錯誤。您無法設定新密碼。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "請在移除前輸入您的密碼。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "密碼尚未被移除。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "您的 zeny 不足。";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "您已成功清除您的倉庫密碼。";
			mes "感謝您使用 " + .@comp_name$ + "。";
			emotion ET_THANKS;
		} else {
			mes "密碼錯誤。我們不會退還您的 1000z。";
			mes "請下次輸入正確的密碼。";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "現在輸入您的^FF0000新密碼^000000，以保護您的倉庫免受小偷的光顧。";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "密碼尚未更改。";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "您的 zeny 不足。";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "您已用一組秘密密碼保護了您的倉庫。";
	mes "感謝您使用 " + .@comp_name$ + "。";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "請輸入一個數字 1000~10000000：";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "您不能使用這麼長的密碼。";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "您不應該使用這麼短的密碼。";
		return 0;
	}
	return .@code_;
}
