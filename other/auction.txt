function	script	F_AuctionWarper	{
	mes "[拍賣場嚮導]";
	mes "你好，你想";
	mes "進入";
	mes "拍賣場嗎？";
	next;
	if (select("是:否") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "[拍賣場嚮導]";
			mes "太好了！那麼，";
			mes "希望你玩得開心";
			mes "並享受拍賣的樂趣?";
		} else {
			mes "[拍賣場嚮導]";
			mes "好好享受拍賣吧。";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "[拍賣場嚮導]";
	mes "那好吧，";
	mes "再見囉。";
	mes "如果你改變了";
	mes "心意，請隨時來";
	mes "享受拍賣的樂趣?";
	close;
}
