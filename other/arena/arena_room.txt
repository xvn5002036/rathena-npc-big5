	mes "你可以用 ^3131FF"+ .@arena_point +" 點競技場點數兌換 "+ .@amount +" 個 "+ getitemname(.@item_id) +"^000000。";
	mes "如果想取消，請輸入 0。否則，請輸入你想花費競技場點數的 ^3131FF次數^000000。";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[紀芙]";
		mes "你已取消請求。";
		close;
	} else if (.@reward > 1500) {
		mes "[紀芙]";
		mes "你已超出最大容量。";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[紀芙]";
		mes "你的競技場點數不足。";
		mes "請檢查你擁有的競技場點數總額。";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[紀芙]";
	mes "謝謝，請再次光臨。";
	close;
}
