-	script	傭兵物品商人#dummy::MercMerchant	-1,{
	mes "[傭兵物品商人]";
	mes "你好，我販售";
	mes "傭兵可以使用的物品。";
	mes "請問你有在找什麼";
	mes "特定的東西嗎？";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[傭兵物品商人]";
	mes getitemname(.@item[.@m]);
	mes "每個售價 " + .@str_cost$ + " zeny。";
	mes "你想要多少個？";
	next;
	input .@input;
	if (!.@input) {
		mes "[傭兵物品商人]";
		mes "你改變主意了？";
		mes "好吧，只要你想";
		mes "購買任何傭兵藥水，";
		mes "隨時歡迎回來找我。";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[傭兵物品商人]";
		mes "很抱歉，但是你";
		mes "一次最多只能購買";
		mes "10,000 瓶藥水。";
		mes "請輸入一個";
		mes "1 到 10,000 的數字。";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[傭兵物品商人]";
		mes "很抱歉，但是你沒有";
		mes "足夠的 zeny 來購買";
		mes "這麼多藥水。嗯，當你準備好";
		mes "為你的傭兵";
		mes "購買東西時，";
		mes "我會在這裡的。";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[傭兵物品商人]";
		mes "如果我給你這麼多";
		mes "藥水，你會";
		mes "沒辦法帶走它們。";
		mes "請在清理出";
		mes "物品欄的一些空間後";
		mes "再回來。";
		close;
	}
	mes "[傭兵物品商人]";
	if (.@input == 1) 
		mes "這是你的 " + getitemname(.@item[.@m]) + "。";
	else {
		mes "給你，這正是";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + "。";
	}
	mes "謝謝你，當你的傭兵";
	mes "需要更多藥水時，";
	mes "請再來光臨。";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

// Mercenary Switch NPCs
//============================================================
-	script	傭兵開關	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[檢查員]";
	mes "請輸入密碼。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[檢查員]";
	if (.@i == -2)
		mes "錯誤。";
	else if (.@i == 0)
		mes "密碼錯誤。";
	else if (.@i == 1) {
		mes "請選擇。";
		next;
		switch( select( "關閉傭兵 NPC","開啟傭兵 NPC" ) ) {
		case 1:
			mes "NPC 已關閉。";
			disablenpc "傭兵管理員#" + .@name$[.@type];
			disablenpc "傭兵物品商人#" + .@name$[.@type];
			break;
		case 2:
			mes "NPC 已開啟。";
			enablenpc "傭兵管理員#" + .@name$[.@type];
			enablenpc "傭兵物品商人#" + .@name$[.@type];
			break;
		}
	}
	close;
}
